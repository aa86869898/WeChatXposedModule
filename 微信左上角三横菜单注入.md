java
package com.xposed.wxhome;

import android.app.Activity;
import android.os.Build;
import android.util.Log;
import android.view.View;

import de.robv.android.xposed.XposedHelpers;

/**
 * 唯一职责：判断某个 Activity 当前是否正处在"微信主页且未被聊天页覆盖"的状态。
 * 任一环节取不到 → 判定为 false（保守：宁可不显示，也绝不在错误页面弹出）。
 */
public final class WxHomeGate {

    private static final String TAG = "WxHomeGate";

    private WxHomeGate() {}

    public static boolean isWeChatHome(Activity activity, boolean onlyMainTab) {
        try {
            if (activity == null || activity.isFinishing()) return false;
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.JELLY_BEAN_MR1 && activity.isDestroyed()) return false;

            // 1) 必须是 LauncherUI（其它窗口：ChattingUI / SearchBarUI / 设置页… 全部直接排除）
            if (!WxClass.LAUNCHER_UI.equals(activity.getClass().getName())) return false;

            // 2) 取 HomeUI -> MainTabUI
            Object homeUI = XposedHelpers.getObjectField(activity, WxClass.F_HOME_UI);
            if (homeUI == null) return false;
            Object tabUI = XposedHelpers.callMethod(homeUI, WxClass.M_GET_TAB_UI);
            if (tabUI == null) return false;

            // 3) 聊天页在前台 → 不显示（用的是微信自己的判定逻辑）
            if (isChattingForeground(activity)) return false;

            // 4) 当前 tab 判定
            int tab = currentTabIndex(activity, tabUI);
            if (tab < 0) return false;                     // MainTabUI 尚未初始化
            if (onlyMainTab && tab != 0) return false;     // 0=会话列表

            // 5) 当前 tab 的 Fragment 必须真的可见
            return isCurrentFragmentVisible(tabUI);
        } catch (Throwable t) {
            if (WxConfig.debugLog) Log.w(TAG, "gate error: " + t);
            return false;
        }
    }

    /** NewChattingTabUI.m() -> o != null && o.f.i（微信 onCreateOptionsMenu 用的同一个判定） */
    private static boolean isChattingForeground(Activity activity) {
        Object chattingTab = XposedHelpers.getObjectField(activity, WxClass.F_CHATTING_TAB);
        if (chattingTab == null) return true;              // 还没准备好 → 当作不在主页
        Object r = XposedHelpers.callMethod(chattingTab, WxClass.M_CHAT_FOREGROUND);
        return r instanceof Boolean && (Boolean) r;
    }

    /** MainTabUI.e，失败时回退到 LauncherUI.getCurrentTabIndex() */
    private static int currentTabIndex(Activity activity, Object tabUI) {
        try {
            return XposedHelpers.getIntField(tabUI, WxClass.F_TAB_INDEX);
        } catch (Throwable ignored) {
            try {
                Object r = XposedHelpers.callStaticMethod(activity.getClass(), "getCurrentTabIndex");
                return r instanceof Integer ? (Integer) r : -1;
            } catch (Throwable t) {
                return -1;
            }
        }
    }

    /** MainTabUI.g() 取当前 Fragment，判断其 View 是否 attached 且 VISIBLE */
    private static boolean isCurrentFragmentVisible(Object tabUI) {
        Object fragment = XposedHelpers.callMethod(tabUI, WxClass.M_GET_CUR_FRAG);
        if (fragment == null) return false;
        Object view = XposedHelpers.callMethod(fragment, "getView");
        return view instanceof View && ((View) view).isShown();
    }
}

/** 圆角半透明底 + 三条横线，纯 Canvas 绘制，不依赖任何 drawable 资源 */
public class CornerIconView extends View {

    private final Paint bgPaint  = new Paint(Paint.ANTI_ALIAS_FLAG);
    private final Paint barPaint = new Paint(Paint.ANTI_ALIAS_FLAG);
    private final RectF  rect     = new RectF();

    private int bgColor   = 0x59000000;
    private int barColor  = Color.WHITE;

    public CornerIconView(Context context) {
        super(context);
        setClickable(true);
        setFocusable(true);
        bgPaint.setStyle(Paint.Style.FILL);
        barPaint.setStyle(Paint.Style.STROKE);
        barPaint.setStrokeCap(Paint.Cap.ROUND);
    }

    public void setIconColors(int bg, int bar) {
        this.bgColor = bg;
        this.barColor = bar;
        invalidate();
    }

    @Override
    protected void onDraw(Canvas canvas) {
        int w = getWidth(), h = getHeight();
        if (w <= 0 || h <= 0) return;

        float r = Math.min(w, h) / 2f;
        bgPaint.setColor(bgColor);
        rect.set(0, 0, w, h);
        canvas.drawRoundRect(rect, r, r, bgPaint);

        float barH   = Math.max(2f, h * 0.075f);
        float inset  = w * 0.28f;
        float cy     = h / 2f;
        float gap    = h * 0.23f;
        barPaint.setColor(barColor);
        barPaint.setStrokeWidth(barH);
        for (int i = -1; i <= 1; i++) {
            float y = cy + i * gap;
            canvas.drawLine(inset, y, w - inset, y, barPaint);
        }
    }

package com.xposed.wxhome;

import android.app.Activity;
import android.content.Intent;
import android.graphics.Insets;
import android.os.Build;
import android.os.Handler;
import android.os.Looper;
import android.util.Log;
import android.util.TypedValue;
import android.view.Gravity;
import android.view.View;
import android.view.ViewGroup;
import android.view.WindowInsets;
import android.widget.FrameLayout;

import java.lang.ref.WeakReference;

/**
 * 负责：
 *  1) 把按钮挂到 Activity 的 Window DecorView 左上角（状态栏下方 = 标题栏左上角）
 *  2) 依据 WxHomeGate 的结果做显隐
 *  3) 生命周期可控（start / stop / refresh / dispose）
 */
public final class HomeCornerButton {

    private static final String TAG     = "WxHomeCorner";
    private static final String VIEW_TAG = "wx_home_corner";

    private final WeakReference<Activity> ref;
    private final Handler handler = new Handler(Looper.getMainLooper());

    private View     button;
    private boolean  started;
    private int      retry;

    private final Runnable tick = new Runnable() {
        @Override public void run() {
            Activity a = ref.get();
            if (!started || a == null) return;
            sync(a);
            if (a.isFinishing() || (Build.VERSION.SDK_INT >= 17 && a.isDestroyed())) return; // 交给 onDestroy 清理
            handler.postDelayed(this, WxConfig.pollIntervalMs);
        }
    };

    public HomeCornerButton(Activity activity) {
        this.ref = new WeakReference<>(activity);
    }

    // ---------------- 生命周期 ----------------

    public void start() {
        if (started) { syncNow(); return; }
        started = true;
        retry = 0;
        handler.removeCallbacks(tick);
        handler.post(tick);
    }

    public void stop() {                 // Activity onPause：隐藏并停轮询
        started = false;
        handler.removeCallbacks(tick);
        Activity a = ref.get();
        if (a != null && button != null) button.setVisibility(View.GONE);
    }

    public void dispose() {               // Activity onDestroy：彻底移除
        stop();
        View v = button;
        button = null;
        if (v != null && v.getParent() instanceof ViewGroup) {
            ((ViewGroup) v.getParent()).removeView(v);
        }
    }

    public void refresh() {               // 事件驱动刷新（即时性）
        if (Looper.myLooper() == Looper.getMainLooper()) syncNow();
        else handler.post(this::syncNow);
    }

    private void syncNow() {
        Activity a = ref.get();
        if (a != null) sync(a);
    }

    // ---------------- 核心同步 ----------------

    private void sync(Activity a) {
        boolean want = WxHomeGate.isWeChatHome(a, WxConfig.onlyMainTab);

        if (want && button == null) attach(a);

        View v = button;
        if (v == null) return;

        int target = want ? View.VISIBLE : View.GONE;
        if (v.getVisibility() != target) {
            v.setVisibility(target);
            if (want) v.bringToFront();
        }
    }

    // ---------------- 挂载 + 精确定位 ----------------

    private void attach(Activity a) {
        ViewGroup host = findHost(a);
        if (host == null) { retryLater(a); return; }

        boolean onDecor   = isDecorHost(a, host);
        int    size       = dp(a, WxConfig.iconSizeDp);
        int    abHeight   = actionBarSize(a);

        int top = resolveTopInset(a)                 // 状态栏/刘海高度
                + (onDecor ? 0 : abHeight)           // 若退化为 content 根布局则补标题栏高度
                + dp(a, WxConfig.extraTopOffsetDp)
                + Math.max(0, (abHeight - size) / 2);// 在标题栏内垂直居中

        CornerIconView v = new CornerIconView(a);
        v.setTag(VIEW_TAG);
        v.setContentDescription("主页按钮");
        v.setOnClickListener(new View.OnClickListener() {
            @Override public void onClick(View view) { onCornerClick(a); }
        });

        FrameLayout.LayoutParams lp = new FrameLayout.LayoutParams(
                size, size, Gravity.TOP | Gravity.START);
        lp.leftMargin  = dp(a, WxConfig.marginLeftDp) + dp(a, WxConfig.extraLeftOffsetDp);
        lp.topMargin   = Math.max(0, top);

        host.addView(v, lp);
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) v.setElevation(dp(a, 8f));
        v.bringToFront();

        button = v;
        if (WxConfig.debugLog) Log.i(TAG, "attach on " + host.getClass().getName() + " topMargin=" + lp.topMargin);
    }

    private void retryLater(final Activity a) {
        if (++retry > 30) return;                      // 最多重试 ~3.6s
        handler.postDelayed(new Runnable() {
            @Override public void run() {
                if (button == null && a != null && !a.isFinishing()) attach(a);
            }
        }, 120L);
    }

    /** 优先 DecorView（能盖住标题栏）；异常时回退 android.R.id.content */
    private static ViewGroup findHost(Activity a) {
        if (a.getWindow() != null) {
            View decor = a.getWindow().getDecorView();
            if (decor instanceof ViewGroup) return (ViewGroup) decor;
        }
        View content = a.findViewById(android.R.id.content);
        return content instanceof ViewGroup ? (ViewGroup) content : null;
    }

    private static boolean isDecorHost(Activity a, ViewGroup host) {
        return a.getWindow() != null && a.getWindow().getDecorView() == host;
    }

    // ---------------- 点击回调（接你自己的业务） ----------------

    private void onCornerClick(Activity a) {
        String cls = WxConfig.clickActivityClassName;
        if (cls != null && cls.length() > 0) {
            try {
                Intent i = new Intent();
                i.setClassName(a, cls);                                   // 模块内 Activity，需 export="true"
                i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
                a.startActivity(i);
                return;
            } catch (Throwable t) {
                Log.w(TAG, "startActivity failed: " + t);                // 失败则退回调试面板
            }
        }
        DebugPanel.show(a);
    }

    // ---------------- 工具方法 ----------------

    private static int dp(Activity a, float dp) {
        return Math.round(TypedValue.applyDimension(
                TypedValue.COMPLEX_UNIT_DIP, dp, a.getResources().getDisplayMetrics()));
    }

    /** 主题 actionBarSize，失败回退 48dp */
    private static int actionBarSize(Activity a) {
        TypedValue tv = new TypedValue();
        try {
            if (a.getTheme().resolveAttribute(android.R.attr.actionBarSize, tv, true)
                    && tv.type == TypedValue.TYPE_DIMENSION && tv.data > 0) {
                return tv.data;
            }
        } catch (Throwable ignored) { }
        return dp(a, 48f);
    }

    /** 状态栏/刘海顶部高度：Insets 优先，status_bar_height 兜底 */
    @SuppressWarnings("deprecation")
    private static int resolveTopInset(Activity a) {
        int top = 0;
        View decor = a.getWindow() != null ? a.getWindow().getDecorView() : null;
        if (decor != null && Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            try {
                WindowInsets insets = decor.getRootWindowInsets();
                if (insets != null) {
                    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                        Insets bars = insets.getInsets(WindowInsets.Type.systemBars());
                        top = bars.top;
                        Insets cut = insets.getInsets(WindowInsets.Type.displayCutout());
                        if (cut.top > top) top = cut.top;
                    } else {
                        top = insets.getSystemWindowInsetTop();
                        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P && insets.getDisplayCutout() != null) {
                            int c = insets.getDisplayCutout().getSafeInsetTop();
                            if (c > top) top = c;
                        }
                    }
                }
            } catch (Throwable ignored) { }
        }
        if (top <= 0) {
            try {
                int id = a.getResources().getIdentifier("status_bar_height", "dimen", "android");
                if (id > 0) top = a.getResources().getDimensionPixelSize(id);
            } catch (Throwable ignored) { }
        }
        return top;
    }
}

import android.app.Activity;
import android.os.Bundle;
import android.util.Log;

import java.util.Map;
import java.util.WeakHashMap;

import de.robv.android.xposed.IXposedHookLoadPackage;
import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.XposedHelpers;
import de.robv.android.xposed.callbacks.XC_LoadPackage;

/**
 * assets/xposed_init 内容写：com.xposed.wxhome.MainHook
 * 作用域只需勾选「微信」com.tencent.mm
 */
public final class MainHook implements IXposedHookLoadPackage {

    private static final String TAG = "WxHomeCornerHook";

    /** 每个 LauncherUI 实例一个控制器（多开任务互不干扰，弱引用不泄漏） */
    private static final Map<Activity, HomeCornerButton> INSTANCES = new WeakHashMap<>();

    @Override
    public void handleLoadPackage(XC_LoadPackage.LoadPackageParam lpparam) {
        if (!WxConfig.TARGET_PACKAGE.equals(lpparam.packageName)) return;
        if (!WxConfig.TARGET_PACKAGE.equals(lpparam.processName)) return;   // 只在微信主进程

        final ClassLoader cl = lpparam.classLoader;
        Class<?> launcher = XposedHelpers.findClassIfExists(WxClass.LAUNCHER_UI, cl);
        if (launcher == null) { log("找不到 " + WxClass.LAUNCHER_UI + "，请确认微信版本"); return; }

        hookLifecycle(launcher);
        hookChatting(launcher);
        hookTabSwitch(cl);

        log("hook 安装完成 (微信主页左上角按钮)");
    }

    // ---------------- 生命周期：决定"挂/摘" ----------------

    private static void hookLifecycle(final Class<?> launcher) {
        hook(launcher, "onResume", new XC_MethodHook() {
            @Override protected void afterHookedMethod(MethodHookParam p) {
                get((Activity) p.thisObject).start();
            }
        });
        hook(launcher, "onPause", new XC_MethodHook() {
            @Override protected void afterHookedMethod(MethodHookParam p) {
                HomeCornerButton b = peek((Activity) p.thisObject);
                if (b != null) b.stop();
            }
        });
        hook(launcher, "onDestroy", new XC_MethodHook() {
            @Override protected void afterHookedMethod(MethodHookParam p) {
                Activity a = (Activity) p.thisObject;
                synchronized (INSTANCES) {
                    HomeCornerButton b = INSTANCES.remove(a);
                    if (b != null) b.dispose();
                }
            }
        });
    }

    // ---------------- 进入/退出聊天页：即时显隐 ----------------

    private static void hookChatting(final Class<?> launcher) {
        XC_MethodHook refresh = new XC_MethodHook() {
            @Override protected void afterHookedMethod(MethodHookParam p) { refresh((Activity) p.thisObject); }
        };
        hook(launcher, "startChatting", String.class, Bundle.class, boolean.class, refresh);
        hook(launcher, "closeChatting",  boolean.class, refresh);
    }

    // ---------------- Tab 切换：即时显隐 ----------------

    private static void hookTabSwitch(ClassLoader cl) {
        XC_MethodHook refresh = new XC_MethodHook() {
            @Override protected void afterHookedMethod(MethodHookParam p) {
                Object tabUI = p.thisObject;                        // MainTabUI 实例
                if (tabUI == null) return;
                Object home = field(tabUI, "a");                    // MainTabUI.a : MMFragmentActivity
                if (home instanceof Activity) refresh((Activity) home);
            }
        };
        Class<?> mainTab = XposedHelpers.findClassIfExists(WxClass.MAIN_TAB_UI, cl);
        if (mainTab != null) hook(mainTab, "a", int.class, refresh);

        // 补一层 ViewPager 回调（若该版本存在）
        Class<?> adapter = XposedHelpers.findClassIfExists(WxClass.TABS_ADAPTER, cl);
        if (adapter != null) {
            hook(adapter, "onPageSelected", int.class, new XC_MethodHook() {
                @Override protected void afterHookedMethod(MethodHookParam p) {
                    Object outer = field(p.thisObject, "this$0");    // MainTabUI$TabsAdapter.this$0
                    if (outer == null) return;
                    Object home = field(outer, "a");
                    if (home instanceof Activity) refresh((Activity) home);
                }
            });
        }
    }

    // ---------------- 工具 ----------------

    private static void refresh(Activity a) {
        HomeCornerButton b;
        synchronized (INSTANCES) { b = INSTANCES.get(a); }
        if (b != null) b.refresh();
    }

    private static synchronized HomeCornerButton get(Activity a) {
        HomeCornerButton b = INSTANCES.get(a);
        if (b == null) { b = new HomeCornerButton(a); INSTANCES.put(a, b); }
        return b;
    }

    private static HomeCornerButton peek(Activity a) {
        synchronized (INSTANCES) { return INSTANCES.get(a); }
    }

    private static Object field(Object o, String name) {
        try { return XposedHelpers.getObjectField(o, name); } catch (Throwable t) { return null; }
    }

    private static void hook(Class<?> c, String name, XC_MethodHook h) { hook(c, name, null, h); }

    private static void hook(Class<?> c, String name, Class<?>[] params, XC_MethodHook h) {
        try { XposedHelpers.findAndHookMethod(c, name, params, h); }
        catch (Throwable t) { log("hook 失败 " + c.getName() + "#" + name + ": " + t); }
    }

    private static void log(String s) {
        if (!WxConfig.debugLog) return;
        XposedBridge.log(TAG + ": " + s);
        Log.i(TAG, s);
    }
