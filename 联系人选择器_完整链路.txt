  [精准SQL — 仅好友, 排除系统账号/群/公众号]
  ──────────────────────────────────────
    SELECT username, nickname, alias, conRemark, type, sex, verifyFlag,
           deleteFlag, chatroomFlag, pyInitial, quanPin
    FROM rcontact
    WHERE deleteFlag = 0
      AND type = 0
      AND verifyFlag > 0
      AND username NOT LIKE '%@chatroom'
      AND username NOT LIKE 'gh_%'
      AND username NOT IN ('weixin','filehelper','medianote','newsapp','floatbottle')
      AND username NOT LIKE 'qmessage%'
      AND username NOT LIKE 'tmessage%'
      AND username NOT LIKE '%@openim'
      AND username NOT LIKE '%@stranger'
      AND username NOT LIKE '%@app'
      AND username NOT LIKE '%@talkroom'
      AND username NOT LIKE '%@lbsroom'
      AND sex = 1                           -- [可选] 1=男 2=女
    ORDER BY
      CASE WHEN conRemark IS NOT NULL AND conRemark != '' THEN 0 ELSE 1 END,
      CASE WHEN pyInitial IS NOT NULL AND pyInitial != '' THEN pyInitial
           ELSE quanPin END

  过滤说明:
    type=0         → 好友 (2=被删, 4=拉黑)
    verifyFlag>0   → 排除系统账号(它们的昵称是fake_)
    !@chatroom     → 排除群聊账户 (75个)
    !gh_           → 排除公众号
    NOT IN         → 排除微信内置角色
    !qmessage%     → 排除QQ消息助手
    sex=1/2        → 性别筛选

  /**
   * 加载过滤后的好友列表 (仅好友, 排除群/公众号/系统账号)
   */
  public static List<ContactItem> loadFriends(Context ctx) {
    List<ContactItem> list = new ArrayList<>();
    SQLiteDatabase db = null;
    Cursor c = null;
    try {
      // Step 1: DB路径
      long uin = ctx.getSharedPreferences("system_config_prefs", 0)
          .getLong("default_uin", 0);
      if (uin == 0) uin = ctx.getSharedPreferences("system_config_prefs", 0)
          .getInt("default_uin", 0);
      if (uin == 0) return list;
      String hash = md5(String.valueOf(uin));
      String dbPath = "/data/user/0/com.tencent.mm/MicroMsg/" + hash + "/EnMicroMsg.db";
      if (!new File(dbPath).exists())
        dbPath = "/data/data/com.tencent.mm/MicroMsg/" + hash + "/EnMicroMsg.db";
      if (!new File(dbPath).exists()) return list;

      // Step 2: 打开DB
      db = SQLiteDatabase.openDatabase(dbPath, null, SQLiteDatabase.OPEN_READONLY);

      // Step 3: 精准过滤SQL
      String sql = "SELECT username,nickname,alias,conRemark,type,sex,verifyFlag,"
          + "chatroomFlag,deleteFlag,pyInitial,quanPin "
          + "FROM rcontact WHERE deleteFlag=0 AND type=0 AND verifyFlag>0 "
          + "AND username NOT LIKE '%@chatroom' "
          + "AND username NOT LIKE 'gh_%' "
          + "AND username NOT IN ('weixin','filehelper','medianote','newsapp','floatbottle') "
          + "AND username NOT LIKE 'qmessage%' "
          + "AND username NOT LIKE 'tmessage%' "
          + "AND username NOT LIKE '%@openim' "
          + "AND username NOT LIKE '%@stranger' "
          + "ORDER BY CASE WHEN conRemark IS NOT NULL AND conRemark!='' THEN 0 ELSE 1 END, "
          + "CASE WHEN pyInitial IS NOT NULL AND pyInitial!='' THEN pyInitial ELSE quanPin END";

      // Step 4: 遍历
      c = db.rawQuery(sql, null);
      while (c.moveToNext()) {
        ContactItem item = new ContactItem();
        item.username  = c.getString(c.getColumnIndex("username"));
        item.nickname  = c.getString(c.getColumnIndex("nickname"));
        item.alias     = c.getString(c.getColumnIndex("alias"));
        item.conRemark = c.getString(c.getColumnIndex("conRemark"));
        item.type      = c.getInt(c.getColumnIndex("type"));
        item.sex       = c.getInt(c.getColumnIndex("sex"));
        item.verifyFlag= c.getInt(c.getColumnIndex("verifyFlag"));
        item.chatroomFlag = c.getInt(c.getColumnIndex("chatroomFlag"));
        item.pyInitial = c.getString(c.getColumnIndex("pyInitial"));

        // Step 5: 显示名
        if (item.conRemark != null && !item.conRemark.isEmpty())
          item.displayName = item.conRemark;
        else if (item.alias != null && !item.alias.isEmpty()
            && !item.alias.startsWith("wxid_"))
          item.displayName = item.alias;
        else if (item.nickname != null && !item.nickname.isEmpty())
          item.displayName = item.nickname;
        else
          item.displayName = item.username;

        // Step 6: 头像路径
        item.avatarPath = "/data/user/0/com.tencent.mm/MicroMsg/"
            + hash + "/avatar/avatar_" + item.username + "_96.jpg";

        list.add(item);
      }
    } catch (Throwable t) {
      Log.e("ContactPicker", "loadFriends: " + t.getMessage());
    } finally {
      if (c != null) c.close();
      if (db != null) db.close();
    }
    return list;
  }

  /**
   * 数据模型
   */
  public static class ContactItem {
    public String username;
    public String nickname;
    public String alias;
    public String conRemark;
    public String displayName;
    public String avatarPath;
    public String pyInitial;
    public int type;
    public int sex;
    public int verifyFlag;
    public int chatroomFlag;
  }

  /**
   * MD5
   */
  private static String md5(String s) {
    try {
      MessageDigest md = MessageDigest.getInstance("MD5");
      byte[] d = md.digest(s.getBytes("UTF-8"));
      StringBuilder sb = new StringBuilder();
      for (byte b : d) sb.append(String.format("%02x", b));
      return sb.toString();
    } catch (Throwable t) { return ""; }
  }

第9层：动态监听变化
================================================================================

  // 在模块 Hook 入口中注册
  public static void hookContactChange(ClassLoader cl) {
    Class<?> j4 = XposedHelpers.findClass("com.tencent.mm.storage.j4", cl);

    // 监听联系人更新
    XposedBridge.hookAllMethods(j4, "h0", new XC_MethodHook() {
      @Override protected void afterHookedMethod(MethodHookParam param) {
        boolean success = (Boolean) param.getResult();
        if (success) {
          // 联系人已变化 → 发送广播通知选择器刷新
          Intent intent = new Intent("com.leshao.v3.CONTACT_CHANGED");
          sAppContext.sendBroadcast(intent);
        }
      }
    });
  }
