package com.fakeaddsource;

import java.util.List;

import de.robv.android.xposed.IXposedHookLoadPackage;
import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.callbacks.XC_LoadPackage;

/**
 * 微信添加好友来源伪造模块
 *
 * 核心原理：
 *   所有添加好友请求最终都通过 com.tencent.mm.pluginsdk.model.m3 (NetSceneVerifyUser)
 *   发送到服务器。该类的构造函数接受一个 int 类型的 Scene 参数，
 *   最终写入 protobuf 字段 a65.u07.i (LinkedList&lt;Integer&gt;)。
 *
 *   服务器根据 Scene 值决定"添加来源"文案：
 *     3  = QQ号         10 = 搜索手机号     12 = 群聊
 *     14 = 附近的人      15 = QQ(另一方式)   17 = 摇一摇
 *     18 = 扫一扫       30 = 名片推荐       44 = 二维码
 *     48 = 雷达加朋友    52 = 搜索(另一方式)
 *
 * @author LSPilot Analysis
 */
public class MainHook implements IXposedHookLoadPackage {

    // ─── 配置：要伪装的来源 Scene 值 ─────────────────────
    //  10 = "通过搜索手机号添加" （最常用伪装）
    //  12 = "通过群聊添加"
    //  30 = "通过名片推荐添加"
    //  14 = "通过附近的人添加"
    //   6 = "通过搜索微信号添加"
    private static final int FAKE_SCENE = 10;

    // 微信包名
    private static final String WECHAT_PKG = "com.tencent.mm";

    // NetSceneVerifyUser 全限定类名（混淆后保持不变）
    private static final String M3_CLASS =
            "com.tencent.mm.pluginsdk.model.m3";

    // 备份 Hook 点：h55.r.l(String, int, String)
    private static final String R_CLASS = "h55.r";
    private static final String R_METHOD = "l";

    // ─────────────────────────────────────────────────────

    @Override
    public void handleLoadPackage(XC_LoadPackage.LoadPackageParam lpparam) {
        // 只处理微信进程
        if (!WECHAT_PKG.equals(lpparam.packageName)) {
            return;
        }

        XposedBridge.log("[FakeAddSource] 微信已加载，开始注入...");

        // ====== Hook 1: NetSceneVerifyUser 所有构造函数 ======
        hookM3Constructors(lpparam);

        // ====== Hook 2 (备份): h55.r.l() 静态方法 ======
        hookRStaticMethod(lpparam);

        XposedBridge.log("[FakeAddSource] 注入完成！FakeScene=" + FAKE_SCENE);
    }

    /**
     * Hook com.tencent.mm.pluginsdk.model.m3 的所有构造函数。
     *
     * m3 有 6 个构造函数，Scene 参数位置：
     *   构造 1-2 (4参/8参):  args[3] 是 int scene
     *   构造 3-6 (多用户):   args[2] 是 List&lt;Integer&gt; sceneList
     *
     * 策略：hookAllConstructors → 遍历参数 → 智能修改
     */
    private void hookM3Constructors(XC_LoadPackage.LoadPackageParam lpp) {
        try {
            Class<?> m3Class = lpp.classLoader.loadClass(M3_CLASS);

            XposedBridge.hookAllConstructors(m3Class, new XC_MethodHook() {
                @Override
                protected void beforeHookedMethod(MethodHookParam param) {
                    try {
                        modifySceneParam(param);
                    } catch (Throwable t) {
                        XposedBridge.log("[FakeAddSource] Hook m3 异常: " + t.getMessage());
                    }
                }
            });

            XposedBridge.log("[FakeAddSource] ✓ m3 构造函数 Hook 成功");
        } catch (ClassNotFoundException e) {
            XposedBridge.log("[FakeAddSource] ✗ m3 类未找到: " + e.getMessage());
        }
    }

    /**
     * 智能修改 Scene 参数：兼容 4参/8参（单 int scene）和
     * 多用户构造函数（List&lt;Integer&gt; sceneList）。
     */
    private void modifySceneParam(MethodHookParam param) {
        Object[] args = param.args;
        if (args == null || args.length < 3) return;

        int originalScene = -1;

        // ── 路径 A：4参/8参构造函数 → args[3] 是 Integer scene ──
        if (args.length >= 4 && args[3] instanceof Integer) {
            originalScene = (Integer) args[3];
            if (originalScene != FAKE_SCENE) {
                args[3] = FAKE_SCENE;
                logSceneChange(originalScene, FAKE_SCENE, "单Scene模式");
            }
            return;
        }

        // ── 路径 B：多用户构造函数 → args[2] 是 List<Integer> ──
        if (args.length >= 3 && args[2] instanceof List) {
            List<?> list = (List<?>) args[2];
            if (!list.isEmpty() && list.get(0) instanceof Integer) {
                originalScene = (Integer) list.get(0);
                if (originalScene != FAKE_SCENE) {
                    @SuppressWarnings("unchecked")
                    List<Integer> sceneList = (List<Integer>) list;
                    sceneList.clear();
                    sceneList.add(FAKE_SCENE);
                    logSceneChange(originalScene, FAKE_SCENE, "多用户模式(size=" + list.size() + ")");
                }
            }
        }
    }

    /**
     * 备份 Hook：h55.r.l(String username, int addScene, String source)
     * 这是 SayHiWithSnsPermissionUI → coroutine → 发请求的路径。
     */
    private void hookRStaticMethod(XC_LoadPackage.LoadPackageParam lpp) {
        try {
            Class<?> rClass = lpp.classLoader.loadClass(R_CLASS);

            XposedBridge.hookAllMethods(rClass, R_METHOD, new XC_MethodHook() {
                @Override
                protected void beforeHookedMethod(MethodHookParam param) {
                    try {
                        // args[1] 是 int addScene
                        if (param.args.length >= 2 && param.args[1] instanceof Integer) {
                            int original = (Integer) param.args[1];
                            if (original != FAKE_SCENE) {
                                param.args[1] = FAKE_SCENE;
                                logSceneChange(original, FAKE_SCENE, "h55.r.l 备份路径");
                            }
                        }
                    } catch (Throwable t) {
                        XposedBridge.log("[FakeAddSource] Hook r.l 异常: " + t.getMessage());
                    }
                }
            });

            XposedBridge.log("[FakeAddSource] ✓ h55.r.l 备份 Hook 成功");
        } catch (ClassNotFoundException e) {
            XposedBridge.log("[FakeAddSource] ✗ h55.r 类未找到（非致命）: " + e.getMessage());
        }
    }

    /**
     * 统一日志输出
     */
    private void logSceneChange(int from, int to, String path) {
        XposedBridge.log(
            "[FakeAddSource] 来源伪造 [" + path + "] " +
            sceneName(from) + "(" + from + ") → " +
            sceneName(to) + "(" + to + ")"
        );
    }

    /**
     * Scene 值 → 中文名称映射
     */
    private static String sceneName(int scene) {
        switch (scene) {
            case 1:  return "通讯录";
            case 3:  return "搜索QQ号";
            case 6:  return "搜索微信号";
            case 10: return "搜索手机号";
            case 12: return "群聊";
            case 14: return "附近的人";
            case 15: return "搜索QQ号(2)";
            case 17: return "摇一摇";
            case 18: return "扫一扫";
            case 30: return "名片推荐";
            case 44: return "二维码";
            case 48: return "雷达加朋友";
            case 52: return "搜索(其他)";
            default: return "未知";
        }
    }
}
