# 微信「听一听」音乐悬浮球模块

> 功能：聊天窗口显示音乐悬浮球 → 点击进入听一听搜索 → 播放自动捕获 → 下载音频到本地缓存 → 回调交给你复制/转 silk/发送

---

## 一、完整代码（TingMusicModule.java）

```java
package com.your.module;   // ← 改成你的包名

import android.app.Activity;
import android.content.Context;
import android.content.Intent;
import android.graphics.Color;
import android.graphics.drawable.GradientDrawable;
import android.os.Handler;
import android.os.Looper;
import android.view.Gravity;
import android.view.MotionEvent;
import android.view.View;
import android.view.ViewGroup;
import android.widget.FrameLayout;
import android.widget.TextView;

import java.io.File;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.util.List;

import de.robv.android.xposed.IXposedHookLoadPackage;
import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.XposedHelpers;
import de.robv.android.xposed.callbacks.XC_LoadPackage;

/**
 * 微信「听一听」音乐悬浮球 + 搜索 + 播放捕获 + 音频下载到本地缓存
 *
 * 完整链路：
 *   聊天窗口悬浮球(♪) → 点击进入听一听 → 搜索播放 → 自动捕获音乐 → 悬浮球变✓
 *   → 点击✓ → 下载音频到缓存 → 回调 onReady(chatUser, info.localPath) → 你自己复制/转silk/发送
 */
public class TingMusicModule implements IXposedHookLoadPackage {

    // ==================== 回调接口 ====================
    /** 音频下载完成回调：info.localPath 就是下载到本地的音频文件绝对路径 */
    public interface OnMusicReady {
        void onReady(String chatUser, TingMusicInfo info);
    }
    private static OnMusicReady listener;
    public static void setOnMusicReady(OnMusicReady l) { listener = l; }

    // ==================== 数据模型 ====================
    public static class TingMusicInfo {
        public String listenId;    // 歌曲ID
        public String srcId;       // 来源ID
        public String title;       // 歌名
        public String author;      // 歌手
        public String cover;       // 封面
        public String dataUrl;     // 音频直链（下载源）
        public String webUrl;      // 网页链接
        public String bizUsername;
        public int    type;        // 3=普通音乐, 4=语音
        public String localPath;   // ★ 下载后的本地文件绝对路径（下载完成才有）
    }

    // ==================== 全局状态 ====================
    private static Activity      currentChatting;   // 当前聊天页
    private static String        currentUser;       // 当前聊天对象 username
    private static TingMusicInfo pendingMusic;      // 已捕获待发送的音乐
    private static TextView      ball;              // 悬浮球

    @Override
    public void handleLoadPackage(XC_LoadPackage.LoadPackageParam lp) {
        if (!"com.tencent.mm".equals(lp.packageName)) return;
        final ClassLoader cl = lp.classLoader;
        hookChattingUI(cl);
        hookTingPlayer(cl);
    }

    // ==================== ① 聊天页悬浮球 ====================
    private static void hookChattingUI(final ClassLoader cl) {
        try {
            Class<?> chattingUI = XposedHelpers.findClass("com.tencent.mm.ui.chatting.ChattingUI", cl);
            // 进入聊天页：记录 Activity + 聊天对象，挂悬浮球
            XposedHelpers.findAndHookMethod(chattingUI, "onResume", new XC_MethodHook() {
                @Override protected void afterHookedMethod(MethodHookParam p) {
                    currentChatting = (Activity) p.thisObject;
                    currentUser = currentChatting.getIntent().getStringExtra("Chat_User");
                    new Handler(Looper.getMainLooper()).postDelayed(() -> showBall(currentChatting, cl), 300);
                }
            });
            // 离开聊天页：移除悬浮球
            XposedHelpers.findAndHookMethod(chattingUI, "onPause", new XC_MethodHook() {
                @Override protected void afterHookedMethod(MethodHookParam p) {
                    removeBall();
                }
            });
        } catch (Throwable t) {
            XposedBridge.log("[Ting] 聊天页 hook 失败: " + t);
        }
    }

    private static void showBall(final Activity act, final ClassLoader cl) {
        try {
            if (ball != null) { updateBallState(); return; }
            ViewGroup root = (ViewGroup) act.getWindow().getDecorView().findViewById(android.R.id.content);
            if (root == null) return;

            ball = new TextView(act);
            ball.setTextColor(Color.WHITE);
            ball.setTextSize(22);
            ball.setGravity(Gravity.CENTER);
            ball.setBackground(makeCircle(0xE607C160));

            int size = dp(act, 52);
            FrameLayout.LayoutParams lp = new FrameLayout.LayoutParams(size, size);
            lp.gravity = Gravity.TOP | Gravity.END;
            lp.topMargin = dp(act, 120);
            lp.rightMargin = dp(act, 12);
            ball.setLayoutParams(lp);

            ball.setOnTouchListener(new TouchListener(act, cl, lp));
            root.addView(ball);
            updateBallState();
        } catch (Throwable t) {
            XposedBridge.log("[Ting] 悬浮球失败: " + t);
        }
    }

    /** 悬浮球状态：空闲=♪(绿,点去搜索)  就绪=✓(红,点发送) */
    private static void updateBallState() {
        if (ball == null) return;
        if (pendingMusic != null) {
            ball.setText("\u2713");
            ball.setBackground(makeCircle(0xE6FA5151));
        } else {
            ball.setText("\u266A");
            ball.setBackground(makeCircle(0xE607C160));
        }
    }

    private static void removeBall() {
        try {
            if (ball != null && ball.getParent() != null) {
                ((ViewGroup) ball.getParent()).removeView(ball);
            }
        } catch (Throwable ignored) {}
        ball = null;
    }

    private static GradientDrawable makeCircle(int color) {
        GradientDrawable g = new GradientDrawable();
        g.setShape(GradientDrawable.OVAL);
        g.setColor(color);
        return g;
    }

    // 拖动 + 点击
    private static class TouchListener implements View.OnTouchListener {
        private final Activity act; private final ClassLoader cl;
        private final FrameLayout.LayoutParams lp;
        private float downX, downY, lastX, lastY;
        private boolean moved;

        TouchListener(Activity a, ClassLoader c, FrameLayout.LayoutParams p) { act = a; cl = c; lp = p; }

        @Override public boolean onTouch(View v, MotionEvent e) {
            switch (e.getActionMasked()) {
                case MotionEvent.ACTION_DOWN:
                    downX = lastX = e.getRawX(); downY = lastY = e.getRawY(); moved = false; return true;
                case MotionEvent.ACTION_MOVE:
                    float dx = e.getRawX() - lastX, dy = e.getRawY() - lastY;
                    if (Math.abs(e.getRawX()-downX) > 10 || Math.abs(e.getRawY()-downY) > 10) moved = true;
                    if (moved) {
                        lp.leftMargin += (int) dx; lp.topMargin += (int) dy;
                        lp.gravity = Gravity.TOP | Gravity.START; ball.setLayoutParams(lp);
                    }
                    lastX = e.getRawX(); lastY = e.getRawY(); return true;
                case MotionEvent.ACTION_UP:
                    if (!moved) onClickBall();
                    return true;
            }
            return false;
        }

        private void onClickBall() {
            if (pendingMusic != null) {
                // ===== 状态2：就绪 → 下载音频 =====
                XposedBridge.log("[Ting] 开始下载: " + pendingMusic.title + "  url=" + pendingMusic.dataUrl);
                downloadAudio(pendingMusic.dataUrl, act, new DownloadCallback() {
                    @Override public void onSuccess(String localPath) {
                        pendingMusic.localPath = localPath;
                        XposedBridge.log("[Ting] 下载完成: " + localPath);
                        if (listener != null) listener.onReady(currentUser, pendingMusic);  // 交给你
                        pendingMusic = null;
                        updateBallState();
                    }
                    @Override public void onFail(String err) {
                        XposedBridge.log("[Ting] 下载失败: " + err);
                    }
                });
            } else {
                // ===== 状态1：空闲 → 进入听一听搜索 =====
                try {
                    Class<?> ting = XposedHelpers.findClass("com.tencent.mm.plugin.ting.TingFlutterActivity", cl);
                    Intent i = new Intent(act, ting);
                    i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
                    act.startActivity(i);
                } catch (Throwable t) {
                    XposedBridge.log("[Ting] 启动听一听失败: " + t);
                }
            }
        }
    }

    // ==================== ② 播放捕获 ====================
    private static void hookTingPlayer(final ClassLoader cl) {
        Class<?> tingService = XposedHelpers.findClass("ul4.a9", cl);
        XC_MethodHook grabber = new XC_MethodHook() {
            @Override protected void afterHookedMethod(MethodHookParam p) {
                Object zr0 = p.args.length >= 2 ? p.args[1] : p.args[0];
                if (zr0 == null) return;
                TingMusicInfo info = extract(cl, zr0);
                if (info == null || info.dataUrl == null || info.dataUrl.isEmpty()) return;
                pendingMusic = info;
                XposedBridge.log("[Ting] 捕获音乐: " + info.title + " / " + info.author);
                XposedBridge.log("[Ting] 音频链接: " + info.dataUrl);
                updateBallState();
            }
        };
        try {
            XposedHelpers.findAndHookMethod(tingService, "Ej", String.class,
                    XposedHelpers.findClass("ix5.zr0", cl),
                    XposedHelpers.findClass("mm4.e", cl), grabber);
        } catch (Throwable ignored) {}
        try {
            XposedHelpers.findAndHookMethod(tingService, "Fj",
                    XposedHelpers.findClass("ix5.zr0", cl), List.class,
                    XposedHelpers.findClass("ix5.n70", cl),
                    XposedHelpers.findClass("ix5.ds0", cl),
                    XposedHelpers.findClass("mm4.e", cl), grabber);
        } catch (Throwable ignored) {}
        try {
            XposedHelpers.findAndHookMethod(tingService, "qj",
                    XposedHelpers.findClass("ix5.zr0", cl), float.class, grabber);
        } catch (Throwable ignored) {}
    }

    private static TingMusicInfo extract(ClassLoader cl, Object zr0) {
        TingMusicInfo info = new TingMusicInfo();
        try {
            Object w90 = XposedHelpers.callMethod(zr0, "d");
            if (w90 == null) return null;
            info.listenId = (String) XposedHelpers.callMethod(w90, "getListenId");
            info.title    = (String) XposedHelpers.callMethod(w90, "getTitle");
            info.author   = (String) XposedHelpers.callMethod(w90, "c");
            info.cover    = (String) XposedHelpers.callMethod(w90, "f");
            info.type     = XposedHelpers.getIntField(w90, "e");
            info.srcId    = (String) XposedHelpers.callMethod(zr0, "b");
            if (info.type == 3) {
                Object h60 = XposedHelpers.callMethod(w90, "b");
                if (h60 != null) {
                    info.dataUrl     = (String) XposedHelpers.callMethod(h60, "d");   // ★ 音频直链
                    info.webUrl      = (String) XposedHelpers.callMethod(h60, "getUrl");
                    info.bizUsername = (String) XposedHelpers.callMethod(h60, "getBizUsername");
                }
            } else if (info.type == 4) {
                Object f90 = XposedHelpers.callMethod(w90, "g");
                if (f90 != null) info.dataUrl = (String) XposedHelpers.callMethod(f90, "getTid");
            }
        } catch (Throwable t) {
            XposedBridge.log("[Ting] 提取失败: " + t);
        }
        return info;
    }

    // ==================== ③ 下载音频到本地缓存 ====================
    public interface DownloadCallback {
        void onSuccess(String localPath);
        void onFail(String error);
    }

    private static void downloadAudio(final String url, final Context ctx, final DownloadCallback cb) {
        new Thread(() -> {
            String result = null;
            String error = null;
            try {
                HttpURLConnection conn = (HttpURLConnection) new URL(url).openConnection();
                conn.setConnectTimeout(10000);
                conn.setReadTimeout(60000);
                conn.setRequestProperty("User-Agent", "Mozilla/5.0 (Linux; Android)");
                conn.setInstanceFollowRedirects(true);
                conn.connect();
                int code = conn.getResponseCode();
                if (code != 200) {
                    error = "HTTP " + code;
                } else {
                    // 下载到缓存目录 /cache/ting_music/
                    File dir = new File(ctx.getCacheDir(), "ting_music");
                    if (!dir.exists()) dir.mkdirs();
                    File file = new File(dir, "ting_" + System.currentTimeMillis() + guessExt(url));
                    InputStream in = conn.getInputStream();
                    FileOutputStream out = new FileOutputStream(file);
                    byte[] buf = new byte[8192];
                    int n;
                    while ((n = in.read(buf)) != -1) out.write(buf, 0, n);
                    out.close();
                    in.close();
                    result = file.getAbsolutePath();
                }
                conn.disconnect();
            } catch (Throwable t) {
                error = t.getMessage();
            }
            // 切回主线程回调
            final String r = result, e = error;
            new Handler(Looper.getMainLooper()).post(() -> {
                if (r != null) cb.onSuccess(r); else cb.onFail(e);
            });
        }).start();
    }

    private static String guessExt(String url) {
        String path = url;
        int q = path.indexOf('?');
        if (q > 0) path = path.substring(0, q);
        int dot = path.lastIndexOf('.');
        if (dot > 0) {
            String ext = path.substring(dot);
            if (ext.length() <= 5) return ext;   // .mp3 .m4a .aac .wav
        }
        return ".mp3";
    }

    // ==================== 工具 ====================
    private static int dp(Context c, int v) { return (int)(c.getResources().getDisplayMetrics().density * v + 0.5f); }
}

## 三、完整交互流程

```
【聊天窗口】悬浮球 ♪（绿色）
     │ 点击（空闲态）
     ▼
【听一听】搜索 → 播放歌曲
     │ Hook 自动捕获 zr0 → 提取 dataUrl/歌名/歌手
     ▼
悬浮球变 ✓（红色 = 就绪）
     │ 点击（就绪态）
     ▼
【下载】dataUrl → /cache/ting_music/ting_xxx.mp3
     │ 下载完成
     ▼
onReady(chatUser, info)   ← info.localPath = 本地音频路径
     │
     ▼
你的流程：复制到微信内部路径 → 转 silk → 调语音发送

## 四、关键说明

| 项 | 说明 |
|---|---|
| **下载位置** | `context.getCacheDir()/ting_music/ting_时间戳.扩展名`，扩展名从 URL 自动推断，推断不出默认 `.mp3` |
| **下载线程** | 子线程下载，`Handler` 切回主线程回调，不卡 UI |
| **重定向/超时** | 已处理 302 重定向，连接超时 10s、读取 60s |
| **发给谁** | `currentUser` = `ChattingUI` 的 `Intent.getStringExtra("Chat_User")` |
| **悬浮球状态** | ♪ 绿=点击去搜索；✓ 红=已就绪，点击下载并回调 |
| **下载失败兜底** | 若 `dataUrl` 带鉴权且过期会下载失败，需重新在听一听里播放一次刷新链接 |

## 五、版本风险

1. **混淆类名随微信版本变**：`ul4.a9`、`ix5.zr0/w90/h60`、方法名 `Ej/Fj/qj/c/f/d/b` 是混淆的。稳定锚点是 `getListenId/getTitle/getUrl/getBizUsername`（protobuf 显式 getter）和 `ChattingUI`、`TingFlutterActivity`。
2. **`Ej/Fj/qj` 三选一命中即可**：签名对不上会自动跳过不崩。若三个都失败（微信大版本更新），用日志 `"MicroMsg.TingServicesImpl"` 反查 `ul4.a9` 新类名，再更新方法名。
