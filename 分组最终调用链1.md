/**
 * UI 注入器 —— 最终可靠版
 *
 * ====== 功能1: 标题栏下方标签筛选栏 ======
 * 策略: 给 ConversationListView (extends ListView) 添加 HeaderView
 *   - ListView.addHeaderView() 会将 View 固定在列表顶部
 *   - 视觉效果 = 标题栏正下方、聊天列表上方
 *   - 不依赖任何资源ID，不依赖父容器结构
 *
 * ====== 功能2: 长按菜单注入 ======
 * 策略: Hook r3.onCreateContextMenu() → MenuItem.setOnMenuItemClickListener()
 */
public class UiInjector {

    private static final String TAG = "WxGroup.UI";
    private static View sTagBarView;
    private static LinearLayout sTagChips;
    private static int sSelectedLabelId = -1;
    private static List<LabelInfo> sCachedLabels;
    private static boolean sHeaderAdded = false;

    // ================================================================
    // 功能1: ListView.addHeaderView() 注入标签栏
    // ================================================================

    public static void installTagFilterBar(ClassLoader cl) {
        try {
            // Hook MainUI.onResume —— 会话列表 Fragment 可见时
            Class<?> mainUI = XposedHelpers.findClass(
                "com.tencent.mm.ui.conversation.MainUI", cl);

            XposedHelpers.findAndHookMethod(mainUI, "onResume",
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam param) {
                        new Handler(Looper.getMainLooper()).postDelayed(() -> {
                            injectHeaderToConversationList(param.thisObject);
                        }, 300);
                    }
                });

            // 监听标签变更 → 刷新
            EventBus.subscribe(EventBus.Event.LABEL_CREATED,  (e,d) -> refreshChips());
            EventBus.subscribe(EventBus.Event.LABEL_DELETED,  (e,d) -> refreshChips());
            EventBus.subscribe(EventBus.Event.LABEL_RENAMED,  (e,d) -> refreshChips());
            EventBus.subscribe(EventBus.Event.LABELS_SYNCED,  (e,d) -> refreshChips());

            XposedBridge.log(TAG + ": ✓ 标签栏 HeaderView Hook 已安装");
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": ✗ installTagFilterBar: " + e.getMessage());
        }
    }

    /**
     * 找到 ConversationListView 并添加 HeaderView
     */
    private static void injectHeaderToConversationList(Object mainUI) {
        try {
            if (sHeaderAdded && sTagBarView != null && sTagBarView.getParent() != null) {
                refreshChips();
                return;
            }

            // 获取 MainUI 的根 View
            View root = (View) XposedHelpers.getObjectField(mainUI, "G");
            if (root == null) return;

            // ★ 遍历找 ConversationListView (extends ListView) ★
            ListView listView = findConversationListView(root);
            if (listView == null) {
                // Fallback: 也尝试找 RecyclerView
                View rv = findRecyclerView(root);
                if (rv != null) {
                    injectAboveRecyclerView(rv);
                    return;
                }
                return;
            }

            // 避免重复添加
            if (listView.getHeaderViewsCount() > 0) {
                for (int i = 0; i < listView.getHeaderViewsCount(); i++) {
                    // 简单检测: 第一个 header 是否已有我们的标记尺寸
                }
            }

            // 构建标签栏
            sTagBarView = buildTagBar(listView.getContext());

            // ★ 关键: addHeaderView ★
            listView.addHeaderView(sTagBarView, null, false);

            refreshChips();
            sHeaderAdded = true;
            XposedBridge.log(TAG + ": ✓ 标签栏已通过 addHeaderView 注入");

        } catch (Throwable e) {
            XposedBridge.log(TAG + ": injectHeader error: " + e.getMessage());
        }
    }

    /**
     * Fallback: RecyclerView 上方注入
     */
    private static void injectAboveRecyclerView(View recyclerView) {
        ViewGroup parent = (ViewGroup) recyclerView.getParent();
        if (parent == null) return;

        sTagBarView = buildTagBar(recyclerView.getContext());
        int idx = parent.indexOfChild(recyclerView);
        parent.addView(sTagBarView, idx,
            new ViewGroup.LayoutParams(
                ViewGroup.LayoutParams.MATCH_PARENT,
                dp(44, recyclerView.getContext())));
        refreshChips();
        sHeaderAdded = true;
    }

    // 标签栏构建

    private static View buildTagBar(Context ctx) {
        // 外层用 FrameLayout 确保在 ListView header 中正确显示
        FrameLayout wrapper = new FrameLayout(ctx);
        wrapper.setLayoutParams(new AbsListView.LayoutParams(
            ViewGroup.LayoutParams.MATCH_PARENT, dp(44, ctx)));

        HorizontalScrollView hsv = new HorizontalScrollView(ctx);
        hsv.setHorizontalScrollBarEnabled(false);
        hsv.setBackgroundColor(Color.parseColor("#EDEDED"));

        // 底部细线
        View line = new View(ctx);
        line.setBackgroundColor(Color.parseColor("#D9D9D9"));
        FrameLayout.LayoutParams lineLp = new FrameLayout.LayoutParams(
            ViewGroup.LayoutParams.MATCH_PARENT, dp(1, ctx));
        lineLp.gravity = Gravity.BOTTOM;
        wrapper.addView(hsv);
        wrapper.addView(line);

        LinearLayout ll = new LinearLayout(ctx);
        ll.setOrientation(LinearLayout.HORIZONTAL);
        ll.setGravity(Gravity.CENTER_VERTICAL);
        int p = dp(10, ctx);
        ll.setPadding(p, dp(4, ctx), p, dp(4, ctx));
        hsv.addView(ll);
        sTagChips = ll;

        return wrapper;
    }

    private static void refreshChips() {
        if (sTagChips == null) return;
        sCachedLabels = LabelManager.getAll();
        Context ctx = sTagChips.getContext();
        sTagChips.removeAllViews();

        addChip(ctx, "全部", -1, sSelectedLabelId == -1);

        for (LabelInfo l : sCachedLabels) {
            addChip(ctx, l.labelName, l.labelId, sSelectedLabelId == l.labelId);
        }

        // ＋ 按钮
        TextView plus = new TextView(ctx);
        plus.setText("＋");
        plus.setTextSize(18);
        plus.setTextColor(Color.parseColor("#07C160"));
        plus.setPadding(dp(10, ctx), 0, dp(10, ctx), 0);
        plus.setGravity(Gravity.CENTER);
        plus.setOnClickListener(v -> showCreateDialog(ctx, null));
        sTagChips.addView(plus);
    }

    private static void addChip(Context ctx, String text, int id, boolean sel) {
        TextView tv = new TextView(ctx);
        tv.setText(text);
        tv.setTextSize(13);
        tv.setPadding(dp(12, ctx), dp(5, ctx), dp(12, ctx), dp(5, ctx));
        tv.setGravity(Gravity.CENTER);
        tv.setSingleLine(true);
        tv.setMaxWidth(dp(120, ctx));

        GradientDrawable bg = new GradientDrawable();
        bg.setCornerRadius(dp(14, ctx));
        if (sel) {
            bg.setColor(Color.parseColor("#07C160"));
            tv.setTextColor(Color.WHITE);
        } else {
            bg.setColor(Color.WHITE);
            tv.setTextColor(Color.parseColor("#353535"));
            bg.setStroke(dp(1, ctx), Color.parseColor("#D9D9D9"));
        }
        tv.setBackground(bg);

        LinearLayout.LayoutParams lp = new LinearLayout.LayoutParams(
            ViewGroup.LayoutParams.WRAP_CONTENT, ViewGroup.LayoutParams.WRAP_CONTENT);
        int m = dp(4, ctx);
        lp.setMargins(m, 0, m, 0);
        tv.setLayoutParams(lp);

        tv.setOnClickListener(v -> {
            sSelectedLabelId = (sSelectedLabelId == id) ? -1 : id;
            refreshChips();
            Toast.makeText(ctx, id <= 0 ? "全部" : "筛选: " + text, Toast.LENGTH_SHORT).show();
        });

        if (id > 0) {
            tv.setOnLongClickListener(v -> {
                showManageDialog(ctx, id, text);
                return true;
            });
        }
        sTagChips.addView(tv);
    }

    public static void installContextMenuHook(ClassLoader cl) {
        try {
            Class<?> r3 = XposedHelpers.findClass(
                "com.tencent.mm.ui.conversation.r3", cl);

            XposedHelpers.findAndHookMethod(r3, "onCreateContextMenu",
                ContextMenu.class, View.class, ContextMenu.ContextMenuInfo.class,
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam p) {
                        injectMenu(p);
                    }
                });
            XposedBridge.log(TAG + ": ✓ 长按菜单 Hook 已安装");
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": ✗ context menu hook: " + e.getMessage());
        }
    }

    private static void injectMenu(MethodHookParam p) {
        try {
            ContextMenu menu = (ContextMenu) p.args[0];
            Object r3 = p.thisObject;
            String username = (String) XposedHelpers.getObjectField(r3, "g");
            Activity act = (Activity) XposedHelpers.getObjectField(r3, "f");
            if (TextUtils.isEmpty(username) || act == null) return;

            int[] cur = ContactHelper.getLabelIds(username);
            Set<Integer> curSet = new HashSet<>();
            for (int id : cur) curSet.add(id);
            List<LabelInfo> all = LabelManager.getAll();
            if (all.isEmpty()) return;

            ContextMenu.ContextMenuInfo info = (ContextMenu.ContextMenuInfo) p.args[2];
            int pos = 0;
            if (info instanceof AdapterView.AdapterContextMenuInfo) {
                pos = ((AdapterView.AdapterContextMenuInfo) info).position;
            }

            for (int i = 0; i < Math.min(all.size(), 8); i++) {
                final LabelInfo lb = all.get(i);
                final boolean has = curSet.contains(lb.labelId);
                MenuItem item = menu.add(pos, 50001 + lb.labelId, 0,
                    (has ? "☑ " : "☐ ") + lb.labelName);
                item.setOnMenuItemClickListener(clicked -> {
                    if (has) ContactHelper.removeLabel(username, lb.labelId);
                    else ContactHelper.addLabel(username, lb.labelId);
                    Toast.makeText(act, (has ? "已移除" : "已添加"), Toast.LENGTH_SHORT).show();
                    return true;
                });
            }

            if (all.size() > 8) {
                MenuItem more = menu.add(pos, 50999, 8, "管理所有分组...");
                more.setOnMenuItemClickListener(clicked -> {
                    showFullDialog(act, username, curSet, all);
                    return true;
                });
            }

            MenuItem create = menu.add(pos, 50998, 9, "＋ 新建分组");
            create.setOnMenuItemClickListener(clicked -> {
                showCreateDialog(act, username);
                return true;
            });
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": injectMenu: " + e.getMessage());
        }
    }

    private static void showFullDialog(Context ctx, String u, Set<Integer> cur, List<LabelInfo> all) {
        String[] ns = new String[all.size()];
        boolean[] chk = new boolean[all.size()];
        for (int i = 0; i < all.size(); i++) {
            ns[i] = all.get(i).labelName;
            chk[i] = cur.contains(all.get(i).labelId);
        }
        new AlertDialog.Builder(ctx)
            .setTitle("管理分组")
            .setMultiChoiceItems(ns, chk, (d, i, b) -> {
                LabelInfo lb = all.get(i);
                if (b) ContactHelper.addLabel(u, lb.labelId);
                else ContactHelper.removeLabel(u, lb.labelId);
            })
            .setPositiveButton("完成", null)
            .setNeutralButton("＋ 新建", (d, w) -> showCreateDialog(ctx, u))
            .show();
    }

    private static void showCreateDialog(Context ctx, String username) {
        EditText et = new EditText(ctx);
        et.setHint("分组名称");
        et.setPadding(50, 35, 50, 35);
        new AlertDialog.Builder(ctx)
            .setTitle("新建分组")
            .setView(et)
            .setPositiveButton("创建", (d, w) -> {
                String n = et.getText().toString().trim();
                if (TextUtils.isEmpty(n)) return;
                LabelInfo lb = LabelManager.create(n);
                if (lb != null) {
                    if (username != null) ContactHelper.addLabel(username, lb.labelId);
                    refreshChips();
                    Toast.makeText(ctx, "已创建「" + n + "」", Toast.LENGTH_SHORT).show();
                }
            })
            .setNegativeButton("取消", null)
            .show();
    }

    private static void showManageDialog(Context ctx, int id, String name) {
        int cnt = LabelManager.getContacts(id).size();
        new AlertDialog.Builder(ctx)
            .setTitle(name)
            .setItems(new String[]{
                "查看联系人 (" + cnt + "人)",
                "重命名", "删除", "合并到其他分组"
            }, (d, w) -> {
                switch (w) {
                    case 0: {
                        List<String> cs = LabelManager.getContacts(id);
                        StringBuilder sb = new StringBuilder();
                        for (int i = 0; i < Math.min(cs.size(), 50); i++)
                            sb.append(cs.get(i)).append("\n");
                        if (cs.size() > 50) sb.append("...共").append(cs.size()).append("人");
                        new AlertDialog.Builder(ctx).setTitle(name).setMessage(sb.toString())
                            .setPositiveButton("关闭", null).show();
                        break;
                    }
                    case 1: {
                        EditText et = new EditText(ctx);
                        et.setText(name);
                        et.setPadding(50, 35, 50, 35);
                        new AlertDialog.Builder(ctx).setTitle("重命名").setView(et)
                            .setPositiveButton("确定", (dd, ww) -> {
                                String n = et.getText().toString().trim();
                                if (!TextUtils.isEmpty(n)) {
                                    LabelManager.rename(String.valueOf(id), n);
                                    refreshChips();
                                }
                            }).setNegativeButton("取消", null).show();
                        break;
                    }
                    case 2:
                        new AlertDialog.Builder(ctx)
                            .setTitle("删除").setMessage("删除「" + name + "」？")
                            .setPositiveButton("删除", (dd, ww) -> {
                                LabelManager.delete(String.valueOf(id));
                                if (sSelectedLabelId == id) sSelectedLabelId = -1;
                                refreshChips();
                            }).setNegativeButton("取消", null).show();
                        break;
                    case 3: {
                        List<LabelInfo> all = LabelManager.getAll();
                        String[] ns = new String[all.size()];
                        for (int i = 0; i < all.size(); i++) ns[i] = all.get(i).labelName;
                        new AlertDialog.Builder(ctx).setTitle("合并到...")
                            .setItems(ns, (dd, i2) -> {
                                LabelInfo t = all.get(i2);
                                if (t.labelId != id) {
                                    BatchOperator.mergeLabels(id, t.labelId);
                                    if (sSelectedLabelId == id) sSelectedLabelId = -1;
                                    refreshChips();
                                }
                            }).setNegativeButton("取消", null).show();
                        break;
                    }
                }
            }).setNegativeButton("取消", null).show();
    }

    /** 遍历找 ConversationListView (ListView 子类) */
    private static ListView findConversationListView(View root) {
        if (root instanceof ListView
            && root.getClass().getName().contains("ConversationListView")) {
            return (ListView) root;
        }
        if (root instanceof ViewGroup) {
            ViewGroup vg = (ViewGroup) root;
            for (int i = 0; i < vg.getChildCount(); i++) {
                ListView lv = findConversationListView(vg.getChildAt(i));
                if (lv != null) return lv;
            }
        }
        return null;
    }

    /** 遍历找 RecyclerView */
    private static View findRecyclerView(View root) {
        if (root.getClass().getName().contains("RecyclerView")) return root;
        if (root instanceof ViewGroup) {
            ViewGroup vg = (ViewGroup) root;
            for (int i = 0; i < vg.getChildCount(); i++) {
                View rv = findRecyclerView(vg.getChildAt(i));
                if (rv != null) return rv;
            }
        }
        return null;
    }

    private static int dp(int d, Context c) {
        return (int)(d * c.getResources().getDisplayMetrics().density + 0.5f);
    }
