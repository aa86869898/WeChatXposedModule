package com.yourname.wechatiso;

import android.content.Context;
import android.content.SharedPreferences;
import android.os.Process;

/**
 * ---------------------------------------------------------------------------
 * 【分身隔离核心】
 * ---------------------------------------------------------------------------
 * 原理：
 *   Android 系统分身（应用分身/第二空间/Work Profile）本质是多用户。
 *   每个 user 有独立 userId（0, 10, 11...）、独立进程、独立 /data/user/{userId}/ 目录。
 *   本类用 userId 作为实例标识，配置寄生在微信进程的 SharedPreferences 中，
 *   系统会自动按 user 隔离存储路径，实现主微信与分身微信配置零干扰。
 *
 * 使用要求：
 *   1. 必须在 Application.attachBaseContext 时调用 init(Context)
 *   2. 所有业务 Hook 通过 userId() / isPrimary() / feature() 判断当前实例
 *   3. 严禁把配置写到模块自身包名下，分身 user 可能无权访问
 * ---------------------------------------------------------------------------
 */
public final class InstanceManager {

    /** SharedPreferences 文件名，落在微信进程数据目录，按 user 自动隔离 */
    private static final String PREF_NAME = "xposed_wechat_iso";
    /** Android 多用户 UID 间隔，固定 100000 */
    private static final int PER_USER_RANGE = 100000;

    private static volatile int sUserId = -1;
    private static volatile String sDataDir = "";
    private static volatile boolean sPrimary = false;
    private static volatile SharedPreferences sPrefs;

    private InstanceManager() {}

    /**
     * 初始化当前实例。每个进程只需调用一次。
     *
     * @param ctx Application 的 BaseContext（来自 attachBaseContext）
     */
    public static void init(Context ctx) {
        if (sUserId != -1) return;

        sUserId = Process.myUid() / PER_USER_RANGE;
        sPrimary = (sUserId == 0);
        sDataDir = ctx.getApplicationInfo().dataDir;

        // 关键：存到微信进程自己的 SP，系统按 user 隔离文件路径
        sPrefs = ctx.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE);
    }

    /** 当前实例 userId：0=主微信，>0=系统分身 */
    public static int userId() {
        ensureInit();
        return sUserId;
    }

    /** 是否为主微信（user 0） */
    public static boolean isPrimary() {
        ensureInit();
        return sPrimary;
    }

    /** 当前微信实例数据目录，用于日志排查 */
    public static String dataDir() {
        ensureInit();
        return sDataDir;
    }

    /**
     * 当前实例总开关。
     * 默认：主微信开，分身关。如需分身默认开，改为 return true;
     */
    public static boolean isEnabled() {
        ensureInit();
        return sPrefs.getBoolean("enabled", isPrimary());
    }

    /** 功能开关。每个功能一个 key，各实例独立存储，互不干扰 */
    public static boolean feature(String key) {
        ensureInit();
        return sPrefs.getBoolean(key, false);
    }

    /** 设置总开关（仅影响当前实例） */
    public static void setEnabled(boolean on) {
        ensureInit();
        sPrefs.edit().putBoolean("enabled", on).apply();
    }

    /** 设置功能开关（仅影响当前实例） */
    public static void setFeature(String key, boolean on) {
        ensureInit();
        sPrefs.edit().putBoolean(key, on).apply();
    }

    private static void ensureInit() {
        if (sUserId == -1) {
            throw new IllegalStateException(
                "InstanceManager not initialized! Call init(Context) in Application.attachBaseContext first.");
        }
    }
}

package com.yourname.wechatiso;

import android.content.Context;
import de.robv.android.xposed.IXposedHookLoadPackage;
import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.XposedHelpers;
import de.robv.android.xposed.callbacks.XC_LoadPackage.LoadPackageParam;

/**
 * ---------------------------------------------------------------------------
 * 【Xposed 模块入口】
 * ---------------------------------------------------------------------------
 * 流程：
 *   1. 过滤目标包名 com.tencent.mm
 *   2. 过滤非主进程（微信有 :tools/:push/:appbrand 等大量子进程，无需重复加载）
 *   3. Hook Application.attachBaseContext，在微信核心初始化前拿到 Context
 *   4. 初始化 InstanceManager，识别当前是主微信还是哪个分身
 *   5. 实例总开关关闭则直接 return，不加载任何 Hook
 *   6. 加载功能模块（每个功能内部还有独立开关）
 * ---------------------------------------------------------------------------
 */
public class MainHook implements IXposedHookLoadPackage {

    private static final String TAG = "WeChatIso";
    private static final String WECHAT_PKG = "com.tencent.mm";

    @Override
    public void handleLoadPackage(LoadPackageParam lpparam) throws Throwable {
        if (!WECHAT_PKG.equals(lpparam.packageName)) return;

        // 【进程过滤】只处理微信主进程，避免子进程重复 init / 多进程写 SP 冲突
        if (!lpparam.processName.equals(lpparam.packageName)) {
            XposedBridge.log("[" + TAG + "] Skip non-main process: " + lpparam.processName);
            return;
        }

        XposedBridge.log("[" + TAG + "] Injecting main process: " + lpparam.processName);

        // Hook Application.attachBaseContext 拿最原始的 Context
        // 注意：android.app.Application 是系统类，classLoader 传 null（由 BootClassLoader 查找）
        XposedHelpers.findAndHookMethod(
                "android.app.Application",
                null,
                "attachBaseContext",
                Context.class,
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam param) throws Throwable {
                        Context ctx = (Context) param.args[0];

                        // 初始化实例隔离管理器
                        InstanceManager.init(ctx);

                        XposedBridge.log("[" + TAG + "] Instance ready: userId="
                                + InstanceManager.userId()
                                + ", primary=" + InstanceManager.isPrimary()
                                + ", enabled=" + InstanceManager.isEnabled()
                                + ", dataDir=" + InstanceManager.dataDir());

                        // 实例总开关：主微信默认开，分身默认关（可在各自 SP 中修改）
                        if (!InstanceManager.isEnabled()) {
                            XposedBridge.log("[" + TAG + "] Instance disabled, skip all features.");
                            return;
                        }

                        // 加载功能，各功能内部用 HookHelper 做二级开关
                        FeatureLoader.load(lpparam);
                    }
                });
    }
}

package com.yourname.wechatiso;

import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.XposedHelpers;

/**
 * ---------------------------------------------------------------------------
 * 【Hook 辅助类】
 * ---------------------------------------------------------------------------
 * 在标准 XposedHelpers 之上增加【实例级开关】。
 * 只有当前 user 的微信打开了对应 feature，才会真正注册 Hook。
 * 确保主微信与分身微信的 Hook 完全独立，互不干扰。
 *
 * 所有业务 Hook 必须通过本类注册，不要直接用 XposedHelpers.findAndHookMethod。
 * ---------------------------------------------------------------------------
 */
public final class HookHelper {

    /** 调试日志开关，发包时可改 false 关闭日志 */
    public static final boolean DEBUG = true;

    private HookHelper() {}

    private static void log(String msg) {
        if (DEBUG) XposedBridge.log("[WeChatIso] " + msg);
    }

    /**
     * 带实例开关的方法 Hook。
     *
     * @param featureKey     功能开关 key，对应 InstanceManager.feature(key)
     * @param className      目标类全名（混淆后用 DexKit 定位后填入）
     * @param cl             类加载器，传 LoadPackageParam.classLoader
     * @param methodName     目标方法名
     * @param callback       Hook 回调
     * @param parameterTypes 方法参数类型（无参数则留空），如 String.class, long.class
     */
    public static void hookIfEnabled(String featureKey,
                                      String className,
                                      ClassLoader cl,
                                      String methodName,
                                      XC_MethodHook callback,
                                      Object... parameterTypes) {
        if (!InstanceManager.feature(featureKey)) {
            log("feature '" + featureKey + "' disabled for userId="
                    + InstanceManager.userId() + ", skip " + className + "." + methodName);
            return;
        }

        try {
            // XposedHelpers 签名：callback 在前，parameterTypes 在后
            XposedHelpers.findAndHookMethod(className, cl, methodName, callback, parameterTypes);
            log("hooked: " + className + "." + methodName + " (userId=" + InstanceManager.userId() + ")");
        } catch (Throwable t) {
            log("hook FAILED: " + className + "." + methodName);
            XposedBridge.log(t);
        }
    }

    /**
     * 带实例开关的构造器 Hook。
     */
    public static void hookConstructorIfEnabled(String featureKey,
                                                 String className,
                                                 ClassLoader cl,
                                                 XC_MethodHook callback,
                                                 Object... parameterTypes) {
        if (!InstanceManager.feature(featureKey)) return;
        try {
            XposedHelpers.findAndHookConstructor(className, cl, callback, parameterTypes);
            log("hooked constructor: " + className + " (userId=" + InstanceManager.userId() + ")");
        } catch (Throwable t) {
            log("constructor hook FAILED: " + className);
            XposedBridge.log(t);
        }
    }
}

package com.yourname.wechatiso;

import de.robv.android.xposed.callbacks.XC_LoadPackage.LoadPackageParam;

/**
 * ---------------------------------------------------------------------------
 * 【功能加载器】
 * ---------------------------------------------------------------------------
 * 在这里注册你的微信功能模块。每加一个功能，init 一下即可。
 * 每个功能内部必须使用 HookHelper.hookIfEnabled，确保实例级隔离。
 *
 * 适配说明：
 *   - 新增功能时，建议用 try-catch 包住，防止单个功能类名写错导致模块整体崩溃。
 * ---------------------------------------------------------------------------
 */
public final class FeatureLoader {

    public static void load(LoadPackageParam lp) {
        // 示例功能：防撤回（演示用，类名需自行适配）
        safeInit("AntiRevoke", () -> com.yourname.wechatiso.demo.AntiRevokeHook.init(lp));

        // TODO: 在这里继续注册你的功能，例如：
        // safeInit("StepCounter", () -> com.yourname.wechatiso.demo.StepHook.init(lp));
        // safeInit("RedPacket", () -> com.yourname.wechatiso.demo.RedPacketHook.init(lp));
    }

    /** 统一异常保护，避免单个功能崩溃影响其他功能 */
    private static void safeInit(String name, Runnable init) {
        try {
            init.run();
        } catch (Throwable t) {
            XposedBridge.log("[WeChatIso] Feature '" + name + "' init failed");
            XposedBridge.log(t);
        }
    }
}

package com.yourname.wechatiso.demo;

import com.yourname.wechatiso.HookHelper;
import com.yourname.wechatiso.InstanceManager;
import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.callbacks.XC_LoadPackage.LoadPackageParam;

/**
 * ---------------------------------------------------------------------------
 * 【示例功能：消息防撤回】
 * ---------------------------------------------------------------------------
 * 重要：微信类名高度混淆，下面的 TARGET_CLASS / TARGET_METHOD 是【占位符】。
 *       你必须用自己的 DexKit / Jadx / Smali 定位真实类名后替换。
 *
 * 定位思路：
 *   1. 搜索字符串 "revokemsg" 或日志 tag "MicroMsg.MsgRevoke"
 *   2. 找到处理撤回通知的类，通常形如 com.tencent.mm.modelmessage.xxx 或 model.xxx
 *   3. Hook 其接收/处理撤回的方法，在 beforeHookedMethod 里 param.setResult(...) 阻止执行
 *
 * 隔离体现：
 *   - 本 Hook 只在当前微信实例的 feature_anti_revoke 开关打开时注册。
 *   - 主微信和分身微信的开关分别存在各自 SP 中，互不影响。
 *   - Hook 回调执行时，InstanceManager.userId() 告诉你当前是哪个实例。
 * ---------------------------------------------------------------------------
 */
public final class AntiRevokeHook {

    // TODO: 适配你的微信版本，替换为真实类名
    private static final String TARGET_CLASS = "com.tencent.mm.modelmessage.MsgRevokeService";

    // TODO: 适配你的微信版本，替换为真实方法名
    private static final String TARGET_METHOD = "onRevokeMsg";

    // 功能开关 key，存在各实例独立的 SP 中
    private static final String FEATURE_KEY = "feature_anti_revoke";

    public static void init(LoadPackageParam lp) {
        HookHelper.hookIfEnabled(
                FEATURE_KEY,
                TARGET_CLASS,
                lp.classLoader,
                TARGET_METHOD,
                new XC_MethodHook() {
                    @Override
                    protected void beforeHookedMethod(MethodHookParam param) throws Throwable {
                        // 当前实例身份（0=主微信，>0=分身）
                        int userId = InstanceManager.userId();

                        XposedBridge.log("[WeChatIso] AntiRevoke triggered in userId=" + userId);

                        // -------------------------------------------------------------------
                        // 防撤回核心逻辑（根据实际 Hook 点调整）：
                        // 如果是 void 方法且想阻止撤回，直接 setResult(null) 即可。
                        // 如果方法有非 null 返回类型，必须返回对应类型的值，例如：
                        //   param.setResult(false);  // boolean 返回
                        //   param.setResult(0L);     // long 返回
                        // -------------------------------------------------------------------
                        param.setResult(null);
                    }
                }
                // 如果目标方法有参数，在后面追加参数类型，例如：
                // , String.class, long.class
        );
    }
}
