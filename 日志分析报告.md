class WxBubbleModule : IXposedHookLoadPackage {
    override fun handleLoadPackage(lp: XC_LoadPackage.LoadPackageParam) {
        if (lp.packageName != PKG) return
        WxEnv.lp = lp
        WxEnv.apkPath = lp.appInfo?.sourceDir ?: return

        // ❌ 删掉：Thread({ WxBoot.start() }, "WxBubble-boot").start()
        // ✅ 改为：等微信真实 ClassLoader 就绪

        // ① 主路径：attachBaseContext 后 CL 可用（3180 实测）
        XposedHelpers.findAndHookMethod("android.content.ContextWrapper", lp.classLoader,
            "attachBaseContext", android.content.Context::class.java, object : XC_MethodHook() {
            override fun afterHookedMethod(p: MethodHookParam) {
                val ctx = p.thisObject as? android.content.Context ?: return
                if (ctx !is android.app.Application) return
                WxEnv.adopt(ctx.classLoader, "attachBaseContext")
            }
        })

object WxEnv {
    lateinit var lp: XC_LoadPackage.LoadPackageParam
    var apkPath = ""; var modulePath = ""; var verTag = ""
    @Volatile var started = false

    /** 真 CL 采纳 + 自建 CL 三级兜底 + 启动（只跑一次） */
    fun adopt(cl: ClassLoader?, src: String) {
        if (started) return
        val real = pickRealCl(cl) ?: return
        CL = real
        started = true
        log("real CL adopted from $src → $real")
        Thread({ WxBoot.start() }, "WxBubble-boot").start()
    }

    @Volatile var CL: ClassLoader = lp.classLoader     // ← 所有 Reflect.load 改用它

    private fun pickRealCl(cl: ClassLoader?): ClassLoader? {
        val probe = "com.tencent.mm.ui.chatting.viewitems.bq"
        if (cl != null && cl.canLoad(probe)) return cl
        // 自建 PathClassLoader over base.apk（日志 DatabaseProvider STRATEGY A 同款，3180 可用）
        runCatching {
            val c = Class.forName("dalvik.system.PathClassLoader")
            val k = c.getConstructor(String::class.java, String::class.java, ClassLoader::class.java)
                .newInstance(apkPath, null, ClassLoader.getSystemClassLoader()) as ClassLoader
            if (k.canLoad(probe)) return k
        }
        runCatching {
            val at = XposedHelpers.callStaticMethod(
                XposedHelpers.findClass("android.app.ActivityThread", null), "currentActivityThread")
            val app = XposedHelpers.callMethod(at, "getSystemContext")
            val k = XposedHelpers.callMethod(app, "getClassLoader") as? ClassLoader
            if (k != null && k.canLoad(probe)) return k
        }
        return null
    }
    private fun ClassLoader.canLoad(n: String) =
        runCatching { loadClass(n); true }.getOrDefault(false)
}
同步改动： object Reflect { fun load(n) = ... XposedHelpers.findClass(n, WxEnv.CL) ... } （原来写死  WxEnv.cl ，改成动态  WxEnv.CL ）。
