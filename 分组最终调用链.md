    public static void installTagFilterBar(ClassLoader cl) {
        try {
            Class<?> launcherUI = XposedHelpers.findClass(
                "com.tencent.mm.ui.LauncherUI", cl);

            XposedHelpers.findAndHookMethod(launcherUI, "onResume",
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam param) {
                        new Handler(Looper.getMainLooper()).postDelayed(() -> {
                            injectTagBar((Activity) param.thisObject);
                        }, 500);
                    }
                });

            // 也监听标签变更事件来刷新
            EventBus.subscribe(EventBus.Event.LABEL_CREATED,  (e, d) -> refreshTagBar());
            EventBus.subscribe(EventBus.Event.LABEL_DELETED,  (e, d) -> refreshTagBar());
            EventBus.subscribe(EventBus.Event.LABEL_RENAMED,  (e, d) -> refreshTagBar());
            EventBus.subscribe(EventBus.Event.LABELS_SYNCED,  (e, d) -> refreshTagBar());

            XposedBridge.log(TAG + ": ✓ 标签筛选栏 Hook 已安装");
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": ✗ installTagFilterBar: " + e.getMessage());
        }
    }

    /**
     * 执行注入 —— 在 ActionBarContainer 正下方插入标签栏
     */
    private static void injectTagBar(Activity activity) {
        try {
            if (activity == null) return;

            // 已注入且仍在视图树中则跳过
            if (sTagBarView != null && sTagBarView.getParent() != null) {
                refreshTagBar();
                return;
            }

            // ★ 关键: 通过资源ID找到 ActionBarContainer ★
            View actionBarContainer = activity.findViewById(ACTION_BAR_CONTAINER_ID);
            if (actionBarContainer == null) {
                // fallback: 从 decorView 逐层查找
                actionBarContainer = findActionBarByTraversal(
                    activity.getWindow().getDecorView());
            }
            if (actionBarContainer == null) return;

            ViewGroup parent = (ViewGroup) actionBarContainer.getParent();
            if (parent == null) return;

            // 计算插入位置: ActionBarContainer 的下一个位置
            int insertIndex = parent.indexOfChild(actionBarContainer) + 1;

            // 避免重复注入
            if (insertIndex < parent.getChildCount()) {
                View existing = parent.getChildAt(insertIndex);
                if (existing != null && existing.getTag() != null
                    && "wxgroup_tagbar".equals(existing.getTag())) {
                    sTagBarView = existing;
                    refreshTagBar();
                    return;
                }
            }

            // 构建标签栏
            sTagBarView = buildTagBar(activity);

            // 标记以便识别
            sTagBarView.setTag("wxgroup_tagbar");

            // 插入到 ActionBar 正下方
            parent.addView(sTagBarView, insertIndex,
                new ViewGroup.LayoutParams(
                    ViewGroup.LayoutParams.MATCH_PARENT,
                    dp(44, activity)));

            refreshTagBar();
            XposedBridge.log(TAG + ": ✓ 标签栏已注入到 ActionBar 下方 (index=" + insertIndex + ")");

        } catch (Throwable e) {
            XposedBridge.log(TAG + ": injectTagBar error: " + e.getMessage());
        }
    }

    /**
     * 备用方案: 从 DecorView 遍历查找 ActionBarContainer
     */
    private static View findActionBarByTraversal(View root) {
        if (root.getId() == ACTION_BAR_CONTAINER_ID) return root;
        if (root instanceof ViewGroup) {
            ViewGroup vg = (ViewGroup) root;
            for (int i = 0; i < vg.getChildCount(); i++) {
                View found = findActionBarByTraversal(vg.getChildAt(i));
                if (found != null) return found;
            }
        }
        return null;
    }

    private static View buildTagBar(Context ctx) {
        HorizontalScrollView hsv = new HorizontalScrollView(ctx);
        hsv.setHorizontalScrollBarEnabled(false);
        hsv.setBackgroundColor(Color.parseColor("#EDEDED"));

        // 底部分割线
        View divider = new View(ctx);
        divider.setBackgroundColor(Color.parseColor("#D9D9D9"));
        // 分割线通过 elevation 实现

        LinearLayout container = new LinearLayout(ctx);
        container.setOrientation(LinearLayout.HORIZONTAL);
        container.setGravity(Gravity.CENTER_VERTICAL);
        int padH = dp(10, ctx);
        int padV = dp(6, ctx);
        container.setPadding(padH, padV, padH, padV);
        hsv.addView(container);
        sTagContainer = container;

        return hsv;
    }

    private static void refreshTagBar() {
        if (sTagContainer == null) return;
        sCachedLabels = LabelManager.getAll();
        Context ctx = sTagContainer.getContext();
        sTagContainer.removeAllViews();

        // "全部" chip
        addChip(ctx, "全部", -1, sSelectedLabelId == -1);
        addDivider(ctx);

        for (LabelInfo label : sCachedLabels) {
            addChip(ctx, label.labelName, label.labelId,
                sSelectedLabelId == label.labelId);
        }

        addDivider(ctx);
        addPlusButton(ctx);
    }

    private static void addChip(Context ctx, String text, int labelId, boolean selected) {
        TextView tv = new TextView(ctx);
        tv.setText(text);
        tv.setTextSize(13);
        tv.setPadding(dp(12, ctx), dp(5, ctx), dp(12, ctx), dp(5, ctx));
        tv.setGravity(Gravity.CENTER);
        tv.setSingleLine(true);
        tv.setMaxLines(1);
        tv.setEllipsize(android.text.TextUtils.TruncateAt.END);
        tv.setMaxWidth(dp(120, ctx));

        GradientDrawable bg = new GradientDrawable();
        bg.setCornerRadius(dp(14, ctx));
        if (selected) {
            bg.setColor(Color.parseColor("#07C160"));
            tv.setTextColor(Color.WHITE);
        } else {
            bg.setColor(Color.WHITE);
            tv.setTextColor(Color.parseColor("#353535"));
            bg.setStroke(dp(1, ctx), Color.parseColor("#D9D9D9"));
        }
        tv.setBackground(bg);

        LinearLayout.LayoutParams lp = new LinearLayout.LayoutParams(
            ViewGroup.LayoutParams.WRAP_CONTENT, ViewGroup.LayoutParams.WRAP_CONTENT);
        int m = dp(4, ctx);
        lp.setMargins(m, 0, m, 0);
        tv.setLayoutParams(lp);

        // 点击筛选
        tv.setOnClickListener(v -> {
            if (sSelectedLabelId == labelId) {
                sSelectedLabelId = -1; // 取消选中
            } else {
                sSelectedLabelId = labelId;
            }
            refreshTagBar();
            onFilterChanged(sSelectedLabelId);
        });

        // 长按管理标签
        if (labelId > 0) {
            tv.setOnLongClickListener(v -> {
                showLabelManageDialog(ctx, labelId, text);
                return true;
            });
        }

        sTagContainer.addView(tv);
    }

    private static void addPlusButton(Context ctx) {
        TextView tv = new TextView(ctx);
        tv.setText("＋");
        tv.setTextSize(18);
        tv.setTextColor(Color.parseColor("#07C160"));
        tv.setPadding(dp(8, ctx), dp(3, ctx), dp(8, ctx), dp(3, ctx));
        tv.setGravity(Gravity.CENTER);

        GradientDrawable bg = new GradientDrawable();
        bg.setCornerRadius(dp(14, ctx));
        bg.setColor(Color.WHITE);
        bg.setStroke(dp(1, ctx), Color.parseColor("#07C160"));
        tv.setBackground(bg);

        tv.setOnClickListener(v -> showCreateLabelDialog(ctx, null));
        sTagContainer.addView(tv);
    }

    private static void addDivider(Context ctx) {
        View v = new View(ctx);
        v.setLayoutParams(new LinearLayout.LayoutParams(dp(2, ctx),
            ViewGroup.LayoutParams.MATCH_PARENT));
        sTagContainer.addView(v);
    }

    private static void onFilterChanged(int labelId) {
        if (sTagContainer == null) return;
        String msg;
        if (labelId <= 0) {
            msg = "显示全部会话";
        } else {
            List<String> contacts = LabelManager.getContacts(labelId);
            msg = "「" + getLabelName(labelId) + "」— " + contacts.size() + " 人";
        }
        Toast.makeText(sTagContainer.getContext(), msg, Toast.LENGTH_SHORT).show();
    }

    private static String getLabelName(int labelId) {
        if (sCachedLabels != null) {
            for (LabelInfo l : sCachedLabels) {
                if (l.labelId == labelId) return l.labelName;
            }
        }
        return "分组" + labelId;
    }

    /**
     * 在微信原生菜单之后注入"分组管理"菜单项
     */
    private static void injectContextMenu(MethodHookParam param) {
        try {
            ContextMenu menu = (ContextMenu) param.args[0];
            Object r3Instance = param.thisObject;

            String username = (String) XposedHelpers.getObjectField(r3Instance, "g");
            Activity activity = (Activity) XposedHelpers.getObjectField(r3Instance, "f");
            if (TextUtils.isEmpty(username) || activity == null) return;

            // 获取该联系人当前的标签
            int[] currentIds = ContactHelper.getLabelIds(username);
            Set<Integer> currentSet = new HashSet<>();
            for (int id : currentIds) currentSet.add(id);

            List<LabelInfo> allLabels = LabelManager.getAll();
            if (allLabels.isEmpty()) return;

            ContextMenu.ContextMenuInfo info =
                (ContextMenu.ContextMenuInfo) param.args[2];
            int position = 0;
            if (info instanceof AdapterView.AdapterContextMenuInfo) {
                position = ((AdapterView.AdapterContextMenuInfo) info).position;
            }

            // ---- 显示前8个标签 ----
            int showCount = Math.min(allLabels.size(), 8);
            for (int i = 0; i < showCount; i++) {
                final LabelInfo label = allLabels.get(i);
                final boolean hasLabel = currentSet.contains(label.labelId);
                final String title = (hasLabel ? "☑ " : "☐ ") + label.labelName;

                MenuItem item = menu.add(
                    position,           // groupId (使用 position)
                    MENU_ADD_TO_GROUP + label.labelId,  // 唯一 itemId
                    0,                  // order
                    title               // title
                );

                // ★ 关键: 直接在 MenuItem 上设置监听器 ★
                item.setOnMenuItemClickListener(clicked -> {
                    if (hasLabel) {
                        ContactHelper.removeLabel(username, label.labelId);
                        toast(activity, "已从「" + label.labelName + "」移除");
                    } else {
                        ContactHelper.addLabel(username, label.labelId);
                        toast(activity, "已添加到「" + label.labelName + "」");
                    }
                    return true;
                });
            }

            // ---- "管理所有分组..." ----
            if (allLabels.size() > 8) {
                MenuItem manageItem = menu.add(
                    position, MENU_MANAGE_GROUPS, showCount, "管理所有分组...");
                manageItem.setOnMenuItemClickListener(clicked -> {
                    showFullGroupDialog(activity, username, currentSet, allLabels);
                    return true;
                });
            }

            // ---- "＋ 新建分组" ----
            MenuItem createItem = menu.add(
                position, MENU_CREATE_GROUP, showCount + 1, "＋ 新建分组");
            createItem.setOnMenuItemClickListener(clicked -> {
                showCreateLabelDialog(activity, username);
                return true;
            });

        } catch (Throwable e) {
            XposedBridge.log(TAG + ": injectContextMenu error: " + e.getMessage());
        }
    }

    private static void showFullGroupDialog(Context ctx, String username,
                                            Set<Integer> currentSet,
                                            List<LabelInfo> allLabels) {
        String[] names = new String[allLabels.size()];
        boolean[] checked = new boolean[allLabels.size()];
        for (int i = 0; i < allLabels.size(); i++) {
            names[i] = allLabels.get(i).labelName;
            checked[i] = currentSet.contains(allLabels.get(i).labelId);
        }

        new AlertDialog.Builder(ctx)
            .setTitle("管理分组 — " + truncate(username, 20))
            .setMultiChoiceItems(names, checked, (dialog, which, isChecked) -> {
                LabelInfo label = allLabels.get(which);
                if (isChecked) {
                    ContactHelper.addLabel(username, label.labelId);
                } else {
                    ContactHelper.removeLabel(username, label.labelId);
                }
            })
            .setPositiveButton("完成", null)
            .setNeutralButton("＋ 新建", (d, w) -> showCreateLabelDialog(ctx, username))
            .show();
    }

    private static void showCreateLabelDialog(Context ctx, String username) {
        EditText input = new EditText(ctx);
        input.setHint("输入分组名称");
        input.setPadding(50, 35, 50, 35);

        new AlertDialog.Builder(ctx)
            .setTitle("新建分组")
            .setView(input)
            .setPositiveButton("创建", (dialog, which) -> {
                String name = input.getText().toString().trim();
                if (!TextUtils.isEmpty(name)) {
                    LabelInfo created = LabelManager.create(name);
                    if (created != null) {
                        if (username != null) {
                            ContactHelper.addLabel(username, created.labelId);
                        }
                        refreshTagBar();
                        toast(ctx, "已创建「" + name + "」");
                    } else {
                        toast(ctx, "创建失败（可能已存在同名分组）");
                    }
                }
            })
            .setNegativeButton("取消", null)
            .show();
    }

    private static void showLabelManageDialog(Context ctx, int labelId, String labelName) {
        int contactCount = LabelManager.getContacts(labelId).size();
        String[] options = {
            "查看联系人 (" + contactCount + "人)",
            "重命名",
            "删除分组",
            "合并到其他分组"
        };

        new AlertDialog.Builder(ctx)
            .setTitle(labelName)
            .setItems(options, (dialog, which) -> {
                switch (which) {
                    case 0: { // 查看联系人
                        List<String> contacts = LabelManager.getContacts(labelId);
                        StringBuilder sb = new StringBuilder();
                        int max = Math.min(contacts.size(), 50);
                        for (int i = 0; i < max; i++) {
                            if (i > 0) sb.append("\n");
                            sb.append(truncate(contacts.get(i), 30));
                        }
                        if (contacts.size() > 50)
                            sb.append("\n\n... 共 ").append(contacts.size()).append(" 人");
                        new AlertDialog.Builder(ctx)
                            .setTitle(labelName + " (" + contacts.size() + "人)")
                            .setMessage(sb.toString())
                            .setPositiveButton("关闭", null)
                            .show();
                        break;
                    }
                    case 1: { // 重命名
                        EditText et = new EditText(ctx);
                        et.setText(labelName);
                        et.setPadding(50, 35, 50, 35);
                        new AlertDialog.Builder(ctx)
                            .setTitle("重命名")
                            .setView(et)
                            .setPositiveButton("确定", (d, w) -> {
                                String n = et.getText().toString().trim();
                                if (!TextUtils.isEmpty(n)) {
                                    LabelManager.rename(String.valueOf(labelId), n);
                                    refreshTagBar();
                                }
                            })
                            .setNegativeButton("取消", null)
                            .show();
                        break;
                    }
                    case 2: // 删除
                        new AlertDialog.Builder(ctx)
                            .setTitle("删除分组")
                            .setMessage("确定删除「" + labelName + "」？\n联系人关联将被清除，标签本身会被删除。")
                            .setPositiveButton("删除", (d, w) -> {
                                LabelManager.delete(String.valueOf(labelId));
                                if (sSelectedLabelId == labelId) sSelectedLabelId = -1;
                                refreshTagBar();
                            })
                            .setNegativeButton("取消", null)
                            .show();
                        break;
                    case 3: { // 合并
                        List<LabelInfo> labels = LabelManager.getAll();
                        String[] ns = new String[labels.size()];
                        for (int i = 0; i < labels.size(); i++)
                            ns[i] = labels.get(i).labelName;
                        new AlertDialog.Builder(ctx)
                            .setTitle("合并「" + labelName + "」到...")
                            .setItems(ns, (d, i2) -> {
                                LabelInfo tgt = labels.get(i2);
                                if (tgt.labelId != labelId) {
                                    BatchOperator.mergeLabels(labelId, tgt.labelId);
                                    if (sSelectedLabelId == labelId) sSelectedLabelId = -1;
                                    refreshTagBar();
                                }
                            })
                            .setNegativeButton("取消", null)
                            .show();
                        break;
                    }
                }
            })
            .setNegativeButton("取消", null)
            .show();
    }

    private static void toast(Context ctx, String msg) {
        Toast.makeText(ctx, msg, Toast.LENGTH_SHORT).show();
    }

    private static String truncate(String s, int maxLen) {
        if (s == null) return "";
        return s.length() > maxLen ? s.substring(0, maxLen) + "..." : s;
    }

    private static int dp(int dpi, Context ctx) {
        return (int) (dpi * ctx.getResources().getDisplayMetrics().density + 0.5f);
    }
