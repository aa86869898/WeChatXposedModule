import android.content.Context;

/** 全局配置；微信进程内单例，跨 Activity 共享。 */
public final class Config {

    public static final String PREFS = "wx_corner_cfg";
    public static final String K_SCOPE  = "scope";     // all | chat | conversation
    public static final String K_TOAST  = "toast";
    public static final String K_DEBUG  = "debug";

    /** DexKit 自检上报（主进程写，模块 UI 读） */
    public static final String K_REPORT = "dexkit_report";
    public static final String K_HIT    = "hook_hit";

    private static volatile Config sInst;

    private final SharedPreferences sp;

    private Config(Context ctx) {
        sp = ctx.getApplicationContext()
                .getSharedPreferences(PREFS, Context.MODE_PRIVATE);
    }

    public static Config get(Context ctx) {
        Config c = sInst;
        if (c == null) {
            synchronized (Config.class) {
                c = sInst;
                if (c == null) {
                    c = new Config(ctx);
                    sInst = c;
                }
            }
        }
        return c;
    }

    public String scope() { return sp.getString(K_SCOPE, "chat"); }
    public boolean toast() { return sp.getBoolean(K_TOAST, true); }
    public boolean debug() { return sp.getBoolean(K_DEBUG, true); }

    public void setScope(String v)   { sp.edit().putString(K_SCOPE, v).apply(); }
    public void setToast(boolean v)  { sp.edit().putBoolean(K_TOAST, v).apply(); }
    public void setDebug(boolean v) { sp.edit().putBoolean(K_DEBUG, v).apply(); }

    public void putReport(String report) {
        sp.edit().putString(K_REPORT, report).apply();
    }

    public String report() { return sp.getString(K_REPORT, ""); }

    public void hit(String where) {
        int n = sp.getInt(K_HIT, 0) + 1;

    }

    public String lastHit() { return sp.getString(K_HIT + "_last", "-"); }

package com.example.wxcorner.targets;

import android.content.Context;
import android.content.pm.ApplicationInfo;
import android.content.pm.PackageInfo;
import android.content.pm.PackageManager;
import android.util.Log;

import com.luckypray.dexkit.DexKitBridge;
import com.luckypray.dexkit.result.ClassDef;
import com.luckypray.dexkit.result.MethodDef;
import com.luckypray.dexkit.query.FindClassQuery;
import com.luckypray.dexkit.query.FindMethodQuery;
import com.luckypray.dexkit.query.enums.MethodUsingType;

import java.io.File;
import java.lang.reflect.Constructor;
import java.lang.reflect.Method;
import java.util.List;

import de.robv.android.xposed.XposedHelpers;

/**
 * 微信更新后的动态适配核心。
 *
 * 定位策略（全部基于「不随混淆变化的字符串常量」与「类结构特征」）：
 *   1. 主目标：com.tencent.mm.ui.j  → ClassFinder("MicroMsg.ActionBarCustomArea")
 *   2. 降级  ：HeaderComponent       → ClassFinder("MicroMsg.ChattingUI.HeaderComponent")
 *              + MethodFinder("handleActionBar isCurrentActivity=")
 *   3. 二次校验：目标类必须同时满足结构特征，避免同名日志误命中
 */
public final class DexKitResolver {

    private static final String TAG = "WxCorner/DexKit";
    private static final String PKG = "com.tencent.mm";

    public static Resolved resolve(Context ctx) {
        Resolved r = new Resolved();
        try {
            String apk = apkPath(ctx);
            if (apk == null) {
                r.t("[FATAL] 无法定位微信 APK 路径");
                return r;
            }
            r.t("APK = " + apk);
            DexKitBridge.create(apk);
            r.t("DexKitBridge.create() OK");

            resolveActivityNames(ctx, r);
            resolveCustomArea(r);
            resolveHeaderComponent(r);

            r.t("--- RESULT ---");
            r.t("primary(customArea)   = " + (r.customAreaCtor != null));
            r.t("secondary(headerComp) = " + (r.headerComponentInit != null));
        } catch (Throwable t) {
            r.t("[EX] " + t);
            Log.e(TAG, "resolve failed", t);
        }
        return r;
    }

    // ------------------------------------------------------------------ APK

    private static String apkPath(Context ctx) {
        try {
            ApplicationInfo ai = ctx.getPackageManager()
                    .getApplicationInfo(PKG, PackageManager.GET_META_DATA);
            return new File(ai.sourceDir).getAbsolutePath();
        } catch (Throwable t) {
            return null;
        }
    }

    // ------------------------------------------------------------ 解析 Activity 名

    /** 通过 log tag 常量反查类名，避免硬编码混淆名。 */
    private static void resolveActivityNames(Context ctx, Resolved r) {
        try {
            ClassDef c = DexKitBridge.findClassUsingStrings("MicroMsg.LauncherUI")
                    .single();
            r.launcherActivityName = c.getName();
            r.t("LauncherActivity  = " + r.launcherActivityName);
        } catch (Throwable t) {
            // 兜底：直接用包名字符串
            r.launcherActivityName = PKG + ".ui.LauncherUI";
            r.t("LauncherActivity  = 硬编码兜底 (" + t + ")");
        }

        try {
            ClassDef c = DexKitBridge.findClassUsingStrings("MicroMsg.ChattingUI")
                    .single();
            r.chattingActivityName = c.getName();
            r.t("ChattingActivity   = " + r.chattingActivityName);
            // 变体包前缀
            String pkg = r.chattingActivityName;
            pkg = pkg.substring(0, pkg.lastIndexOf('.'));
            r.chattingPackagePrefix = pkg + ".";
            r.t("ChattingPrefix     = " + r.chattingPackagePrefix);
        } catch (Throwable t) {
            r.chattingActivityName = PKG + ".ui.chatting.ChattingUI";
            r.t("ChattingActivity   = 硬编码兜底 (" + t + ")");
        }
    }

    // --------------------------------------------- 解析主目标 com.tencent.mm.ui.j

    /**
     * com.tencent.mm.ui.j（混淆名会变，靠字符串 + 结构定位）
     * 特征：持有字段类型 com.tencent.mm.ui.widget.imageview.WeImageView 的字段
     *       持有 21 个 View 字段
     *       有 <init>(Landroid/view/View;)V
     */
    private static void resolveCustomArea(Resolved r) {
        // —— 尝试 A：字符串锚点 ——
        List<ClassDef> cands = new ArrayList<>();
        try {
            cands = DexKitBridge.findClassUsingStrings("MicroMsg.ActionBarCustomArea")
                    .list();
            r.t("[A] findClassUsingStrings(ActionBarCustomArea) = " + cands.size());
        } catch (Throwable t) {
            r.t("[A] findClassUsingStrings 异常: " + t);
        }

        // —— 尝试 B：结构锚点 ——
        // 「布局 XML inflate + 一次 findViewById 绑定 20+ 子控件」的唯一类：
        // 构造函数参数 1 个 View，且类内有多个 View/WeImageView/TextView 字段
        try {
            List<ClassDef> struct = DexKitBridge.findClasses(
                            new FindClassQuery()
                                    .methodUsingStrings("expand back touch right:%s")
                                    .methodParamTypes("(Landroid/view/View;)V"))
                    .list();
            r.t("[B] findClasses(expand back touch right) = " + struct.size());
            for (ClassDef cd : struct) {
                if (looksLikeCustomArea(cd)) {
                    r.t("[B] 结构命中: " + cd.getName());
                    cands.add(cd);
                }
            }
        } catch (Throwable t) {
            r.t("[B] 结构查找异常: " + t);
        }

        // —— 逐个尝试拿到 <init>(View) 并做语义校验 ——
        for (ClassDef cd : cands) {
            try {
                Class<?> clazz = XposedHelpers.findClass(cd.getName(),
                        DexKitResolver.class.getClassLoader());
                Constructor<?> ctor = clazz.getDeclaredConstructor(View.class);
                // 语义校验：必须存在 g(CharSequence)（设标题）与 c(OnClickListener)（设返回键）
                clazz.getDeclaredMethod("g", CharSequence.class);
                clazz.getDeclaredMethod("c", View.OnClickListener.class);
                r.customAreaCtor = ctor;
                r.t("PRIMARY OK: " + cd.getName() + "#<init>(View)");
                return;
            } catch (Throwable t) {
                r.t("候选 " + cd.getName() + " 校验失败: " + t);
            }
        }
        r.t("PRIMARY MISS");
    }

    /** 结构特征校验：View 字段数 >= 10，且至少一个 WeImageView 字段。 */
    private static boolean looksLikeCustomArea(ClassDef cd) {
        try {
            int viewFields = 0;
            boolean hasWeImageView = false;
            for (String typeName : cd.getFields().stream()
                    .map(f -> f.getType()).toArray(String[]::new)) {
                if ("android.view.View".equals(typeName)
                        || "android.view.ViewGroup".equals(typeName)
                        || "android.widget.TextView".equals(typeName)
                        || "android.widget.ImageView".equals(typeName)
                        || "com.tencent.mm.ui.widget.imageview.WeImageView".equals(typeName)) {
                    viewFields++;
                }
                if (typeName != null && typeName.endsWith("WeImageView")) {
                    hasWeImageView = true;
                }
            }
            return viewFields >= 10 && hasWeImageView;
        } catch (Throwable t) {
            return false;
        }
    }

    // ---------------------------------------- 降级目标 HeaderComponent#F0()

    private static void resolveHeaderComponent(Resolved r) {
        try {
            List<ClassDef> cands = DexKitBridge.findClassUsingStrings(
                            "MicroMsg.ChattingUI.HeaderComponent").list();
            r.t("[C] findClassUsingStrings(HeaderComponent) = " + cands.size());
            for (ClassDef cd : cands) {
                try {
                    Class<?> clazz = XposedHelpers.findClass(cd.getName(),
                            DexKitResolver.class.getClassLoader());
                    Method m = DexKitBridge.findMethodUsingStrings(
                                    "handleActionBar isCurrentActivity=")
                            .inClass(clazz)
                            .single();
                    // 反射拿到实际方法（methodName 才是 obfuscated 真名）
                    Method real = findNoArgVoid(clazz, m.getName());
                    if (real != null) {
                        r.headerComponentClass = clazz;
                        r.headerComponentInit = real;
                        r.t("SECONDARY OK: " + cd.getName() + "#" + real.getName() + "()");
                        return;
                    }
                } catch (Throwable t) {
                    r.t("[C] 候选 " + cd.getName() + " 失败: " + t);
                }
            }
        } catch (Throwable t) {
            r.t("[C] 异常: " + t);
        }
        r.t("SECONDARY MISS");
    }

    private static Method findNoArgVoid(Class<?> c, String name) {
        for (Method m : c.getDeclaredMethods()) {
            if (m.getName().equals(name) && m.getParameterCount() == 0
                    && m.getReturnType() == void.class) {
                return m;
            }
        }
        return null;
    }
}

package com.example.wxcorner.inject;

import android.app.Activity;
import android.content.Context;
import android.content.ContextWrapper;

public final class Scope {

    private final String mode;                 // all | chat | conversation
    private final String launcherName;         // com.tencent.mm.ui.LauncherUI
    private final String chattingName;         // com.tencent.mm.ui.chatting.ChattingUI
    private final String chattingPrefix;       // com.tencent.mm.ui.chatting.

    public Scope(String mode, String launcherName,
                 String chattingName, String chattingPrefix) {
        this.mode = mode == null ? "chat" : mode;
        this.launcherName = launcherName;
        this.chattingName = chattingName;
        this.chattingPrefix = chattingPrefix;
    }

    public boolean accept(Context ctx) {
        if ("all".equals(mode)) return true;
        Activity act = activityOf(ctx);
        if (act == null) return false;
        String n = act.getClass().getName();
        if ("conversation".equals(mode)) {
            return n.equals(launcherName);
        }
        // 默认：仅聊天页（含 variants / 半屏 / 折叠屏等所有变体）
        return n.equals(chattingName) || n.startsWith(chattingPrefix);
    }

    public static Activity activityOf(Context c) {
        Context cur = c;
        int guard = 0;
        while (cur instanceof ContextWrapper && guard++ < 12) {
            if (cur instanceof Activity) return (Activity) cur;
            Context base = ((ContextWrapper) cur).getBaseContext();
            if (base == null) return null;
            cur = base;
        }
        return null;
    }

/**
 * 标题栏左上角按钮注入器。
 *
 * 关键设计：
 *  1) 零资源 ID 依赖 —— 参考 root.getChildAt(0)（微信返回键）复制 LayoutParams
 *  2) 幂等           —— 用 View.setTag(Object) 打标记，遍历子 view 查重
 *  3) 作用域过滤     —— 延后到 post 中剥离 ContextWrapper 取 Activity 判定
 *  4) 不动宿主 tag   —— 只给「我们自己的 View」设 tag，不碰 root 的 tag
 *     （ActionBarOverlayLayout 用 setTag 存 WindowCallback，动了会崩）
 */
public final class CustomBarInjector {

    private static final String TAG = "WxCorner/Inject";
    /** 用自定义 Class 实例作为标记对象，天然唯一且不会与宿主 tag 冲突 */
    private static final Object MARK = new Object() {
        @Override public String toString() { return "WxCorner::injected"; }
    };

    private final Config cfg;
    private final Scope scope;
    private final int dpWidth;

    public CustomBarInjector(Config cfg, Scope scope) {
        this.cfg = cfg;
        this.scope = scope;
        this.dpWidth = dp(64, 1f);   // 触发时用 real context 覆盖
    }

    // ------------------------------------------------------------ 入口

    public void onCustomBarCreated(final View customView) {
        if (customView == null) return;
        if (!(customView instanceof ViewGroup)) {
            log("customView 不是 ViewGroup，跳过: " + customView.getClass());
            return;
        }
        // 延后一帧：此时 root 已 attach，可安全 post 做作用域判定
        customView.post(new Runnable() {
            @Override public void run() {
                try {
                    if (!scope.accept(customView.getContext())) {
                        log("作用域不匹配，跳过 (scope=" + scopeMode() + ")");
                        return;
                    }
                    inject((ViewGroup) customView);
                } catch (Throwable t) {
                    Log.e(TAG, "inject failed", t);
                }
            }
        });
    }

    // ------------------------------------------------------------ 实际插入

    private void inject(ViewGroup root) {
        if (root.getTag() == MARK) {
            return;                                   // 根节点已打标（不适用，兜底）
        }
        for (int i = 0; i < root.getChildCount(); i++) {
            View c = root.getChildAt(i);
            if (c != null && MARK.equals(c.getTag(MARK_KEY))) {
                log("已存在注入按钮，跳过（幂等）");
                return;
            }
        }

        Context ctx = root.getContext();
        ImageView btn = new ImageView(ctx);
        CornerDrawable cd = new CornerDrawable();
        Drawable d = cd;
        btn.setImageDrawable(d);
        btn.setScaleType(ImageView.ScaleType.FIT_CENTER);
        int p = dp(ctx, 18);
        btn.setPadding(p, p, p, p);
        btn.setClickable(true);
        btn.setFocusable(true);
        btn.setContentDescription("注入按钮");
        try {
            btn.setColorFilter(tintOf(root), PorterDuff.Mode.SRC_IN);
        } catch (Throwable ignored) { }
        btn.setTag(MARK_KEY, MARK);
        btn.setOnClickListener(new View.OnClickListener() {
            @Override public void onClick(View v) {
                cfg.hit("click");
                onButtonClick(v);
            }
        });
        btn.setOnLongClickListener(new View.OnLongClickListener() {
            @Override public boolean onLongClick(View v) {
                onButtonLongClick(v);
                return true;
            }
        });

        ViewGroup.LayoutParams lp = buildParams(root, ctx);
        root.addView(btn, 0, lp);                   // ★ index 0 = 最左侧（返回键之前）
        cfg.hit("inject");
        log("注入成功, childCount=" + root.getChildCount() + ", lp=" + lp);
    }

    // ------------------------------------------------------- LayoutParams 复制

    private ViewGroup.LayoutParams buildParams(ViewGroup root, Context ctx) {
        View ref = root.getChildCount() > 0 ? root.getChildAt(0) : null;   // 微信返回键
        ViewGroup.LayoutParams src = ref != null ? ref.getLayoutParams() : null;

        int width = dp(ctx, 40);
        int height = ViewGroup.LayoutParams.MATCH_PARENT;
        if (src != null && src.height > 0) height = src.height;

        LinearLayout.LayoutParams lp;
        if (src instanceof LinearLayout.LayoutParams) {
            lp = new LinearLayout.LayoutParams((LinearLayout.LayoutParams) src);
        } else {
            lp = new LinearLayout.LayoutParams(width, height);
        }
        lp.width  = width;
        lp.height = height;
        lp.weight = 0f;
        lp.gravity = Gravity.CENTER_VERTICAL;
        try {
            lp.setMarginStart(dp(ctx, 4));
            lp.setMarginEnd(dp(ctx, 2));
        } catch (Throwable ignored) { }
        return lp;
    }

    // -------------------------------------------------------------- 主题取色

    /** 尽量跟随微信标题文字色（深色模式自动变浅色）。 */
    private int tintOf(ViewGroup root) {
        // 1) 优先：找根下的第一个 TextView 的 currentTextColor
        try {
            View back = root.getChildAt(0);
            if (back instanceof ViewGroup) {
                ViewGroup bg = (ViewGroup) back;
                for (int i = 0; i < bg.getChildCount(); i++) {
                    View v = bg.getChildAt(i);
                    if (v instanceof android.widget.TextView) {
                        return ((android.widget.TextView) v).getCurrentTextColor();
                    }
                }
            }
        } catch (Throwable ignored) { }
        // 2) 兜底：取 window 的 statusBarColor 是否为浅色
        try {
            Context c = root.getContext();
            while (c instanceof android.content.ContextWrapper
                    && !(c instanceof android.app.Activity)) {
                c = ((android.content.ContextWrapper) c).getBaseContext();
            }
            if (c instanceof android.app.Activity) {
                int sbc = ((android.app.Activity) c).getWindow().getStatusBarColor();
                double lum = (0.299 * ((sbc >> 16) & 0xFF)
                        + 0.587 * ((sbc >> 8) & 0xFF)
                        + 0.114 * (sbc & 0xFF));
                return lum > 150 ? 0xFF333333 : 0xFFE8E8E8;
            }
        } catch (Throwable ignored) { }
        return 0xFF7A7A7A;
    }

    // -------------------------------------------------------------- 点击行为

    private void onButtonClick(View v) {
        if (cfg.toast()) {
            Context c = v.getContext();
            String title = "(标题栏已注入)";
            try {
                android.view.ViewGroup root = (android.view.ViewGroup) v.getParent();
                android.widget.TextView tv = firstTextView(root);
                if (tv != null && tv.getText() != null) {
                    title = "★ " + tv.getText();
                }
            } catch (Throwable ignored) { }
            Toast.makeText(c, title, Toast.LENGTH_SHORT).show();
        }
    }

    private void onButtonLongClick(View v) {
        try {
            Context c = v.getContext();
            Intent i = new Intent();
            i.setClassName(c, "com.example.wxcorner.ui.MainActivity");
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            c.startActivity(i);
        } catch (Throwable t) {
            Log.e(TAG, "open settings failed", t);
        }
    }

    private android.widget.TextView firstTextView(ViewGroup g) {
        if (g == null) return null;
        for (int i = 0; i < g.getChildCount(); i++) {
            View v = g.getChildAt(i);
            if (v instanceof android.widget.TextView) return (android.widget.TextView) v;
            if (v instanceof ViewGroup) {
                android.widget.TextView r = firstTextView((ViewGroup) v);
                if (r != null) return r;
            }
        }
        return null;
    }

    // ------------------------------------------------------------------ 工具

    /** 独立 tag key，避免与宿主 setTag(Object) 冲突。 */
    private static final int MARK_KEY = 0x0C0F00E1;

    private String scopeMode() { return "?"; }

    private static int dp(Context c, float v) {
        return Math.round(TypedValue.applyDimension(TypedValue.COMPLEX_UNIT_DIP, v,
                c.getResources().getDisplayMetrics()));
    }

    private static int dp(float v, float scale) { return Math.round(v * scale); }

    private static void log(String s) {
        Log.i(TAG, s);
    }

public class XposedEntry implements IXposedHookLoadPackage {

    private static final String TAG = "WxCorner";
    private static volatile Resolved sResolved;
    private static volatile CustomBarInjector sInjector;

    @Override
    public void handleLoadPackage(final XC_LoadPackage.LoadPackageParam lpparam) {
        if (!"com.tencent.mm".equals(lpparam.packageName)) return;
        XposedBridge.log(TAG + " loaded: " + lpparam.packageName);

        // 进程判定：只注入主进程，避免子进程重复注入 / 多进程 SharedPreferences 竞争
        if (!lpparam.processName.endsWith(lpparam.packageName)) {
            XposedBridge.log(TAG + " skip sub-process: " + lpparam.processName);
            return;
        }

        XposedHelpers.findAndHookMethod(Application.class, "attach",
                Context.class, new XC_MethodHook() {
            @Override protected void afterHookedMethod(MethodHookParam p) {
                Context ctx = (Context) p.args[0];
                if (!(ctx instanceof Application)) return;
                Application app = (Application) ctx;
                init(app);
            }
        });
    }

    private void init(final Application app) {
        final Config cfg = Config.get(app);
        final Resolved r = DexKitResolver.resolve(app);
        sResolved = r;
        cfg.putReport(r.trace.toString());
        XposedBridge.log(TAG + " DexKit trace:\n" + r.trace);

        if (!r.primaryReady() && !r.secondaryReady()) {
            XposedBridge.log(TAG, "!! 全部目标解析失败，模块不生效。请检查微信版本并更新锚点。");
            return;
        }

        final Scope scope = new Scope(cfg.scope(), r.launcherActivityName,
                r.chattingActivityName, r.chattingPackagePrefix);
        final CustomBarInjector injector = new CustomBarInjector(cfg, scope);
        sInjector = injector;

        // ---------------- Hook 1（主）：CustomAreaHelper#<init>(View) ----------------
        if (r.primaryReady()) {
            final Constructor<?> ctor = r.customAreaCtor;
            XposedBridge.hookMethod(ctor, new XC_MethodHook() {
                @Override protected void afterHookedMethod(MethodHookParam p) {
                    if (p.args == null || p.args.length == 0) return;
                    Object arg = p.args[0];
                    if (arg instanceof android.view.View) {
                        try {
                            injector.onCustomBarCreated((android.view.View) arg);
                        } catch (Throwable t) {
                            Log.e(TAG, "primary hook", t);
                        }
                    }
                }
            });
            XposedBridge.log(TAG, "HOOKED primary: " + ctor.getDeclaringClass().getName()
                    + "#<init>(View)");
        }

        // ---------------- Hook 2（降级）：HeaderComponent#handleActionBar() ----------------
        if (r.secondaryReady() && !r.primaryReady()) {
            Method m = r.headerComponentInit;
            XposedBridge.hookMethod(m, new XC_MethodHook() {
                @Override protected void afterHookedMethod(MethodHookParam p) {
                    try {
                        android.view.View bar = customViewOf(p.thisObject);
                        if (bar != null) injector.onCustomBarCreated(bar);
                    } catch (Throwable t) {

                    }

            });

        }
    }

     * 降级路径：从 HeaderComponent 实例中按「类型」而非「字段名」捞出 customView。
     * fg 的 View 类型字段有 E / F / q 三个，q 才是标题栏容器。
     * 判定：ViewGroup + 子节点里含 TextView（标题/未读数）。
     */
    private static android.view.View customViewOf(Object component) {
        if (component == null) return null;
        for (Class<?> c = component.getClass(); c != null && c != Object.class; c = c.getSuperclass()) {
            for (Field f : c.getDeclaredFields()) {
                if (!android.view.View.class.isAssignableFrom(f.getType())) continue;
                if (java.lang.reflect.Modifier.isStatic(f.getModifiers())) continue;
                try {
                    f.setAccessible(true);
                    Object v = f.get(component);
                    if (v instanceof android.view.ViewGroup) {
                        android.view.ViewGroup g = (android.view.ViewGroup) v;
                        if (g.getChildCount() >= 3) {
                            for (int i = 0; i < g.getChildCount(); i++) {
                                if (g.getChildAt(i) instanceof android.view.ViewGroup) return g;
                            }
                            return g;
                        }
                    }
                } catch (Throwable ignored) { }
            }
        }
        return null;
    }

附：结论摘要
1.  
可行。微信聊天页顶部标题栏是一个  LinearLayout ，其唯一构造器是  com.tencent.mm.ui.j#<init>(Landroid/view/View;)V （日志 TAG  MicroMsg.ActionBarCustomArea ），构造入参就是已 attach 的标题栏根视图。在该构造器 after 中  addView(v, 0, lp)  即可把按钮放到返回键左侧 = 左上角。
2.  
不依赖第三方插件框架：仅用标准  IXposedHookLoadPackage  +  de.robv.android.xposed:api:82 ；DexKit 仅作为定位库，不含 hook 运行时。
3.  
不依赖混淆字段名、不依赖资源 ID：按钮位置由  root.getChildAt(0) （微信返回键）推导，LayoutParams 从其复制；只依赖「日志 TAG 字符串」这一不随混淆变化的常量。
4.  
三重降级： ActionBarCustomArea  字符串 →  expand back touch right  结构特征 →  HeaderComponent#handleActionBar ；全部失败时安全失败并把定位链路写进 SharedPreferences 供自检页查看。
5.  
二次审查共修正 14 处判断错误（详见 §4），其中最关键的 3 处：
•  
走  addIconOptionMenu  会落到右上角，不是左上角；
•  
读  fg  的混淆字段  q  不可靠，应改为读  j  的构造入参；
•  
 j  有 6 个调用方，必须做 Activity 作用域过滤，且  F0()  会重入，必须做 tag 幂等。
