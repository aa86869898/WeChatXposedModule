package com.wxgroup;

import java.util.List;
import java.util.Map;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.ConcurrentHashMap;

/**
 * 模块内轻量事件总线
 * 用于 LabelSyncHook → AutoGroupEngine 等跨模块通信
 */
public class EventBus {

    public enum Event {
        /** 标签已创建 (data: int labelId, String labelName) */
        LABEL_CREATED,
        /** 标签已删除 (data: int labelId) */
        LABEL_DELETED,
        /** 标签已重命名 (data: int labelId, String newName) */
        LABEL_RENAMED,
        /** 联系人标签已变更 (data: String username) */
        CONTACT_LABEL_CHANGED,
        /** 标签列表已从服务器同步 (data: int count) */
        LABELS_SYNCED,
        /** 自动分组规则已执行 (data: AutoRule rule, int matched) */
        AUTO_RULE_EXECUTED,
    }

    public interface Listener {
        void onEvent(Event event, Object data);
    }

    private static final Map<Event, List<Listener>> listeners = new ConcurrentHashMap<>();

    public static void subscribe(Event event, Listener listener) {
        listeners.computeIfAbsent(event, k -> new CopyOnWriteArrayList<>()).add(listener);
    }

    public static void unsubscribe(Event event, Listener listener) {
        List<Listener> list = listeners.get(event);
        if (list != null) list.remove(listener);
    }

    public static void post(Event event, Object data) {
        List<Listener> list = listeners.get(event);
        if (list != null) {
            for (Listener l : list) {
                try { l.onEvent(event, data); } catch (Throwable ignored) {}
            }
        }
    }
}

package com.wxgroup;

import android.app.Application;
import android.content.Context;
import de.robv.android.xposed.IXposedHookLoadPackage;
import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.XposedHelpers;
import de.robv.android.xposed.callbacks.XC_LoadPackage;

/**
 * 总入口 — ClassLoader 捕获 + 实时 Hook 注册 + 核心服务获取
 */
public class MainHook implements IXposedHookLoadPackage {

    private static final String WECHAT = "com.tencent.mm";
    private static final String TAG = "WxGroup";

    // ===== 核心实例 =====
    private static volatile Object sLabelStorage;       // g4
    private static volatile Object sContactStorage;     // j4
    private static volatile ClassLoader sClassLoader;
    private static volatile Context sWeChatContext;
    private static volatile boolean sHooksInstalled = false;
    private static volatile boolean sInitDone = false;

    @Override
    public void handleLoadPackage(XC_LoadPackage.LoadPackageParam lpparam) {
        if (!WECHAT.equals(lpparam.packageName)) return;
        if (!lpparam.processName.equals(WECHAT)) return;

        sClassLoader = lpparam.classLoader;
        XposedBridge.log(TAG + ": [1/4] ClassLoader 捕获完成");

        // 在微信 Application.onCreate 后完成初始化
        XposedHelpers.findAndHookMethod(
            Application.class, "onCreate",
            new XC_MethodHook() {
                @Override
                protected void afterHookedMethod(MethodHookParam param) {
                    if (sInitDone) return;
                    sWeChatContext = (Context) param.thisObject;
                    initCoreServices();
                    installRealTimeHooks();
                    startSubSystems();
                    sInitDone = true;
                    XposedBridge.log(TAG + ": [4/4] 全部就绪 ✓");
                }
            }
        );
    }

    // ================================================================
    // 阶段1: 获取微信核心服务实例
    // ================================================================
    private static void initCoreServices() {
        try {
            // g4 标签存储
            Class<?> holder = locateHolder();
            sLabelStorage = XposedHelpers.callStaticMethod(holder, "hj");
            XposedBridge.log(TAG + ": [2/4] LabelStorage(g4) ✓ — " + holder.getName());

            // j4 联系人存储
            Class<?> j1 = XposedHelpers.findClass("hm0.j1", sClassLoader);
            Class<?> sc4 = XposedHelpers.findClass("sh3.c4", sClassLoader);
            sContactStorage = XposedHelpers.callMethod(
                XposedHelpers.callStaticMethod(j1, "s", sc4), "ij"
            );
            XposedBridge.log(TAG + ": [2/4] ContactStorage(j4) ✓");

            // 配置初始化
            ConfigManager.initInWeChat(sWeChatContext);
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": [FAIL] " + e.getMessage());
        }
    }

    private static Class<?> locateHolder() {
        String[] cand = {"x93.r","x93.s","x93.q","x93.t","y93.r","w93.r"};
        for (String c : cand) {
            try {
                Class<?> cls = XposedHelpers.findClass(c, sClassLoader);
                Object r = XposedHelpers.callStaticMethod(cls, "hj");
                if (r != null && "com.tencent.mm.storage.g4".equals(r.getClass().getName()))
                    return cls;
            } catch (Throwable ignored) {}
        }
        try { return XposedHelpers.findClass("x93.r", sClassLoader); }
        catch (Throwable e) { return null; }
    }

    // ================================================================
    // 阶段2: 安装实时 Hook
    // ================================================================
    private static void installRealTimeHooks() {
        if (sHooksInstalled) return;
        sHooksInstalled = true;
        XposedBridge.log(TAG + ": [3/4] 安装实时 Hook...");

        // Hook 1: 标签创建 g4.insert(c4)
        hookLabelInsert();

        // Hook 2: 标签删除 g4.d(String)
        hookLabelDelete();

        // Hook 3: 标签更新 g4.update(c4, String[])
        hookLabelUpdate();

        // Hook 4: 联系人更新 j4.p0(String, y3)
        hookContactUpdate();

        // Hook 5: 网络同步完成 aa3.d.onGYNetEnd
        LabelSyncHook.install(sClassLoader);

        XposedBridge.log(TAG + ": [3/4] 5个实时Hook已安装");
    }

    // ---- Hook 1: 标签插入 ----
    private static void hookLabelInsert() {
        try {
            Class<?> g4 = XposedHelpers.findClass("com.tencent.mm.storage.g4", sClassLoader);
            XposedHelpers.findAndHookMethod(g4, "insert",
                "s85.f0",  // 参数类型 → c4 的父类
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam p) {
                        try {
                            Object label = p.args[0];
                            int id = XposedHelpers.getIntField(label, "field_labelID");
                            String name = (String) XposedHelpers.getObjectField(label, "field_labelName");
                            boolean ok = (boolean) p.getResult();
                            if (ok) {
                                XposedBridge.log(TAG + ": [Hook] 标签创建: [" + id + "] " + name);
                                EventBus.post(EventBus.Event.LABEL_CREATED,
                                    new int[]{id, name.hashCode()});
                            }
                        } catch (Throwable ignored) {}
                    }
                });
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": hookLabelInsert 失败: " + e.getMessage());
        }
    }

    private static void hookLabelDelete() {
        try {
            Class<?> g4 = XposedHelpers.findClass("com.tencent.mm.storage.g4", sClassLoader);
            XposedHelpers.findAndHookMethod(g4, "d", String.class,
                new XC_MethodHook() {
                    @Override
                    protected void beforeHookedMethod(MethodHookParam p) {
                        String labelId = (String) p.args[0];
                        XposedBridge.log(TAG + ": [Hook] 标签删除: ID=" + labelId);

                    protected void afterHookedMethod(MethodHookParam p) {
                        if ((boolean) p.getResult()) {

                });

            Class<?> g4 = XposedHelpers.findClass("com.tencent.mm.storage.g4", sClassLoader);
            XposedHelpers.findAndHookMethod(g4, "update",
                "s85.f0", "[Ljava.lang.String;",
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam p) {
                        try {
                            Object label = p.args[0];
                            int id = XposedHelpers.getIntField(label, "field_labelID");
                            String name = (String) XposedHelpers.getObjectField(label, "field_labelName");
                            if ((boolean) p.getResult()) {
                                EventBus.post(EventBus.Event.LABEL_RENAMED,
                                    new Object[]{id, name});
                                XposedBridge.log(TAG + ": [Hook] 标签更新: [" + id + "] " + name);
                            }
                        } catch (Throwable ignored) {}
                    }
                });
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": hookLabelUpdate 失败: " + e.getMessage());
        }
    }

    // ---- Hook 4: 联系人更新 ----
    private static void hookContactUpdate() {
        try {
            Class<?> j4 = XposedHelpers.findClass("com.tencent.mm.storage.j4", sClassLoader);
            XposedHelpers.findAndHookMethod(j4, "p0",
                String.class, "com.tencent.mm.storage.y3",
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam p) {
                        try {
                            String username = (String) p.args[0];
                            Object contact = p.args[1];
                            String labelIds = (String) XposedHelpers.callMethod(contact, "A0");
                            XposedBridge.log(TAG + ": [Hook] 联系人标签变更: " + username
                                + " → [" + labelIds + "]");
                            EventBus.post(EventBus.Event.CONTACT_LABEL_CHANGED, username);
                        } catch (Throwable ignored) {}
                    }
                });
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": hookContactUpdate 失败: " + e.getMessage());
        }
    }

    // ================================================================
    // 阶段3: 启动子系统
    // ================================================================
    private static void startSubSystems() {
        // 自动备份（如启用）
        if (ConfigManager.isBackupEnabled()) {
            LabelBackup.scheduleAutoBackup(sWeChatContext);
        }
        // 自动分组引擎
        if (ConfigManager.isAutoGroupEnabled()) {
            AutoGroupEngine.start();
        }
        // 监听事件总线
        AutoGroupEngine.subscribeToEvents();
    }

    // ================================================================
    // 公开访问器
    // ================================================================
    public static Object getLabelStorage() {
        if (!sInitDone && sClassLoader != null) initCoreServices();
        return sLabelStorage;
    }

    public static Object getContactStorage() {
        if (!sInitDone && sClassLoader != null) initCoreServices();
        return sContactStorage;
    }

    public static ClassLoader getClassLoader() { return sClassLoader; }
    public static Context getWeChatContext() { return sWeChatContext; }
    public static boolean isReady() { return sInitDone && sLabelStorage != null && sContactStorage != null; }
}

 *   1. 标签同步完成后

    }

    public static void subscribeToEvents() {
        EventBus.subscribe(EventBus.Event.LABELS_SYNCED, (event, data) -> {
            XposedBridge.log(TAG + ": 收到 LABELS_SYNCED，开始执行自动分组...");
            runOnce();
        });
        EventBus.subscribe(EventBus.Event.CONTACT_LABEL_CHANGED, (event, data) -> {
            // 联系人标签变更后重新评估该联系人
            String username = (String) data;
            evaluateContact(username);
        });
    }

    // ==================== 规则管理 ====================

    public static synchronized void addRule(AutoRule rule) {
        sRules.add(rule);
        saveRules();
        XposedBridge.log(TAG + ": 规则已添加: " + rule);
    }

    public static synchronized boolean removeRule(int ruleId) {
        boolean removed = sRules.removeIf(r -> r.ruleId == ruleId);
        if (removed) saveRules();
        return removed;
    }

    public static List<AutoRule> getRules() {
        return new ArrayList<>(sRules);
    }

    public static void enableRule(int ruleId, boolean enabled) {
        for (AutoRule r : sRules) {
            if (r.ruleId == ruleId) { r.enabled = enabled; break; }
        }
        saveRules();
    }

    // ==================== 执行引擎 ====================

    /**
     * 全量扫描执行一次
     */
    public static synchronized void runOnce() {
        if (sRunning) return;
        sRunning = true;
        try {
            XposedBridge.log(TAG + ": 开始全量扫描...");
            int totalMatched = 0;

            for (AutoRule rule : sRules) {
                if (!rule.enabled) continue;
                int matched = executeRule(rule);
                totalMatched += matched;
                rule.matchCount += matched;
                rule.lastRunTime = System.currentTimeMillis();
            }
            saveRules();
            XposedBridge.log(TAG + ": 扫描完成，共命中 " + totalMatched + " 次");
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": runOnce error: " + e.getMessage());
        } finally {
            sRunning = false;
        }
    }

    private static int executeRule(AutoRule rule) {
        int matched = 0;
        try {
            Object st = MainHook.getContactStorage();
            if (st == null) return 0;

            // 获取所有联系人（从 rcontact 表）
            // 使用 j4 的 K() 或 D() 获取全部联系人 cursor
            // 备选: 通过 g4 缓存获取所有有标签的联系人
            Set<String> allUsers = getAllContactUsernames(st);

            for (String username : allUsers) {
                if (matchRule(username, rule)) {
                    ContactHelper.addLabel(username, rule.targetLabelId);
                    matched++;
                }
            }
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": executeRule error: " + e.getMessage());
        }
        if (matched > 0) {
            EventBus.post(EventBus.Event.AUTO_RULE_EXECUTED,
                new Object[]{rule, matched});
        }
        return matched;
    }

            if (!rule.enabled) continue;

    private static boolean matchRule(String username, AutoRule rule) {
        try {
            Object st = MainHook.getContactStorage();
            if (st == null) return false;
            Object contact = XposedHelpers.callMethod(st, "m", username);
            if (contact == null) return false;

            switch (rule.matchType) {
                case REMARK_CONTAINS: {
                    // dm.f2 有 conRemark 相关字段
                    // 尝试通过 y3 的 getter 获取备注
                    String remark = getContactField(contact, "field_conRemark");
                    return remark != null && remark.contains(rule.matchValue);
                }
                case NICKNAME_CONTAINS: {
                    String nickname = getContactField(contact, "field_nickname");
                    return nickname != null && nickname.contains(rule.matchValue);
                }
                case USERNAME_MATCHES: {
                    return username.matches(rule.matchValue);
                }
                case LABEL_NAME_MATCHES: {
                    // 检查联系人已有标签名是否匹配
                    int[] ids = ContactHelper.getLabelIds(username);

                        if (label != null) {
                            String name = (String) label.get("labelName");
                            if (name != null && name.contains(rule.matchValue)) return true;
                        }
                    }
                    return false;
                }
                default:
                    return false;
            }
        } catch (Throwable e) {
            return false;
        }
    }

              .append(r.matchValue).append("|")
              .append(r.targetLabelId).append("|")
              .append(r.enabled);
        }
        ConfigManager.getRaw().edit().putString("auto_rules", sb.toString()).apply();
    }

     * @param filePath JSON文件路径
     * @param overwrite 是否覆盖已存在的同名标签
     * @return 导入数量
     */
    public static int importFromJson(String filePath, boolean overwrite) {
        int count = 0;
        try {
            File file = new File(filePath);
            if (!file.exists()) {
                XposedBridge.log(TAG + ": 文件不存在: " + filePath);
                return 0;
            }

            String content = readFile(file);
            // 简单解析（生产环境建议用Gson/org.json）
            List<Map<String, Object>> parsed = parseSimpleJson(content);
            for (Map<String, Object> item : parsed) {
                String name = (String) item.get("labelName");
                if (name == null || name.isEmpty()) continue;

                // 检查重复
                LabelInfo exist = LabelManager.getByName(name);
                if (exist != null) {
                    if (overwrite) {
                        LabelManager.rename(String.valueOf(exist.labelId), name);
                    } else {
                        continue;
                    }
                }

                LabelInfo created = LabelManager.create(name);
                if (created != null) {
                    count++;
                    // 恢复联系人关联
                    @SuppressWarnings("unchecked")
                    List<String> contacts = (List<String>) item.get("contacts");
                    if (contacts != null && !contacts.isEmpty()) {
                        for (String u : contacts) {
                            ContactHelper.addLabel(u, created.labelId);
                        }
                    }
                }
            }
            XposedBridge.log(TAG + ": 导入完成 → " + count + " 个标签从 " + filePath);
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": 导入失败: " + e.getMessage());
        }
        return count;
    }

            hours, hours, TimeUnit.HOURS);

    }

        long last = ConfigManager.getLastBackupTime();
        long interval = ConfigManager.getBackupIntervalHours() * 3600000L;
        if (System.currentTimeMillis() - last >= interval) {
            exportToJson(ctx);
        }
    }

    private static String escape(String s) {
        if (s == null) return "";
        return s.replace("\\", "\\\\")
                .replace("\"", "\\\"")
                .replace("\n", "\\n")
                .replace("\r", "\\r")
                .replace("\t", "\\t");
    }

    private static String readFile(File file) throws IOException {
        StringBuilder sb = new StringBuilder();
        try (BufferedReader br = new BufferedReader(new FileReader(file))) {
            String line;
            while ((line = br.readLine()) != null) sb.append(line).append("\n");
        }
        return sb.toString();
    }

    private static List<Map<String, Object>> parseSimpleJson(String json) {
        // 生产环境应使用 Gson 或 org.json
        // 此处提供占位，实际请引入 Gson
        List<Map<String, Object>> list = new ArrayList<>();
        try {
            // 尝试用 org.json
            org.json.JSONArray arr = new org.json.JSONArray(json);
            for (int i = 0; i < arr.length(); i++) {
                org.json.JSONObject obj = arr.getJSONObject(i);
                Map<String, Object> map = new HashMap<>();
                map.put("labelId", obj.optInt("labelId"));
                map.put("labelName", obj.optString("labelName"));
                map.put("labelPYFull", obj.optString("labelPYFull"));
                map.put("labelPYShort", obj.optString("labelPYShort"));
                map.put("createTime", obj.optLong("createTime"));
                map.put("isTemporary", obj.optBoolean("isTemporary"));

                List<String> contacts = new ArrayList<>();
                org.json.JSONArray ca = obj.optJSONArray("contacts");
                if (ca != null) {
                    for (int j = 0; j < ca.length(); j++) {
                        contacts.add(ca.optString(j));
                    }
                }
                map.put("contacts", contacts);
                list.add(map);
            }
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": JSON解析失败: " + e.getMessage());
        }
        return list;
    }

/**
 * 广播命令接收器
 * 
 * 外部应用可通过发送广播来控制分组功能:
 * 
 * adb shell am broadcast -a com.wxgroup.action.EXECUTE \
 *   --es command "create_label" --es labelName "测试分组"
 * 
 * 支持的命令:
 *   create_label     — 创建标签
 *   delete_label     — 删除标签
 *   rename_label     — 重命名标签
 *   add_contact      — 添加联系人到标签
 *   remove_contact   — 从标签移除联系人
 *   backup           — 导出备份
 *   restore          — 从备份恢复
 *   run_auto_group   — 执行自动分组
 *   merge_labels     — 合并标签
 *   stats            — 获取统计（通过结果广播返回）
 */
public class CommandReceiver extends BroadcastReceiver {

    private static final String TAG = "WxGroup.Cmd";

    @Override
    public void onReceive(Context context, Intent intent) {
        if (!MainHook.isReady()) {
            XposedBridge.log(TAG + ": 模块未就绪，忽略命令");
            return;
        }

        String action = intent.getAction();
        if ("com.wxgroup.action.EXECUTE".equals(action)) {
            execute(context, intent);
        } else if ("com.wxgroup.action.QUERY".equals(action)) {
            query(context, intent);
        } else if ("com.wxgroup.action.BACKUP".equals(action)) {
            LabelBackup.exportToJson(context);
        } else if ("com.wxgroup.action.RESTORE".equals(action)) {
            String path = intent.getStringExtra("path");
            boolean overwrite = intent.getBooleanExtra("overwrite", false);
            if (path != null) LabelBackup.importFromJson(path, overwrite);
        }
    }

    private void execute(Context ctx, Intent intent) {
        String cmd = intent.getStringExtra("command");
        if (cmd == null) return;

        try {
            switch (cmd) {
                case "create_label": {
                    String name = intent.getStringExtra("labelName");
                    if (name != null) LabelManager.create(name);
                    break;
                }
                case "delete_label": {
                    String id = intent.getStringExtra("labelId");
                    if (id != null) LabelManager.delete(id);
                    break;
                }
                case "rename_label": {
                    String id = intent.getStringExtra("labelId");
                    String name = intent.getStringExtra("labelName");
                    if (id != null && name != null) LabelManager.rename(id, name);
                    break;
                }
                case "add_contact": {
                    String user = intent.getStringExtra("username");
                    int lid = intent.getIntExtra("labelId", -1);
                    if (user != null && lid >= 0) ContactHelper.addLabel(user, lid);
                    break;
                }
                case "remove_contact": {
                    String user = intent.getStringExtra("username");
                    int lid = intent.getIntExtra("labelId", -1);
                    if (user != null && lid >= 0) ContactHelper.removeLabel(user, lid);
                    break;
                }
                case "run_auto_group": {
                    AutoGroupEngine.runOnce();
                    break;
                }
                case "merge_labels": {
                    int src = intent.getIntExtra("sourceId", -1);
                    int dst = intent.getIntExtra("targetId", -1);
                    if (src >= 0 && dst >= 0) BatchOperator.mergeLabels(src, dst);
                    break;
                }
                case "backup": {
                    LabelBackup.exportToJson(ctx);
                    break;
                }
                default:
                    XposedBridge.log(TAG + ": 未知命令: " + cmd);
            }
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": 命令执行异常: " + e.getMessage());
        }
    }

    private void query(Context ctx, Intent intent) {
        String cmd = intent.getStringExtra("command");
        if ("stats".equals(cmd)) {
            var stats = BatchOperator.getStatistics();
            // 通过广播返回结果
            Intent result = new Intent("com.wxgroup.action.RESULT");
            Bundle b = new Bundle();
            for (var e : stats.entrySet()) {
                b.putString(e.getKey(), String.valueOf(e.getValue()));
            }
            result.putExtras(b);
            ctx.sendBroadcast(result);
        }
    }
