    /** 在微信进程、内核初始化完成后调用(建议挂 LauncherUI.onResume 之后 / 你已有 onReady 回调里) */
    public static synchronized void init(ClassLoader cl) {
        if (sReady) return;
        sCl = cl;
        try {
            // 1) ChatroomMembersLogic: 含 "@chatroom" 与 "@im.chatroom" 且含 getMembersByChatRoomName 的类
            C_u1 = findClassWithStrings("@chatroom", "@im.chatroom");

            M_u1_B = getStatic(C_u1, "B", String.class);                 // 是否群
            M_u1_m = getStatic(C_u1, "m", String.class);                 // 取成员
            // 2) 我的群: 静态 List c(), 含 "FTSApiLogic"
            C_n = findSimpleStatic("c");
            M_n_c = getStatic(C_n, "c");

            // 3) 服务定位器 + RoomServiceFactory
            C_n0 = findClassWithMethod("c", Class.class);                // Object c(Class)->service
            M_n0_c = getStatic(C_n0, "c", Class.class);

            C_vf0e = C_n0 == null ? null : findInterfaceWithMethod("bj", String.class);
            C_pe5f = findInterfaceWithMethod("j", String.class, List.class, String.class, Object.class);
            C_opA = findClassExact("com.tencent.mm.roomsdk.model.factory.a");
            if (C_opA == null)
                C_opA = findSuperclassOfImplementer(C_pe5f);             // 兜底: 找返回 factory.a 的抽象父

            // 4) 具体工厂实现(uf0.e)上的 bj(String)->pe5.f ; 及 pe5.f.j(...)
            M_bj = findMethodOnFactory();
            M_j = getMethod(C_pe5f, "j", String.class, List.class, String.class, Object.class);
            M_opB = getMethod(C_opA, "b");                               // 无参 b() 发送
            M_opC = getMethod(C_opA, "c", Class.forName("android.content.Context", false, cl),
                    String.class, String.class, boolean.class, boolean.class,
                    Class.forName("android.content.DialogInterface$OnCancelListener", false, cl));
            F_op_d = getField(C_opA, "d");                               // 回调字段

            C_cb = (F_op_d != null) ? F_op_d.getType() : null;

            // 5) 邀请链接消息 b41.s1: static void c(String,List,String,boolean,String)
            C_s1 = findClassWithMethod("c", String.class, List.class, String.class, boolean.class, String.class);
            M_s1_c = getStatic(C_s1, "c", String.class, List.class, String.class, boolean.class, String.class);

            sReady = M_u1_m != null && M_u1_B != null && M_n_c != null
                    && M_n0_c != null && M_bj != null && M_j != null && M_opB != null;
            android.util.Log.i(TAG, "init ready=" + sReady + " u1=" + (C_u1 != null)
                    + " s1=" + (C_s1 != null) + " opA=" + (C_opA != null));
        } catch (Throwable t) {
            android.util.Log.e(TAG, "init fail: " + t);
            sReady = false;
        }
    }

    public static boolean isReady() { return sReady; }

    /** 返回「target 不在的、我加入的群」username 列表 */
    @SuppressWarnings("unchecked")
    public static List<String> pickCandidateGroups(String targetWxid) {
        List<String> out = new ArrayList<>();
        if (!sReady || TextUtils.isEmpty(targetWxid)) return out;
        try {
            List<String> myRooms = (List<String>) M_n_c.invoke(null);
            if (myRooms == null) return out;
            for (String room : myRooms) {
                if (TextUtils.isEmpty(room) || !isRoom(room)) continue;
                List<String> members = membersOf(room);
                if (members == null) continue;             // 取不到成员信息: 保守跳过(可能未同步)
                boolean in = false;
                for (String m : members) if (targetWxid.equals(m)) { in = true; break; }
                if (!in) out.add(room);
            }
        } catch (Throwable t) {
            android.util.Log.e(TAG, "pickCandidateGroups: " + t);
        }
        return out;
    }

    public static boolean isRoom(String u) {

    /** 并发邀请多个群(建议串行+外部节流; 内部对 -2012 自动降级发邀请链接) */
    public static void inviteMulti(List<String> rooms, String targetWxid, String reason, Callback cb) {
        if (!sReady) { if (cb != null) cb.onResult("<init>", false, -1, "not-ready"); return; }
        String rsn = TextUtils.isEmpty(reason) ? "邀请你加入群聊" : reason;
        for (String room : rooms) inviteOne(room, targetWxid, rsn, cb);
    }

    /** 单个群: 先静默强拉; 失败 -2012 自动降级发邀请链接 */
    public static void inviteOne(String room, String targetWxid, String reason, Callback cb) {
        try {
            Object factory = M_n0_c.invoke(null, C_vf0e);          // RoomServiceFactory
            Object roomOp  = M_bj.invoke(factory, room);           // pe5.f 绑定群
            Object op      = M_j.invoke(roomOp, room,
                    Collections.singletonList(targetWxid), reason, null);   // NetSceneAddChatRoomMember

            final Object fOp = op;
            Object cbProxy = newCallback((errType, errCode, errMsg, data) -> {
                List<?> ok = fieldAsList(data, "a");               // 状态0 成功列表
                boolean success = (errType == 0 && errCode == 0 && ok != null && !ok.isEmpty());
                if (success) {
                    if (cb != null) cb.onResult(room, true, 0, "ok");
                } else if (errCode == -2012 || (errMsg != null && errMsg.toLowerCase().contains("need invite"))) {
                    // 需邀请 => 发邀请链接(本账号在群里发 type=10000 系统消息, 对方点确认才进)
                    sendInviteLink(room, Collections.singletonList(targetWxid), reason);
                    if (cb != null) cb.onResult(room, false, errCode, "need-invite -> 已降级发邀请链接");
                } else {
                    if (cb != null) cb.onResult(room, false, errCode, errMsg);
                }
            });
            F_op_d.set(fOp, cbProxy);                              // 设回调字段 d
            // 无 UI 直接发送; 想上原生加载框改用 M_opC.invoke(fOp, ctx, t1, t2, true, true, cancelListener)
            M_opB.invoke(fOp);
        } catch (Throwable t) {
            android.util.Log.e(TAG, "inviteOne " + room + ": " + t);
            if (cb != null) cb.onResult(room, false, -999, String.valueOf(t));
        }
    }

    /** 发邀请链接/系统邀请消息(微信正规降级路径 vn.m.e() 等价物) */
    public static void sendInviteLink(String room, List<String> targets, String reasonTpl) {
        try {
            String tpl = TextUtils.isEmpty(reasonTpl) ? "%s 邀请你加入群聊" : reasonTpl;
            String url = "weixin://findfriend/verifycontact/" + room + "/";   // 注意结尾斜杠
            M_s1_c.invoke(null, room, targets, tpl, Boolean.TRUE, url);
        } catch (Throwable t) {
            android.util.Log.e(TAG, "sendInviteLink: " + t);
        }
    }

    private interface RawCallback { void a(int errType, int errCode, String errMsg, Object data); }

    private static Object newCallback(RawCallback r) {
        try {
            return Proxy.newProxyInstance(sCl, new Class[]{ C_cb }, (proxy, m, args) -> {
                if ("a".equals(m.getName()) && args != null) {
                    int et = toInt(args[0]);
                    int ec = toInt(args[1]);
                    String em = args.length > 2 ? String.valueOf(args[2]) : "";
                    Object data = args.length > 3 ? args[3] : null;
                    r.a(et, ec, em, data);
                } else if ("hashCode".equals(m.getName())) return 0;
                else if ("equals".equals(m.getName())) return proxy == args[0];
                else if ("toString".equals(m.getName())) return "WxGroupInviter$cb";
                return null;
            });
        } catch (Throwable t) { return null; }
    }

    private static List<?> fieldAsList(Object data, String fieldName) {
        if (data == null) return null;
        try {
            Field f = data.getClass().getField(fieldName);
            Object v = f.get(data);
            return (v instanceof List) ? (List<?>) v : null;
        } catch (Throwable ignore) { return null; }
    }

    private static int toInt(Object o) { return o == null ? 0 : ((Number) o).intValue(); }

三个必须知道的坑（都是上面日志/逆向实证）
1.  
是否群/取成员有缓存时序： t73.n.c() （我的群）走 FTS/联系人库，内核/DB 没就绪时返回空——所以第 2 步一定要延迟/挂 onReady。 b41.u1.m(room)  的成员列表若没同步过某群可能返回空，代码里对"空成员"保守跳过；要更准可先用群信息页打开一次该群强制同步。
2.  
 .j  与  .b(...)  别搞反： pe5.f.j(room,list,reason,obj)  = 加人； pe5.f.b(room,list,int)  = 踢人（底层  delchatroommember ）。代码里只调了  j 。
3.  
风控无代码解：日志  由于账号安全原因…无法加入当前群聊  是服务端回执。邀请链接能绕过「需 add 权限」，绕不过「账号安全风控」。风控期降级发链接也可能被吞，先停手养号；代码层最多做到识别该文案自动暂停队列，别再撞墙加重封禁。
