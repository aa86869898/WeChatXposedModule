// ================= FwdFix final（v4） =================
// 依赖你 DexKit 已解析：MsgRetransmitUI / MvvmContactListUI
// 目标：选择器确定后 data==null → 从 MvvmContactListUI 的选中态取目标 → 注入 MsgRetransmitUI.h
//       不 setResult(null)、不调 g7，让微信原生继续跑（保留成功提示）

public static volatile android.app.Activity sLastMvvm;   // 最近一次 MvvmContactListUI

public void installFwdFix(ClassLoader cl) {
    hookRetransmit(cl);   // ① 入口
    hookPickerFinish(cl); // ② 缓存 + 状态 dump
}

/* ---------- ① MsgRetransmitUI.onActivityResult ---------- */
private void hookRetransmit(ClassLoader cl) {
    final XC_MethodHook cb = new XC_MethodHook() {
        @Override protected void beforeHookedMethod(MethodHookParam p) throws Throwable {
            if (p.thisObject == null) return;
            String cls = p.thisObject.getClass().getName();
            if (!cls.equals("com.tencent.mm.ui.transmit.MsgRetransmitUI")) return;   // 只认转发页
            int rc = (Integer) p.args[1];
            android.content.Intent data = (android.content.Intent) p.args[2];
            XposedBridge.log("[FwdFix][B] fired rc=" + rc + " data=" + data + " this=" + cls);

            if (data != null) return;                 // 原生自带结果, 不插手
            if (rc != android.app.Activity.RESULT_OK) return;

            Object self = p.thisObject;
            java.util.List<String> picked = null;
            try { picked = pullStateP(); } catch (Throwable t) { XposedBridge.log("[FwdFix][B] state err " + t); }
            XposedBridge.log("[FwdFix][B] state.p=" + picked);

            if (picked == null || picked.isEmpty()) {
                XposedBridge.log("[FwdFix][B] 无目标, 不注入");
                dumpState(sLastMvvm);                 // 顺手 dump, 定位字段名
                return;
            }
            try {
                java.lang.reflect.Field h = self.getClass().getDeclaredField("h");   // List<String>
                h.setAccessible(true);
                h.set(self, picked);
                XposedBridge.log("[FwdFix][B] 注入 h=" + h.get(self)
                        + " type=" + getF(self,"e")
                        + " contentLen=" + String.valueOf(getF(self,"i")).length());
            } catch (Throwable t) { XposedBridge.log("[FwdFix][B] 注入失败 " + t); return; }
            XposedBridge.log("[FwdFix][B] 注入完成, 让原生继续跑(目标=" + picked + ")");
        }
    };

    // 基类层先装(CFLPMenu 已证明 android.app.Activity 层一定触发, 且能覆盖 MsgRetransmitUI)
    try { XposedHelpers.findAndHookMethod("android.app.Activity", cl,
            "onActivityResult", int.class, int.class, android.content.Intent.class, cb); }
    catch (Throwable t) { XposedBridge.log("[FwdFix] hook act err " + t); }

    // 子类再装一次(若 MsgRetransmitUI 自己重写了, 也能拿到)
    try { XposedHelpers.findAndHookMethod("com.tencent.mm.ui.transmit.MsgRetransmitUI", cl,
            "onActivityResult", int.class, int.class, android.content.Intent.class, cb); }
    catch (Throwable t) { XposedBridge.log("[FwdFix] hook rtx err " + t); }
}

/* ---------- ② MvvmContactListUI.finish() ---------- */
private void hookPickerFinish(ClassLoader cl) {
    try {
        XposedHelpers.findAndHookMethod("com.tencent.mm.ui.mvvm.MvvmContactListUI", cl, "finish",
            new XC_MethodHook() {
                @Override protected void beforeHookedMethod(MethodHookParam p) throws Throwable {
                    sLastMvvm = (android.app.Activity) p.thisObject;
                    XposedBridge.log("[FwdFix][A2] MvvmContactListUI.finish → state.dump");
                    dumpState(sLastMvvm);
                }
            });
    } catch (Throwable t) { XposedBridge.log("[FwdFix] hook finish err " + t); }
}

/* ---------- 工具：MVI StateCenter → state.p ---------- */
@SuppressWarnings("unchecked")
private java.util.List<String> pullStateP() throws Throwable {
    return pullStatePFrom(sLastMvvm);
}

@SuppressWarnings("unchecked")
private java.util.List<String> pullStatePFrom(android.app.Activity act) throws Throwable {
    if (act == null) return null;
    Object center = act.getClass().getMethod("getStateCenter").invoke(act);
    if (center == null) return null;
    Object st = center.getClass().getMethod("getState").invoke(center);
    if (st == null) return null;
    java.lang.reflect.Field f = st.getClass().getDeclaredField("p");   // LinkedList<String> 选中列表
    f.setAccessible(true);
    Object v = f.get(st);
    if (v instanceof java.util.List) {
        java.util.List<String> out = new java.util.ArrayList<>();
        for (Object o : (java.util.List<?>) v) out.add(String.valueOf(o));
        return out;
    }
    return null;
}

/* ---------- 工具：state 全字段 dump（一次定位真字段名） ---------- */
private void dumpState(android.app.Activity act) {
    try {
        Object st = null;
        if (act != null) {
            Object center = act.getClass().getMethod("getStateCenter").invoke(act);
            if (center != null) st = center.getClass().getMethod("getState").invoke(center);
        }
        if (st == null) { XposedBridge.log("[FwdFix][STATE] null"); return; }
        for (java.lang.reflect.Field f : st.getClass().getDeclaredFields()) {
            f.setAccessible(true);
            Object v;
            try { v = f.get(st); } catch (Throwable t) { v = "<err>"; }
            XposedBridge.log("[FwdFix][STATE] " + st.getClass().getSimpleName() + "." + f.getName()
                    + " : " + (v == null ? "null" : v.getClass().getSimpleName()) + " = " + v);
        }
    } catch (Throwable t) { XposedBridge.log("[FwdFix][STATE] dump err " + t); }
}

private Object getF(Object o, String name) throws Throwable {
    java.lang.reflect.Field f = o.getClass().getDeclaredField(name);
    f.setAccessible(true);
    return f.get(o);
}

•  
打出来 → 机器通了，继续看  state.p=  /  [FwdFix][STATE] 。
•  
打不出来 → 说明你这套  findAndHookMethod  在框架里也起不来（和之前  XposedHelpers NoSuchMethodError  同源）。那就把 CFLPMenu 里真正干 install 的那 1~2 行原样贴我，我按它的写法逐字改——CFLPMenu 能触发，说明它用的那个 primitive 在你环境是通的。
2.  
 state.p  不是选中列表就改字段：看日志里的  [FwdFix][STATE] xxx = [...] ，哪个是  LinkedList / List<String>  且内容是 wxid，就把  pullStatePFrom()  里的  "p"  换成那个名字（ q/o/e …都有可能）。
3.
