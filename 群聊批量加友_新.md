 * 1.1 群聊批量添加好友调用链
 *   ChattingUI.onCreate → intent["Chat_User"]="@chatroom" → 注入悬浮按钮

 *        管理员     room.E0(u)   (RoomData Member Flag & 2048)

 *   a4.h(Context, String target, e9 msg, boolean, boolean, String)  ← Hook 逐条转发

/** Xposed 入口 */
public final class WxXposedModule implements IXposedHookLoadPackage {
    private static final Set<String> injected = new HashSet<>();

    @Override
    public void handleLoadPackage(LoadPackageParam lpparam) {
        if (!"com.tencent.mm".equals(lpparam.packageName)) return;
        final ClassLoader cl = lpparam.classLoader;
        try {
            // 群聊批量添加好友入口
            XposedHelpers.findAndHookMethod(WxConstants.CHATTING_UI, cl, "onCreate", Bundle.class,
                    new XC_MethodHook() {
                        @Override protected void afterHookedMethod(MethodHookParam param) {
                            try {
                                Activity act = (Activity) param.thisObject;
                                String talker = act.getIntent().getStringExtra("Chat_User"); // 实测key
                                if (talker != null && talker.endsWith("@chatroom")) {
                                    act.runOnUiThread(() -> injectButton(act, talker, cl));
                                }
                            } catch (Throwable t) { t.printStackTrace(); }
                        }
                    });
            // 语音转发入口
            VoiceForward.hook(cl);
        } catch (Throwable t) { t.printStackTrace(); }
    }

    private static void injectButton(Activity act, String room, ClassLoader cl) {
        if (!injected.add(room)) return;
        try {
            Button btn = new Button(act);
            btn.setText("批量加友"); btn.setTextSize(13);
            btn.setTextColor(Color.WHITE); btn.setBackgroundColor(Color.rgb(238, 80, 40)); btn.setAlpha(0.92f);
            btn.setOnClickListener(v -> new GroupBatchAddFriend().showSelector(act, room, cl, true));
            FrameLayout.LayoutParams lp = new FrameLayout.LayoutParams(
                    FrameLayout.LayoutParams.WRAP_CONTENT, FrameLayout.LayoutParams.WRAP_CONTENT);
            lp.gravity = Gravity.TOP | Gravity.END; lp.setMargins(0, 140, 20, 0);
            ((ViewGroup) act.getWindow().getDecorView()).addView(btn, lp);
        } catch (Throwable t) { t.printStackTrace(); }
    }
}

/** 运行时取微信服务 */
final class WxServices {
    private final ClassLoader cl;
    WxServices(ClassLoader cl) { this.cl = cl; }

    Object svc(String ifaceCls) {
        return XposedHelpers.callStaticMethod(
                XposedHelpers.findClass(WxConstants.J1, cl), "v",
                XposedHelpers.findClass(ifaceCls, cl));
    }
    Object chatStore() { return XposedHelpers.callMethod(svc(WxConstants.Q02_F), "a"); }   // a3
    Object contactStore() { return XposedHelpers.callMethod(svc(WxConstants.TN3_C4), "cj"); } // j4
    Object msgStore() { return XposedHelpers.callMethod(svc(WxConstants.TN3_C4), "lj"); }   // f9
    Object netSceneQueue() {                                                                 // r1
        return XposedHelpers.getObjectField(
                XposedHelpers.callStaticMethod(XposedHelpers.findClass(WxConstants.J1, cl), "q"), "b");
    }
    String myWxid() {
        return (String) XposedHelpers.callStaticMethod(XposedHelpers.findClass(WxConstants.B41_Y1, cl), "u");
    }
}

/** 群聊批量添加好友 */
final class GroupBatchAddFriend {

    static final class Member { String wxid, display; boolean isOwner, isAdmin, isFriend; }

    // ---------- 枚举 + 过滤 ----------
    static List<Member> members(ClassLoader cl, String room) {
        WxServices s = new WxServices(cl);
        Object roomEnt = XposedHelpers.callMethod(s.chatStore(), "t1", room);   // z2
        List<Member> out = new ArrayList<>();
        if (roomEnt == null) return out;
        @SuppressWarnings("unchecked")
        List<String> wxids = (List<String>) XposedHelpers.callMethod(roomEnt, "z0");
        String me = s.myWxid();
        for (String u : wxids) {
            if (u == null || u.length() == 0 || u.equals(me)) continue;
            Member m = new Member();
            m.wxid = u;
            m.isOwner = (Boolean) XposedHelpers.callMethod(roomEnt, "L0", u);   // 群主
            m.isAdmin = (Boolean) XposedHelpers.callMethod(roomEnt, "E0", u);   // 管理员
            m.isFriend = isFriend(cl, u);
            m.display = displayName(cl, u);
            out.add(m);
        }
        return out;
    }

    static boolean isFriend(ClassLoader cl, String u) {                          // 好友: type&1
        try {
            Object c = XposedHelpers.callMethod(new WxServices(cl).contactStore(), "n", u, true);
            if (c == null) return false;
            int type = (Integer) XposedHelpers.callMethod(c, "getType");         // im.f2.getType()
            return (type & 1) != 0;
        } catch (Throwable t) { return false; }
    }

    static String displayName(ClassLoader cl, String u) {                        // g2=备注兜底 i1=wxid
        try {
            Object c = XposedHelpers.callMethod(new WxServices(cl).contactStore(), "n", u, true);
            if (c != null) {
                String s = (String) XposedHelpers.callMethod(c, "g2");
                if (s != null && s.length() > 0) return s;
                s = (String) XposedHelpers.callMethod(c, "i1");
                if (s != null && s.length() > 0) return s;
            }
        } catch (Throwable ignored) {}
        return u;
    }

    static boolean isSystemUser(String u) {                                       // 公众号/群/系统号/OpenIM
        if (u == null) return true;
        if (u.startsWith("gh_")) return true;
        if (u.endsWith("@chatroom") || u.endsWith("@im.chatroom") || u.endsWith("@lbsroom")) return true;
        if (u.endsWith("@openim") || u.endsWith("@bizfansmsg") || u.endsWith("@stranger") || u.endsWith("@bottle")) return true;
        return u.startsWith("filehelper") || u.startsWith("floatbottle") || u.startsWith("medianote");
    }

    // ---------- 勾选弹窗 ----------
    void showSelector(Activity act, String room, ClassLoader cl, boolean skipFriend) {
        List<Member> cand = new ArrayList<>();
        for (Member m : members(cl, room)) {
            if (m.isOwner || m.isAdmin) continue;          // ★必须过滤群主/管理员
            if (isSystemUser(m.wxid)) continue;
            if (skipFriend && m.isFriend) continue;
            cand.add(m);
        }
        if (cand.isEmpty()) return;

        ScrollView scroll = new ScrollView(act);
        LinearLayout box = new LinearLayout(act);
        box.setOrientation(LinearLayout.VERTICAL);
        List<CheckBox> cbs = new ArrayList<>();
        for (Member m : cand) {
            CheckBox cb = new CheckBox(act);
            cb.setText(m.display + "  (" + m.wxid + ")");
            cb.setChecked(true); cb.setTag(m);
            box.addView(cb); cbs.add(cb);
        }
        scroll.addView(box);

        LinearLayout top = new LinearLayout(act);
        top.setOrientation(LinearLayout.HORIZONTAL);
        top.addView(b(act, "全选", v -> cbs.forEach(c -> c.setChecked(true))));
        top.addView(b(act, "反选", v -> cbs.forEach(c -> c.setChecked(!c.isChecked()))));
        top.addView(b(act, "清空", v -> cbs.forEach(c -> c.setChecked(false))));

        final android.widget.EditText etVerify = new android.widget.EditText(act);
        etVerify.setHint("验证内容"); etVerify.setText("你好，我是微信好友");
        final android.widget.EditText etDelay = new android.widget.EditText(act);
        etDelay.setHint("间隔毫秒(≥1500)"); etDelay.setText("1500");

        LinearLayout root = new LinearLayout(act);
        root.setOrientation(LinearLayout.VERTICAL);
        root.addView(top); root.addView(etVerify); root.addView(etDelay); root.addView(scroll);

        new android.app.AlertDialog.Builder(act)
                .setTitle("批量添加好友 " + room)
                .setView(root)
                .setPositiveButton("开始添加", (d, w) -> {
                    List<Member> sel = new ArrayList<>();
                    for (CheckBox c : cbs) if (c.isChecked()) sel.add((Member) c.getTag());
                    int delay = 1500;
                    try { delay = Integer.parseInt(etDelay.getText().toString()); } catch (Throwable ignored) {}
                    String verify = etVerify.getText().toString();
                    final int dl = Math.max(delay, 800);
                    new Thread(() -> batchSend(cl, room, sel, verify, dl)).start();
                })
                .setNegativeButton("取消", null)
                .show();
    }

    private static Button b(Activity a, String t, android.view.View.OnClickListener l) {
        Button b = new Button(a); b.setText(t); b.setTextSize(12); b.setOnClickListener(l); return b;
    }

    // ---------- 批量发送 ----------
    private void batchSend(ClassLoader cl, String room, List<Member> list, String verify, int delayMs) {
        try {
            WxServices s = new WxServices(cl);
            int scene = 2;                                   // 场景值: 2=从群聊(可配)
            Object nsq = s.netSceneQueue();
            Object cb = Proxy.newProxyInstance(cl,
                    new Class<?>[]{ XposedHelpers.findClass(WxConstants.U0, cl) },
                    (proxy, method, args) -> {
                        if (method.getName().equals("onSceneEnd")) {
                            int errType = (Integer) args[1], errCode = (Integer) args[2];
                            String u = (String) XposedHelpers.getAdditionalInstanceField(args[4], "batch_user");
                            android.util.Log.i("WxBatch", u + " -> errType=" + errType + " errCode=" + errCode);
                        }
                        return null;
                    });
            XposedHelpers.callMethod(nsq, "a", 30, cb);       // 注册 NetSceneVerifyUser 回调

            int ok = 0, fail = 0;
            for (Member m : list) {
                if (sendOne(cl, s, room, m.wxid, verify, scene)) ok++; else fail++;
                try { Thread.sleep(delayMs); } catch (InterruptedException ignored) {}
            }
            android.util.Log.i("WxBatch", "done ok=" + ok + " fail=" + fail);
        } catch (Throwable t) { t.printStackTrace(); }
    }

    private boolean sendOne(ClassLoader cl, WxServices s, String room, String wxid,
                            String verify, int scene) {
        try {
            Class<?> p3 = XposedHelpers.findClass(WxConstants.P3, cl);
            LinkedList<String>  users   = new LinkedList<>(); users.add(wxid);
            LinkedList<Integer> scenes  = new LinkedList<>(); scenes.add(scene);
            LinkedList<String>  tickets = new LinkedList<>();
            HashMap<String,Integer> flags = new HashMap<>();
            LinkedList<String>  labels  = new LinkedList<>();

            // ★ 发申请(opcode=1)必须用 11 参构造；4/8参构造断言只允许 opcode==3
            Object sceneObj = XposedHelpers.newInstance(p3,
                    WxConstants.OP_ADDCONTACT,
                    users, scenes, tickets,
                    verify, verify, flags, room, "", "", labels);
            XposedHelpers.setAdditionalInstanceField(sceneObj, "batch_user", wxid);
            return (Boolean) XposedHelpers.callMethod(s.netSceneQueue(), "g", sceneObj);
        } catch (Throwable t) { t.printStackTrace(); return false; }
    }
}

/** 语音转发（拦截 a4.h 逐条转发，type=34 走自定义链路） */
final class VoiceForward {

    static void hook(ClassLoader cl) {
        // a4.h 签名实测: h(Context, String target, e9 msg, boolean, boolean, String)
        XposedHelpers.findAndHookMethod(WxConstants.A4, cl, "h",
                android.content.Context.class, String.class,
                XposedHelpers.findClass(WxConstants.E9, cl),
                boolean.class, boolean.class, String.class,
                new XC_MethodHook() {
                    @Override protected void beforeHookedMethod(MethodHookParam param) {
                        try {
                            Object msg = param.args[2];
                            int type = (Integer) XposedHelpers.callMethod(msg, "getType"); // 语音=34
                            if (type == 34) {
                                String target = (String) param.args[1];
                                new VoiceForward().forward(cl, msg, target);
                                param.setResult(null);   // 跳过原生（原生会丢语音）
                            }
                        } catch (Throwable t) { t.printStackTrace(); }
                    }
                });
    }

    void forward(ClassLoader cl, Object srcMsg, String target) {
        try {
            Object v0 = XposedHelpers.callStaticMethod(XposedHelpers.findClass(WxConstants.V61_V0, cl), "bj");
            Object voiceStore = XposedHelpers.callMethod(v0, "ej");               // v61.m1
            if (voiceStore == null) { log("voiceStore null"); return; }

            String talker = (String) XposedHelpers.callMethod(srcMsg, "N0");      // 会话
            long localId = (Long)  XposedHelpers.callMethod(srcMsg, "getMsgId");
            long svrId   = (Long)  XposedHelpers.callMethod(srcMsg, "F0");

            // 1) 源 voiceinfo: x1(talker,localId) → 兜底 u1(svrId)
            Object vi = XposedHelpers.callMethod(voiceStore, "x1", talker, (int) localId);
            if (vi == null) vi = XposedHelpers.callMethod(voiceStore, "u1", svrId);
            if (vi == null) { log("voiceinfo null"); return; }

            String fileName = (String) XposedHelpers.getObjectField(vi, "b");     // FileName
            int    len      = (Integer) XposedHelpers.getObjectField(vi, "l");    // 时长 s
            int    flag     = (Integer) XposedHelpers.getObjectField(vi, "v");    // VoiceFlag
            String human    = (String) XposedHelpers.getObjectField(vi, "n");     // 原发送人
            int    totalLen = (Integer) XposedHelpers.getObjectField(vi, "h");
            String format   = (String) XposedHelpers.getObjectField(vi, "p");
            byte[] master   = (byte[]) XposedHelpers.getObjectField(vi, "t");
            Object ext      = XposedHelpers.getObjectField(vi, "w");

            // 2) 源文件全路径
            String srcPath = getVoiceFullPath(cl, fileName);
            if (srcPath == null) { log("src path null"); return; }

            // 3) 新文件名 + 目标落盘路径(录制规则 k1.d)
            String prefix = fileName.contains("_")
                    ? fileName.substring(0, fileName.lastIndexOf('_') + 1) : "amr_";
            String newName = (String) XposedHelpers.callStaticMethod(voiceStore.getClass(),
                    "b1", target, prefix);
            String dstPath = (String) XposedHelpers.callStaticMethod(
                    XposedHelpers.findClass(WxConstants.SDK_K1, cl), "d",
                    "voice2/", "msg_", newName, ".amr", 2, true);

            // 4) 复制文件
            XposedHelpers.callStaticMethod(XposedHelpers.findClass(WxConstants.VFS_Z6, cl), "c", srcPath, dstPath);

            // 5) 插新 voiceinfo(User=target, Status=1, 其余照抄源) —— 必须先插, u()内部按fileName查表
            Object c1 = XposedHelpers.newInstance(XposedHelpers.findClass(WxConstants.V61_C1, cl));
            XposedHelpers.setObjectField(c1, "b", newName);
            XposedHelpers.setObjectField(c1, "c", target);
            XposedHelpers.setObjectField(c1, "l", len);
            XposedHelpers.setObjectField(c1, "v", flag | 1);
            XposedHelpers.setObjectField(c1, "n", human);
            XposedHelpers.setObjectField(c1, "h", totalLen);
            XposedHelpers.setObjectField(c1, "p", format);
            XposedHelpers.setObjectField(c1, "t", master);
            XposedHelpers.setObjectField(c1, "w", ext);
            XposedHelpers.setIntField (c1, "i", 1);                  // Status=1 待发
            XposedHelpers.setLongField(c1, "j", System.currentTimeMillis());
            XposedHelpers.callMethod(voiceStore, "y1", c1);          // insert

            // 6) 发送（核心：构造 type=34 消息 → 插消息库 → 解析路径）
            XposedHelpers.callStaticMethod(XposedHelpers.findClass(WxConstants.V61_D1, cl),
                    "u", newName, len, flag | 1, null, "");

            // 7) 踢上传队列
            Object svc = XposedHelpers.callMethod(v0, "dj");          // yl.y0 SceneVoiceService
            XposedHelpers.callMethod(svc, "e");
            log("voice forwarded -> " + target + " name=" + newName);
        } catch (Throwable t) { t.printStackTrace(); }
    }

    /** 源语音全路径：优先 vfs.t7.b("voice2") 目录，兜底 k1.d 按录制规则拼 */
    private static String getVoiceFullPath(ClassLoader cl, String fileName) {
        try {
            Class<?> t7 = XposedHelpers.findClass(WxConstants.VFS_T7, cl);
            if (t7 != null) {
                String dir = (String) XposedHelpers.callStaticMethod(t7, "b", "voice2");
                if (dir != null) {
                    java.io.File f = new java.io.File(dir, fileName);
                    if (f.exists()) return f.getAbsolutePath();
                }
            }
        } catch (Throwable ignored) {}
        try {
            return (String) XposedHelpers.callStaticMethod(
                    XposedHelpers.findClass(WxConstants.SDK_K1, cl), "d",
                    "voice2/", "msg_", fileName, ".amr", 1, false);
        } catch (Throwable t) { return null; }
    }

    private static void log(String s) { android.util.Log.i("VoiceForward", s); }
