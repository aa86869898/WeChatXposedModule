private static final int FRAME_PCM_BYTES = 320 * 1 * 16 / 8;
private static final int SILK_BITRATE = 16000;
private static final int SILK_COMPLEXITY = 5;

// 修复后的encodePcmToSilk编码方法
void encodePcmToSilk(byte[] padPcm, String outPath, int rawPcmLen) {
    SilkWriter writer = yl.g(TARGET_SAMPLE_RATE, SILK_BITRATE);
    writer.setComplexity(SILK_COMPLEXITY);
    writer.b(outPath);
    int frameByteSize = FRAME_PCM_BYTES;
    int totalFrameCount = padPcm.length / frameByteSize;
    for (int i = 0; i < totalFrameCount; i++) {
        int offset = i * frameByteSize;
        byte[] singleFrame = new byte[frameByteSize];
        System.arraycopy(padPcm, offset, singleFrame, 0, frameByteSize);
        boolean isLastFrame = (i == totalFrameCount - 1);
        writer.h0(singleFrame, frameByteSize, isLastFrame);
        writer.silk.a(singleFrame, 0);
    }
    writer.silk.flush();
    writer.d();
}
