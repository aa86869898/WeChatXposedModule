/**
 * QuoteSpeaker — 引用消息自动朗读
 * ====================================
 * 功能: 当别人引用你的消息时，自动通过 TTS 朗读引用内容
 *
 * 原理:
 *   每条新消息入库后会通知到 hk.o.b(e9, j4)
 *   → 检查消息 XML 中是否包含 <refermsg>
 *   → 提取 <fromusr> 与当前用户 wxid 对比
 *   → 匹配成功则调用系统 TTS 朗读: "XX 引用了你: 内容"
 *
 * 微信 8.0.76 实测通过
 */

import android.speech.tts.TextToSpeech;
import android.speech.tts.UtteranceProgressListener;
import java.util.Locale;
import java.util.HashSet;
import java.util.regex.Pattern;
import java.util.regex.Matcher;

// ============================================================
// 全局状态
// ============================================================
tts = null;
spokenMsgIds = new HashSet();
myWxid = "";
initialized = false;

// ============================================================
// 初始化: 获取当前用户 wxid + 创建 TTS 引擎
// ============================================================
init() {
    if (initialized) return;

    // 获取当前用户 wxid
    try {
        import hm0.j1;
        myWxid = j1.u().c().u(2, "");
        log("QuoteSpeaker: 当前用户 wxid = " + myWxid);
    } catch (e) {
        log("QuoteSpeaker: 获取 wxid 失败: " + e.getMessage());
        // 降级方案: 从 kernel 读取
        try {
            import com.tencent.mm.kernel.b;
            myWxid = b.h();
        } catch (e2) {
            log("QuoteSpeaker: 降级获取 wxid 也失败: " + e2.getMessage());
            return;
        }
    }

    if (myWxid == null || myWxid.isEmpty()) {
        log("QuoteSpeaker: wxid 为空，退出初始化");
        return;
    }

    // 初始化 TTS
    try {
        tts = new TextToSpeech(hostContext, status -> {
            if (status == TextToSpeech.SUCCESS) {
                int result = tts.setLanguage(Locale.CHINESE);
                if (result == TextToSpeech.LANG_MISSING_DATA ||
                    result == TextToSpeech.LANG_NOT_SUPPORTED) {
                    log("QuoteSpeaker: 中文不支持，尝试英文");
                    tts.setLanguage(Locale.US);
                }
                log("QuoteSpeaker: TTS 初始化成功");
            } else {
                log("QuoteSpeaker: TTS 初始化失败, status=" + status);
            }
        });
    } catch (e) {
        log("QuoteSpeaker: TTS 创建失败: " + e.getMessage());
    }

    initialized = true;
    log("QuoteSpeaker: 插件初始化完成, myWxid=" + myWxid);
}

// ============================================================
// TTS 朗读方法
// ============================================================
speakOut(text) {
    if (tts == null) {
        log("QuoteSpeaker: TTS 未就绪，跳过朗读");
        return;
    }
    try {
        // Android L+ 使用 QUEUE_FLUSH 覆盖当前朗读队列
        tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, "quote_" + System.currentTimeMillis());
        log("QuoteSpeaker: 已朗读: " + text);
    } catch (e) {
        log("QuoteSpeaker: 朗读异常: " + e.getMessage());
    }
}

// ============================================================
// XML 提取辅助方法
// ============================================================
extractXmlTag(xml, tag) {
    if (xml == null) return "";
    p = Pattern.compile("<" + tag + ">([^<]*)</" + tag + ">");
    m = p.matcher(xml);
    return m.find() ? m.group(1) : "";
}

// ============================================================
// 主逻辑: 通过 DexKit 定位 hk.o 类，Hook b(e9,j4)
// ============================================================
main() {
    log("=== QuoteSpeaker 引用消息朗读插件加载 ===");
    init();

    // DexKit 定位: 包含 "notifyNewMsg" 和 "addMsg" 字符串的类
    targetClass = findClass()
        .usingStrings("notifyNewMsg", "addMsg")
        .single();

    if (targetClass == null) {
        log("QuoteSpeaker: DexKit 未找到目标类，尝试手动类名");
        // 尝试已知类名
        try {
            targetClass = findClass("hk.o", hostLoader);
        } catch (e) {
            log("QuoteSpeaker: 也找不到 hk.o: " + e.getMessage());
        }
    }

    if (targetClass == null) {
        log("QuoteSpeaker: 无法定位消息通知类，插件退出");
        toast("引用朗读: 类定位失败");
        return;
    }

    log("QuoteSpeaker: 找到目标类 = " + targetClass.getName());

    // 查找 b(e9, j4) 方法: 参数个数为 2
    notifyMethod = findMethod()
        .inClass(targetClass)
        .name("b")
        .paramCount(2)
        .single();

    if (notifyMethod == null) {
        // 尝试查找: 参数为 com.tencent.mm.storage.e9 的方法
        notifyMethod = findMethod()
            .inClass(targetClass)
            .params("Lcom/tencent/mm/storage/e9;", "La65/j4;")
            .single();
    }

    if (notifyMethod == null) {
        log("QuoteSpeaker: 未找到 b() 方法，尝试 hookAllMethods");
        hookAllMethods(targetClass, "b", param -> {
            onMsgNotify(param);
        });
        log("QuoteSpeaker: hookAllMethods(b) 已安装");
    } else {
        hookBefore(notifyMethod, param -> {
            onMsgNotify(param);
        });
        log("QuoteSpeaker: hookBefore(b) 已安装");
    }

    toast("引用朗读插件已加载");
    log("=== QuoteSpeaker Hook 安装完成 ===");
}

// ============================================================
// 消息通知回调
// ============================================================
onMsgNotify(param) {
    try {
        msg = param.args[0];

        // 只处理收到的消息 (isSend == 0)
        isSend = getIntField(msg, "field_isSend");
        if (isSend != 0) return;

        // 获取消息内容
        content = callMethod(msg, "j");
        if (content == null || content.isEmpty()) return;

        // 检查是否包含引用
        if (!content.contains("<refermsg>")) return;

        // 提取 fromusr
        fromUsr = extractXmlTag(content, "fromusr");
        if (fromUsr == null || fromUsr.isEmpty()) return;

        // 检查是否引用了"我"
        if (!myWxid.equals(fromUsr)) return;

        // 去重: 用 msgSvrId
        msgSvrId = getLongField(msg, "field_msgSvrId");
        if (spokenMsgIds.contains(msgSvrId)) return;
        spokenMsgIds.add(msgSvrId);

        // 限制去重集合大小
        if (spokenMsgIds.size() > 500) {
            spokenMsgIds.clear();
        }

        // 提取显示名和引用内容
        displayName = extractXmlTag(content, "displayname");
        quoteText = extractXmlTag(content, "content");

        // 获取引用人的备注名(如果为空则用 displayName)
        if (displayName == null || displayName.isEmpty()) {
            displayName = callMethod(msg, "N0");  // N0() = getTalker
            if (displayName == null) displayName = "有人";
        }

        // 构建朗读文本
        speakText = displayName + " 引用你: " + quoteText;

        log("QuoteSpeaker: 检测到引用! from=" + fromUsr + 
            ", displayName=" + displayName + 
            ", text=" + quoteText);

        // Toast 提示
        toast(displayName + " 引用了你的消息");

        // TTS 朗读
        speakOut(speakText);

    } catch (e) {
        log("QuoteSpeaker: onMsgNotify 异常: " + e.getMessage());
    }
}

// 启动
main();
