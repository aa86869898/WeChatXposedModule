package com.leshao.batchadd;

import android.content.Context;
import android.content.pm.PackageInfo;

import java.util.HashMap;
import java.util.Map;

import de.robv.android.xposed.XposedHelpers;

/**
 * 微信符号表（按版本号分发）。
 *
 * 每个微信版本都会重混淆短名（hm0.j1 / e01.v1 / cw1.f ...），
 * 这里用「versionCode -> SymbolSet」映射，命中即用对应版本符号，未命中走 DEFAULT。
 *
 * 用法：
 *   1. 先运行一次，看日志 "[leshao] versionCode=xxxxx"；
 *   2. 把 xxxxx 填进 TABLE，并补上该版本的符号。
 */
public final class SymbolTable {

    public static ClassLoader CL;

    // ================= 运行时符号（init 时按版本填充） =================
    public static String CHATROOM_INFO_UI;
    public static String KERNEL;
    public static String SERVICE_METHOD;
    public static String NET_SERVICE_METHOD;
    public static String NET_QUEUE_FIELD;
    public static String MEMBERS_LOGIC;
    public static String CHATROOM_MGR_IF;
    public static String SELF;
    public static String CONTACT_MGR_IF;
    public static String CONTACT_IMPL;
    public static String NETSCENE_VERIFY;
    public static String LISTENER_IF;

    // ================= 不随版本变的常量 =================
    public static final String EXTRA_ROOM_ID = "RoomInfo_Id";
    public static final int VERIFY_TYPE = 30;   // m3.getType()

    public static int VERSION_CODE = -1;

    private SymbolTable() {}

    public static void init(ClassLoader cl) {
        CL = cl;
        VERSION_CODE = detectVersionCode();
        apply(resolve(VERSION_CODE));
    }

    public static Class<?> cls(String name) {
        return XposedHelpers.findClass(name, CL);
    }

    // ================= 版本分发 =================

    private static final Map<Integer, SymbolSet> TABLE = new HashMap<>();
    private static final SymbolSet DEFAULT;

    static {
        // 已核实的最新版本（兜底默认值）
        DEFAULT = new SymbolSet()
                .chatroomInfoUI("com.tencent.mm.chatroom.ui.ChatroomInfoUI")
                .kernel("hm0.j1")
                .serviceMethod("s")
                .netServiceMethod("n")
                .netQueueField("b")
                .membersLogic("e01.v1")
                .chatroomMgrIf("cw1.f")
                .self("e01.z1")
                .contactMgrIf("sh3.c4")
                .contactImpl("com.tencent.mm.plugin.messenger.foundation.h2")
                .netsceneVerify("com.tencent.mm.pluginsdk.model.m3")
                .listenerIf("com.tencent.mm.modelbase.u0");

        // ===== 按版本覆盖（只写与 DEFAULT 不同的字段） =====
        // 示例（版本号是占位，按日志里的真实 versionCode 填）：
        // TABLE.put(0x28003120, DEFAULT.copy()
        //         .kernel("obf.xxx")
        //         .membersLogic("obf.yyy")
        //         .chatroomMgrIf("obf.zzz"));
    }

    private static SymbolSet resolve(int versionCode) {
        SymbolSet set = TABLE.get(versionCode);
        return set != null ? set : DEFAULT;
    }

    private static void apply(SymbolSet s) {
        CHATROOM_INFO_UI   = s.chatroomInfoUI;
        KERNEL             = s.kernel;
        SERVICE_METHOD     = s.serviceMethod;
        NET_SERVICE_METHOD = s.netServiceMethod;
        NET_QUEUE_FIELD    = s.netQueueField;
        MEMBERS_LOGIC      = s.membersLogic;
        CHATROOM_MGR_IF    = s.chatroomMgrIf;
        SELF               = s.self;
        CONTACT_MGR_IF     = s.contactMgrIf;
        CONTACT_IMPL       = s.contactImpl;
        NETSCENE_VERIFY    = s.netsceneVerify;
        LISTENER_IF        = s.listenerIf;
    }

    /** 通过 ActivityThread 拿 Context，再读微信 versionCode */
    private static int detectVersionCode() {
        try {
            Class<?> at = Class.forName("android.app.ActivityThread");
            Object thread = at.getMethod("currentActivityThread").invoke(null);
            Object app = at.getMethod("getApplication").invoke(thread);
            if (app instanceof Context) {
                PackageInfo pi = ((Context) app).getPackageManager()
                        .getPackageInfo("com.tencent.mm", 0);
                return pi.versionCode;
            }
        } catch (Throwable ignored) {
        }
        return -1;
    }

    // ================= 符号集 =================

    static class SymbolSet {
        String chatroomInfoUI;
        String kernel;
        String serviceMethod;
        String netServiceMethod;
        String netQueueField;
        String membersLogic;
        String chatroomMgrIf;
        String self;
        String contactMgrIf;
        String contactImpl;
        String netsceneVerify;
        String listenerIf;

        SymbolSet chatroomInfoUI(String v) { this.chatroomInfoUI = v; return this; }
        SymbolSet kernel(String v)           { this.kernel = v; return this; }
        SymbolSet serviceMethod(String v)    { this.serviceMethod = v; return this; }
        SymbolSet netServiceMethod(String v) { this.netServiceMethod = v; return this; }
        SymbolSet netQueueField(String v)    { this.netQueueField = v; return this; }
        SymbolSet membersLogic(String v)     { this.membersLogic = v; return this; }
        SymbolSet chatroomMgrIf(String v)    { this.chatroomMgrIf = v; return this; }
        SymbolSet self(String v)             { this.self = v; return this; }
        SymbolSet contactMgrIf(String v)     { this.contactMgrIf = v; return this; }
        SymbolSet contactImpl(String v)      { this.contactImpl = v; return this; }
        SymbolSet netsceneVerify(String v)   { this.netsceneVerify = v; return this; }
        SymbolSet listenerIf(String v)       { this.listenerIf = v; return this; }

        SymbolSet copy() {
            SymbolSet c = new SymbolSet();
            c.chatroomInfoUI   = chatroomInfoUI;
            c.kernel           = kernel;
            c.serviceMethod    = serviceMethod;
            c.netServiceMethod = netServiceMethod;
            c.netQueueField    = netQueueField;
            c.membersLogic     = membersLogic;
            c.chatroomMgrIf    = chatroomMgrIf;
            c.self             = self;
            c.contactMgrIf     = contactMgrIf;
            c.contactImpl      = contactImpl;
            c.netsceneVerify   = netsceneVerify;
            c.listenerIf       = listenerIf;
            return c;
        }
    }
}

package com.leshao.batchadd;

import de.robv.android.xposed.XSharedPreferences;

/** 模块配置（MainActivity 写，HookMain 在微信进程读）。 */
public final class Config {

    public static final String MODULE_PKG = "com.leshao.batchadd";
    public static final String PREFS = "leshao_config";

    public static boolean excludeOwner  = true;
    public static boolean excludeAdmin  = true;
    public static boolean excludeFriend = true;
    public static int     scene         = 0;

    private Config() {}

    /** 在微信进程里读取模块自身配置（XSharedPreferences 跨进程读模块数据目录） */
    public static void load() {
        try {
            XSharedPreferences sp = new XSharedPreferences(MODULE_PKG, PREFS);
            sp.reload();
            excludeOwner  = sp.getBoolean("exclude_owner", true);
            excludeAdmin  = sp.getBoolean("exclude_admin", true);
            excludeFriend = sp.getBoolean("exclude_friend", true);
            scene         = parseInt(sp.getString("scene", "0"), 0);
        } catch (Throwable t) {
            // 读取失败则用默认值
        }
    }

    private static int parseInt(String s, int def) {
        try {
            return Integer.parseInt(s.trim());
        } catch (Throwable t) {
            return def;
        }
    }
}

package com.leshao.batchadd;

import android.app.Activity;
import android.os.Handler;
import android.os.Looper;
import android.view.MenuItem;

import java.lang.reflect.InvocationHandler;
import java.lang.reflect.Method;
import java.lang.reflect.Proxy;
import java.util.List;

import de.robv.android.xposed.IXposedHookLoadPackage;
import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.XposedHelpers;
import de.robv.android.xposed.callbacks.XC_LoadPackage;

public class HookMain implements IXposedHookLoadPackage {

    private static final String PKG_WX = "com.tencent.mm";
    private static final Handler sMain = new Handler(Looper.getMainLooper());

    @Override
    public void handleLoadPackage(XC_LoadPackage.LoadPackageParam lp) {
        if (!PKG_WX.equals(lp.packageName)) return;
        SymbolTable.init(lp.classLoader);
        Config.load();

        XposedBridge.log("[leshao] versionCode=" + SymbolTable.VERSION_CODE
                + "  excludeOwner=" + Config.excludeOwner
                + "  excludeAdmin=" + Config.excludeAdmin
                + "  excludeFriend=" + Config.excludeFriend
                + "  scene=" + Config.scene);

        try {
            XposedHelpers.findAndHookMethod(SymbolTable.CHATROOM_INFO_UI, lp.classLoader,
                    "onCreate", android.os.Bundle.class, new XC_MethodHook() {
                        @Override protected void afterHookedMethod(MethodHookParam p) {
                            injectButton((Activity) p.thisObject);
                        }
                    });
        } catch (Throwable t) {
            XposedBridge.log("[leshao] hook onCreate fail: " + t);
        }
    }

    // ---------- ① 注入按钮 ----------
    private void injectButton(final Activity act) {
        sMain.post(new Runnable() {
            @Override public void run() {
                try {
                    Method addText = XposedHelpers.findMethodExact(
                            "com.tencent.mm.ui.MMActivity", SymbolTable.CL,
                            "addTextOptionMenu",
                            int.class, String.class, int.class,
                            MenuItem.OnMenuItemClickListener.class);
                    addText.invoke(act, 0x5E5E, "批量加友", 0,
                            new MenuItem.OnMenuItemClickListener() {
                                @Override public boolean onMenuItemClick(MenuItem item) {
                                    doBatchAdd(act);
                                    return true;
                                }
                            });
                } catch (Throwable t) {
                    XposedBridge.log("[leshao] injectButton fail: " + t);
                }
            }
        });
    }

    // ---------- ② 核心业务 ----------
    private void doBatchAdd(final Activity act) {
        new Thread(new Runnable() {
            @Override public void run() {
                try {
                    final String roomName = act.getIntent().getStringExtra(SymbolTable.EXTRA_ROOM_ID);
                    if (roomName == null) return;

                    List<String> members = (List<String>) XposedHelpers.callStaticMethod(
                            SymbolTable.cls(SymbolTable.MEMBERS_LOGIC), "m", roomName);
                    if (members == null || members.isEmpty()) return;

                    String owner = getRoomOwner(roomName);
                    String self  = (String) XposedHelpers.callStaticMethod(
                            SymbolTable.cls(SymbolTable.SELF), "r");

                    Object storage = getContactStorage();
                    Object queue   = getNetSceneQueue();

                    int done = 0;
                    for (String u : members) {
                        if (u == null || u.equals(self)) continue;               // 自己（恒排除）

                        if (Config.excludeOwner && u.equals(owner)) continue;   // 群主

                        Object contact = XposedHelpers.callMethod(storage, "n", u, true);
                        if (contact == null) continue;
                        int type = (Integer) XposedHelpers.callMethod(contact, "getType");
                        if (Config.excludeFriend && (type & 0x1) != 0) continue; // 已是好友
                        if (Config.excludeAdmin && isAdmin(contact)) continue;   // 管理员

                        Object scene = XposedHelpers.newInstance(
                                SymbolTable.cls(SymbolTable.NETSCENE_VERIFY),
                                3, u, "", Config.scene);                        // opcode=3
                        XposedHelpers.callMethod(queue, "g", scene);            // doScene
                        done++;
                        Thread.sleep(300);                                      // 限速
                    }

                    final int n = done;
                    sMain.post(new Runnable() {
                        @Override public void run() {
                            android.widget.Toast.makeText(act, "已发起 " + n + " 个好友请求",
                                    android.widget.Toast.LENGTH_SHORT).show();
                        }
                    });
                } catch (Throwable t) {
                    XposedBridge.log("[leshao] doBatchAdd fail: " + t);
                }
            }
        }).start();
    }

    // ---------- ③ 工具 ----------
    private static Object svc(String iface) {
        return XposedHelpers.callStaticMethod(
                SymbolTable.cls(SymbolTable.KERNEL), SymbolTable.SERVICE_METHOD,
                SymbolTable.cls(iface));
    }

    private static Object getChatroomInfo(String roomName) {
        Object mgr     = svc(SymbolTable.CHATROOM_MGR_IF);          // bw1.a
        Object storage = XposedHelpers.callMethod(mgr, "a");        // a3
        return XposedHelpers.callMethod(storage, "H0", roomName);   // z2(y1)
    }

    private static String getRoomOwner(String roomName) {
        Object room = getChatroomInfo(roomName);
        if (room == null) return null;
        return (String) XposedHelpers.getObjectField(room, "field_roomowner");
    }

    private static Object getContactStorage() {
        Object mgr = svc(SymbolTable.CONTACT_MGR_IF);               // messenger.foundation.h2
        return XposedHelpers.callMethod(mgr, "ij");                 // j4
    }

    private static boolean isAdmin(Object contact) {
        if (contact == null) return false;
        try {
            int flag = XposedHelpers.getIntField(contact, "field_chatroomFlag");
            return (flag & 2) != 0;   // TODO: 用 Probe 实测确定管理员 bit
        } catch (Throwable t) {
            try {
                return XposedHelpers.getIntField(contact, "T") == 0; // 本版本 k7() 判据
            } catch (Throwable t2) {
                return false;
            }
        }
    }

    private static Object getNetSceneQueue() {
        Object net = XposedHelpers.callStaticMethod(
                SymbolTable.cls(SymbolTable.KERNEL), SymbolTable.NET_SERVICE_METHOD);
        return XposedHelpers.getObjectField(net, SymbolTable.NET_QUEUE_FIELD); // 字段 -> r1
    }

    // ---------- ④ 可选：结果回调 ----------
    private static void registerCallback(Object queue) {
        Class<?> u0 = SymbolTable.cls(SymbolTable.LISTENER_IF);
        Object listener = Proxy.newProxyInstance(SymbolTable.CL, new Class<?>[]{u0},
                new InvocationHandler() {
                    @Override public Object invoke(Object proxy, Method m, Object[] a) {
                        if ("onSceneEnd".equals(m.getName()) && a != null && a.length >= 3) {
                            XposedBridge.log("[leshao] onSceneEnd errType=" + a[0]
                                    + " errCode=" + a[1] + " msg=" + a[2]);
                        }
                        return null;
                    }
                });
        XposedHelpers.callMethod(queue, "a",
                new Class<?>[]{int.class, u0}, SymbolTable.VERIFY_TYPE, listener);
    }
}
