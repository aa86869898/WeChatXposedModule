    public static void installTagFilterBar(final ClassLoader cl) {
        try {
            Class<?> mainUI = XposedHelpers.findClass(
                "com.tencent.mm.ui.conversation.MainUI", cl);

            XposedHelpers.findAndHookMethod(mainUI, "onResume",
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam p) {
                        new Handler(Looper.getMainLooper()).postDelayed(() -> {
                            tryInjectTagBar(p.thisObject);
                        }, 400);
                    }
                });

            XposedBridge.log(TAG + ": ✓ 标签筛选栏 Hook 已安装");
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": ✗ installTagFilterBar: " + e.getMessage());
        }
    }

    private static void tryInjectTagBar(Object mainUI) {
        try {
            if (sInjectedBar != null && sInjectedBar.getParent() != null) {
                refreshTagChips();
                return;
            }
            View root = (View) XposedHelpers.getObjectField(mainUI, "G");
            if (root == null) return;

            // 查找垂直 LinearLayout 作为父容器
            ViewGroup parent = findVerticalLinearLayout(root);
            if (parent == null) parent = (ViewGroup) root;

            // 找到 RecyclerView/ListView 的位置
            int idx = findListIndex(parent);
            if (idx < 0) idx = 0;

            sInjectedBar = buildTagBar(root.getContext());
            parent.addView(sInjectedBar, idx,
                new ViewGroup.LayoutParams(
                    ViewGroup.LayoutParams.MATCH_PARENT,
                    dp(44, root.getContext())));

            refreshTagChips();
            XposedBridge.log(TAG + ": ✓ 标签栏注入成功");
        } catch (Throwable e) {
            // 静默失败，下次 onResume 重试
        }
    }

    private static View buildTagBar(Context ctx) {
        HorizontalScrollView hsv = new HorizontalScrollView(ctx);
        hsv.setHorizontalScrollBarEnabled(false);
        hsv.setBackgroundColor(Color.parseColor("#F5F5F5"));
        hsv.setElevation(dp(2, ctx));

        LinearLayout ll = new LinearLayout(ctx);
        ll.setOrientation(LinearLayout.HORIZONTAL);
        ll.setGravity(Gravity.CENTER_VERTICAL);
        ll.setPadding(dp(8, ctx), dp(4, ctx), dp(8, ctx), dp(4, ctx));
        hsv.addView(ll);
        sTagContainer = ll;
        sTagBar = hsv;
        return hsv;
    }

    public static void refreshTagChips() {
        if (sTagContainer == null) return;
        sLabels = LabelManager.getAll();
        Context ctx = sTagContainer.getContext();
        sTagContainer.removeAllViews();

        // "全部"
        sTagContainer.addView(makeChip(ctx, "全部", -1, sSelectedLabelId == -1));
        sTagContainer.addView(spacer(ctx));

        for (LabelInfo l : sLabels) {
            sTagContainer.addView(makeChip(ctx, l.labelName,
                l.labelId, sSelectedLabelId == l.labelId));
        }

        sTagContainer.addView(spacer(ctx));
        sTagContainer.addView(makeAddBtn(ctx));
    }

    private static TextView makeChip(Context ctx, String text, int id, boolean sel) {
        TextView tv = new TextView(ctx);
        tv.setText(text);
        tv.setTextSize(13);
        tv.setPadding(dp(12, ctx), dp(6, ctx), dp(12, ctx), dp(6, ctx));
        tv.setGravity(Gravity.CENTER);

        GradientDrawable bg = new GradientDrawable();
        bg.setCornerRadius(dp(16, ctx));
        if (sel) {
            bg.setColor(Color.parseColor("#07C160"));
            tv.setTextColor(Color.WHITE);
            tv.setTypeface(null, Typeface.BOLD);
        } else {
            bg.setColor(Color.WHITE);
            tv.setTextColor(Color.parseColor("#333333"));
            bg.setStroke(1, Color.parseColor("#E0E0E0"));
        }
        tv.setBackground(bg);

        LinearLayout.LayoutParams lp = new LinearLayout.LayoutParams(
            ViewGroup.LayoutParams.WRAP_CONTENT, ViewGroup.LayoutParams.WRAP_CONTENT);
        lp.setMargins(dp(4, ctx), 0, dp(4, ctx), 0);
        tv.setLayoutParams(lp);

        tv.setOnClickListener(v -> {
            sSelectedLabelId = (sSelectedLabelId == id) ? -1 : id;
            refreshTagChips();
            applyFilter(sSelectedLabelId);
        });

        if (id > 0) {
            tv.setOnLongClickListener(v -> {
                showLabelManage(ctx, id, text);
                return true;
            });
        }
        return tv;
    }

    private static TextView makeAddBtn(Context ctx) {
        TextView tv = new TextView(ctx);
        tv.setText("＋");
        tv.setTextSize(16);
        tv.setTextColor(Color.parseColor("#07C160"));
        tv.setPadding(dp(10, ctx), dp(6, ctx), dp(10, ctx), dp(6, ctx));
        tv.setGravity(Gravity.CENTER);
        tv.setTypeface(null, Typeface.BOLD);

        GradientDrawable bg = new GradientDrawable();
        bg.setCornerRadius(dp(16, ctx));
        bg.setColor(Color.WHITE);
        bg.setStroke(1, Color.parseColor("#07C160"));
        tv.setBackground(bg);

        tv.setOnClickListener(v -> showCreateDialog(ctx, null));
        return tv;
    }

    private static View spacer(Context ctx) {
        View v = new View(ctx);
        v.setLayoutParams(new LinearLayout.LayoutParams(dp(1, ctx),
            ViewGroup.LayoutParams.MATCH_PARENT));
        return v;
    }

    private static void applyFilter(int labelId) {
        if (sTagContainer == null) return;
        if (labelId <= 0) {
            Toast.makeText(sTagContainer.getContext(), "显示全部", Toast.LENGTH_SHORT).show();
        } else {
            List<String> cs = LabelManager.getContacts(labelId);
            Toast.makeText(sTagContainer.getContext(),
                "筛选: " + cs.size() + " 个联系人", Toast.LENGTH_SHORT).show();
        }
    }

    /**
     * Hook r3.onCreateContextMenu —— 微信聊天列表长按菜单的真正入口
     * 
     * r3 = com.tencent.mm.ui.conversation.r3 (ConversationLongClickListener)
     * r3.g = 当前长按的 username (String)
     * r3.h = 当前长按的 conversation k4 对象
     * r3.f = Activity 上下文
     */
    public static void installContextMenuHook(final ClassLoader cl) {
        try {
            Class<?> r3Class = XposedHelpers.findClass(
                "com.tencent.mm.ui.conversation.r3", cl);

            XposedHelpers.findAndHookMethod(r3Class,
                "onCreateContextMenu",
                ContextMenu.class,
                View.class,
                ContextMenu.ContextMenuInfo.class,
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam param) {
                        try {
                            injectMenuItems(param);
                        } catch (Throwable e) {
                            XposedBridge.log(TAG + ": injectMenuItems error: " + e.getMessage());
                        }
                    }
                });

            XposedBridge.log(TAG + ": ✓ 长按菜单 Hook 已安装 (r3.onCreateContextMenu)");
        } catch (Throwable e) {
            XposedBridge.log(TAG + ": ✗ installContextMenuHook: " + e.getMessage());
        }
    }

    /**
     * 在微信原生菜单项之后，注入自定义菜单
     */
    private static void injectMenuItems(MethodHookParam param) {
        ContextMenu menu = (ContextMenu) param.args[0];
        Object r3Instance = param.thisObject;

        // 获取当前长按的 username
        String username = (String) XposedHelpers.getObjectField(r3Instance, "g");
        if (TextUtils.isEmpty(username)) return;

        // 获取该联系人当前的标签
        int[] currentLabels = ContactHelper.getLabelIds(username);
        Set<Integer> currentSet = new HashSet<>();
        for (int id : currentLabels) currentSet.add(id);

        List<LabelInfo> allLabels = LabelManager.getAll();
        if (allLabels.isEmpty()) return;

        // ----- 获取 AdapterContextMenuInfo 以获取 position -----
        ContextMenu.ContextMenuInfo info = (ContextMenu.ContextMenuInfo) param.args[2];
        int position = 0;
        if (info instanceof AdapterView.AdapterContextMenuInfo) {
            position = ((AdapterView.AdapterContextMenuInfo) info).position;
        }

        // ----- 添加分隔线 -----
        // 为每个标签创建菜单项（最多显示 8 个标签 + "更多"）
        int maxShow = Math.min(allLabels.size(), 8);

        // 分组标题（不可点击）
        SubMenu groupSubMenu = menu.addSubMenu(MENU_GROUP_ID, View.NO_ID, 0, "▸ 分组管理");

        for (int i = 0; i < maxShow; i++) {
            final LabelInfo label = allLabels.get(i);
            final boolean hasLabel = currentSet.contains(label.labelId);
            final String labelTitle = (hasLabel ? "☑ " : "☐ ") + label.labelName;

            MenuItem item = groupSubMenu.add(
                MENU_GROUP_ID + 1,
                MENU_GROUP_ID + 100 + label.labelId,
                i,
                labelTitle
            );

            // ★ 关键: 直接设置 OnMenuItemClickListener ★
            item.setOnMenuItemClickListener(clickedItem -> {
                if (hasLabel) {
                    ContactHelper.removeLabel(username, label.labelId);
                    Toast.makeText(getContext(r3Instance),
                        "已从「" + label.labelName + "」移除", Toast.LENGTH_SHORT).show();
                } else {
                    ContactHelper.addLabel(username, label.labelId);
                    Toast.makeText(getContext(r3Instance),
                        "已添加到「" + label.labelName + "」", Toast.LENGTH_SHORT).show();
                }
                return true;
            });
        }

        // "更多分组..." — 打开完整管理弹窗
        if (allLabels.size() > maxShow) {
            MenuItem moreItem = groupSubMenu.add(
                MENU_GROUP_ID + 1,
                MENU_GROUP_ID + 999,
                maxShow,
                "更多分组..."
            );
            moreItem.setOnMenuItemClickListener(clickedItem -> {
                Context ctx = getContext(r3Instance);
                showFullGroupDialog(ctx, username, currentSet, allLabels);
                return true;
            });
        }

        // "新建分组" — 快捷创建
        MenuItem newItem = groupSubMenu.add(
            MENU_GROUP_ID + 1,
            MENU_GROUP_ID + 998,
            maxShow + 1,
            "＋ 新建分组"
        );
        newItem.setOnMenuItemClickListener(clickedItem -> {
            Context ctx = getContext(r3Instance);
            showCreateDialog(ctx, username);
            return true;
        });
    }

    /**
     * 完整分组管理弹窗

    private static void showFullGroupDialog(Context ctx, String username,
                                            Set<Integer> currentSet, List<LabelInfo> allLabels) {
        String[] names = new String[allLabels.size()];
        boolean[] checked = new boolean[allLabels.size()];
        for (int i = 0; i < allLabels.size(); i++) {
            names[i] = allLabels.get(i).labelName;
            checked[i] = currentSet.contains(allLabels.get(i).labelId);
        }

        new AlertDialog.Builder(ctx)
            .setTitle("管理分组")
            .setMultiChoiceItems(names, checked, (dialog, which, isChecked) -> {
                LabelInfo label = allLabels.get(which);
                if (isChecked) {
                    ContactHelper.addLabel(username, label.labelId);
                } else {
                    ContactHelper.removeLabel(username, label.labelId);
                }
            })
            .setPositiveButton("完成", null)
            .setNeutralButton("＋ 新建", (dialog, which) -> showCreateDialog(ctx, username))
            .show();
    }

    /**
     * 新建分组弹窗
     */
    private static void showCreateDialog(Context ctx, String username) {
        EditText input = new EditText(ctx);
        input.setHint("输入新分组名称");
        input.setPadding(40, 30, 40, 30);

        new AlertDialog.Builder(ctx)
            .setTitle("新建分组")
            .setView(input)
            .setPositiveButton("创建", (dialog, which) -> {
                String name = input.getText().toString().trim();
                if (!TextUtils.isEmpty(name)) {
                    LabelInfo created = LabelManager.create(name);
                    if (created != null) {
                        if (username != null) {
                            ContactHelper.addLabel(username, created.labelId);
                        }
                        refreshTagChips();
                        Toast.makeText(ctx, "已创建「" + name + "」", Toast.LENGTH_SHORT).show();
                    }
                }
            })
            .setNegativeButton("取消", null)
            .show();
    }

    /**
     * 标签管理弹窗（长按筛选栏 Chip 触发）
     */
    private static void showLabelManage(Context ctx, int labelId, String labelName) {
        int contactCount = LabelManager.getContacts(labelId).size();
        new AlertDialog.Builder(ctx)
            .setTitle("管理: " + labelName)
            .setItems(new String[]{
                "查看联系人 (" + contactCount + "人)",
                "重命名",
                "删除分组",
                "合并到其他分组"
            }, (dialog, which) -> {
                switch (which) {
                    case 0: {
                        List<String> contacts = LabelManager.getContacts(labelId);
                        StringBuilder sb = new StringBuilder();
                        int max = Math.min(contacts.size(), 50);
                        for (int i = 0; i < max; i++) {
                            if (i > 0) sb.append("\n");
                            sb.append(contacts.get(i));
                        }
                        if (contacts.size() > 50)
                            sb.append("\n... 共 ").append(contacts.size()).append(" 人");
                        new AlertDialog.Builder(ctx)
                            .setTitle(labelName)
                            .setMessage(sb.toString())
                            .setPositiveButton("关闭", null)
                            .show();
                        break;
                    }
                    case 1: {
                        EditText et = new EditText(ctx);
                        et.setText(labelName);
                        et.setPadding(40, 30, 40, 30);
                        new AlertDialog.Builder(ctx)
                            .setTitle("重命名")
                            .setView(et)
                            .setPositiveButton("确定", (d, w) -> {
                                String n = et.getText().toString().trim();
                                if (!TextUtils.isEmpty(n)) {
                                    LabelManager.rename(String.valueOf(labelId), n);
                                    refreshTagChips();
                                }
                            })
                            .setNegativeButton("取消", null)
                            .show();
                        break;
                    }
                    case 2:
                        new AlertDialog.Builder(ctx)
                            .setTitle("确认删除")
                            .setMessage("删除「" + labelName + "」？\n联系人关联将被清除。")
                            .setPositiveButton("删除", (d, w) -> {
                                LabelManager.delete(String.valueOf(labelId));
                                refreshTagChips();
                            })
                            .setNegativeButton("取消", null)
                            .show();
                        break;
                    case 3: {
                        List<LabelInfo> labels = LabelManager.getAll();
                        String[] ns = new String[labels.size()];
                        for (int i = 0; i < labels.size(); i++)
                            ns[i] = labels.get(i).labelName;
                        new AlertDialog.Builder(ctx)
                            .setTitle("合并到...")
                            .setItems(ns, (d, i2) -> {
                                LabelInfo tgt = labels.get(i2);
                                if (tgt.labelId != labelId) {
                                    BatchOperator.mergeLabels(labelId, tgt.labelId);
                                    refreshTagChips();
                                }
                            })
                            .setNegativeButton("取消", null)
                            .show();
                        break;
                    }
                }
            })
            .setNegativeButton("取消", null)
            .show();
    }

    /** 从 r3 实例获取 Context */
    private static Context getContext(Object r3Instance) {
        try {
            return (Activity) XposedHelpers.getObjectField(r3Instance, "f");
        } catch (Throwable e) {
            return sTagContainer != null ? sTagContainer.getContext() : null;
        }
    }

    private static ViewGroup findVerticalLinearLayout(View root) {
        if (root instanceof LinearLayout &&
            ((LinearLayout) root).getOrientation() == LinearLayout.VERTICAL &&
            ((ViewGroup) root).getChildCount() >= 2) {
            return (ViewGroup) root;
        }
        if (root instanceof ViewGroup) {
            ViewGroup vg = (ViewGroup) root;
            for (int i = 0; i < vg.getChildCount(); i++) {
                ViewGroup f = findVerticalLinearLayout(vg.getChildAt(i));
                if (f != null) return f;
            }
        }
        return null;
    }

    private static int findListIndex(ViewGroup parent) {
        for (int i = 0; i < parent.getChildCount(); i++) {
            View child = parent.getChildAt(i);
            String cls = child.getClass().getName();
            if (cls.contains("RecyclerView") || cls.contains("ListView")) {
                return i;
            }
        }
        return -1;
    }

    private static int dp(int dpi, Context ctx) {
        return (int) (dpi * ctx.getResources().getDisplayMetrics().density + 0.5f);
    }
