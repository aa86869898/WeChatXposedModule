# 微信 TTS 自动朗读 Xposed 模块 — 完整代码

## 功能列表

1. ✅ 收到文字消息自动播报（格式：某某某说:XXX内容）
2. ✅ 收到语音消息自动播报（格式：某某某发来语音，请在手机上收听）
3. ✅ 收到图片自动播报（格式：某某某发来一张照片）
4. ✅ 收到视频自动播报（格式：某某某发来一段视频）
5. ✅ 收到位置自动播报（格式：某某某发来定位在:详细位置）
6. ✅ 抢到红包播报金额（某某某的红包:恭喜发财，你抢到了X.XX元）
7. ✅ 自动收款/转账播报（收到某某某转账X.XX元）
8. ✅ 是否播报群聊消息（XX群的XXX发来:XXX）
9. ✅ 自定义播报时间段（默认08:00~23:00）
10. ✅ 自定义播报名单（白名单/黑名单）
11. ✅ 零第三方依赖，纯 Xposed + 系统 TTS

# 七、MainHook.java — Xposed 入口

```java
package com.example.wechattts;

import android.content.Context;
import android.content.SharedPreferences;
import de.robv.android.xposed.IXposedHookLoadPackage;
import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.XposedHelpers;
import de.robv.android.xposed.callbacks.XC_LoadPackage;

public class MainHook implements IXposedHookLoadPackage {

    static final String PKG_WECHAT = "com.tencent.mm";
    static final String PREFS_NAME = "wechat_tts_prefs";

    static Context sAppContext;
    static TtsEngine sTtsEngine;
    static MessageHandler sMsgHandler;
    static RedPacketHandler sRedPktHandler;
    static TransferHandler sTransferHandler;
    static FilterManager sFilterMgr;
    static NicknameResolver sNickResolver;
    static SharedPreferences sPrefs;

    @Override
    public void handleLoadPackage(XC_LoadPackage.LoadPackageParam lpparam) throws Throwable {
        if (!PKG_WECHAT.equals(lpparam.packageName)) return;
        XposedBridge.log("[WeChatTTS] 开始Hook...");

        // 1. 在 SplashActivity 初始化所有组件
        XposedHelpers.findAndHookMethod(
            PKG_WECHAT + ".app.WeChatSplashActivity",
            lpparam.classLoader, "onCreate",
            android.os.Bundle.class, new XC_MethodHook() {
                @Override
                protected void afterHookedMethod(MethodHookParam param) {
                    if (sAppContext != null) return;
                    sAppContext = ((android.app.Activity) param.thisObject)
                        .getApplicationContext();
                    sPrefs = sAppContext.getSharedPreferences(PREFS_NAME, 0);
                    sTtsEngine = new TtsEngine(sAppContext);
                    sFilterMgr = new FilterManager(sPrefs);
                    sNickResolver = new NicknameResolver();
                    sMsgHandler = new MessageHandler(sTtsEngine, sFilterMgr, sNickResolver);
                    sRedPktHandler = new RedPacketHandler(sTtsEngine, sFilterMgr, sNickResolver);
                    sTransferHandler = new TransferHandler(sTtsEngine, sFilterMgr, sNickResolver);
                    XposedBridge.log("[WeChatTTS] 初始化完成");
                }
            }
        );

        // 2. Hook 消息入库
        hookMsgInsert(lpparam.classLoader);

        // 3. Hook 红包领取
        hookRedPacket(lpparam.classLoader);

        // 4. Hook 转账收款
        hookTransfer(lpparam.classLoader);
    }

    // ==================== 消息入库 Hook ====================
    private void hookMsgInsert(ClassLoader cl) {
        try {
            Class<?> msgLogicCls = cl.loadClass("e01.x9");
            Class<?> msgInfoCls = cl.loadClass(PKG_WECHAT + ".storage.e9");

            XposedHelpers.findAndHookMethod(msgLogicCls, "e",
                msgInfoCls, boolean.class, new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam param) {
                        try {
                            Object msgInfo = param.args[0];
                            boolean isNew = (boolean) param.args[1];
                            if (!isNew) return;

                            int msgType = (int) XposedHelpers.callMethod(msgInfo, "getType");
                            int isSend = (int) XposedHelpers.callMethod(msgInfo, "O0");
                            if (isSend == 1) return; // 不读自己发的

                            String talker = (String) XposedHelpers.callMethod(msgInfo, "N0");
                            String content = (String) XposedHelpers.callMethod(msgInfo, "I0");

                            if (!sFilterMgr.shouldProcess(talker, msgType)) return;
                            sMsgHandler.handleMessage(talker, msgType, content, msgInfo);

                        } catch (Throwable t) {
                            XposedBridge.log("[WeChatTTS] MsgErr: " + t);
                        }
                    }
                });
            XposedBridge.log("[WeChatTTS] 消息Hook OK");
        } catch (Throwable t) {
            XposedBridge.log("[WeChatTTS] MsgHook fail: " + t);
        }
    }

    // ==================== 红包领取 Hook ====================
    private void hookRedPacket(ClassLoader cl) {
        hookOnSceneEnd(cl, PKG_WECHAT + ".plugin.luckymoney.ui.LuckyMoneyNewReceiveUI");
        hookOnSceneEnd(cl, PKG_WECHAT + ".plugin.luckymoney.ui.LuckyMoneyDetailUI");
    }

    private void hookOnSceneEnd(ClassLoader cl, String className) {
        try {
            Class<?> uiCls = cl.loadClass(className);
            XposedHelpers.findAndHookMethod(uiCls, "onSceneEnd",
                int.class, int.class, String.class,
                cl.loadClass(PKG_WECHAT + ".modelbase.m1"),
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam param) {
                        try {
                            int errType = (int) param.args[0];
                            int errCode = (int) param.args[1];
                            if (errType != 0 || errCode != 0) return;
                            Object resp = param.args[3];
                            sRedPktHandler.handleReceiveResult(param.thisObject, resp);
                        } catch (Throwable t) {
                            XposedBridge.log("[WeChatTTS] RedPktErr: " + t);
                        }
                    }
                });
            XposedBridge.log("[WeChatTTS] 红包Hook OK: " + className);
        } catch (Throwable t) {
            XposedBridge.log("[WeChatTTS] RedPkt fail: " + className);
        }
    }

    // ==================== 转账收款 Hook ====================
    private void hookTransfer(ClassLoader cl) {
        try {
            Class<?> uiCls = cl.loadClass(
                PKG_WECHAT + ".plugin.remittance.ui.RemittanceDetailUI");
            XposedHelpers.findAndHookMethod(uiCls, "onSceneEnd",
                int.class, int.class, String.class,
                cl.loadClass(PKG_WECHAT + ".modelbase.m1"),
                new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam param) {
                        try {
                            int errType = (int) param.args[0];
                            int errCode = (int) param.args[1];
                            if (errType != 0 || errCode != 0) return;
                            Object resp = param.args[3];
                            sTransferHandler.handleReceiveResult(param.thisObject, resp);
                        } catch (Throwable t) {
                            XposedBridge.log("[WeChatTTS] TransferErr: " + t);
                        }
                    }
                });
            XposedBridge.log("[WeChatTTS] 转账Hook OK");
        } catch (Throwable t) {
            XposedBridge.log("[WeChatTTS] TransferHook fail: " + t);
        }
    }
}
```

# 八、TtsEngine.java — TTS 引擎封装

```java
package com.example.wechattts;

import android.content.Context;
import android.speech.tts.TextToSpeech;
import android.speech.tts.UtteranceProgressListener;
import android.os.Build;
import java.util.Locale;
import java.util.LinkedList;
import java.util.Queue;
import de.robv.android.xposed.XposedBridge;

public class TtsEngine {

    private TextToSpeech mTts;
    private boolean mReady = false;
    private final Queue<String> mQueue = new LinkedList<>();

    public TtsEngine(Context ctx) {
        mTts = new TextToSpeech(ctx, status -> {
            if (status == TextToSpeech.SUCCESS) {
                int result = mTts.setLanguage(Locale.CHINESE);
                if (result == TextToSpeech.LANG_MISSING_DATA
                    || result == TextToSpeech.LANG_NOT_SUPPORTED) {
                    mTts.setLanguage(Locale.CHINESE);
                }
                mTts.setSpeechRate(1.1f);
                mTts.setPitch(1.0f);
                mReady = true;
                XposedBridge.log("[WeChatTTS] TTS初始化成功");
                flushQueue();
            } else {
                XposedBridge.log("[WeChatTTS] TTS初始化失败: " + status);
            }
        });
        mTts.setOnUtteranceProgressListener(new UtteranceProgressListener() {
            @Override public void onStart(String utteranceId) {}
            @Override public void onDone(String utteranceId) { flushQueue(); }
            @Override public void onError(String utteranceId) { flushQueue(); }
        });
    }

    /** 打断当前朗读，立即播报新文字 */
    public void speak(String text) {
        if (text == null || text.isEmpty()) return;
        if (!mReady || mTts == null) { mQueue.offer(text); return; }
        String id = "tts_" + System.currentTimeMillis();
        if (Build.VERSION.SDK_INT >= 21) {
            mTts.speak(text, TextToSpeech.QUEUE_FLUSH, null, id);
        } else {
            mTts.speak(text, TextToSpeech.QUEUE_FLUSH, null);
        }
    }

    /** 排队追加朗读 */
    public void speakQueued(String text) {
        if (text == null || text.isEmpty()) return;
        if (!mReady || mTts == null) { mQueue.offer(text); return; }
        String id = "tts_" + System.currentTimeMillis();
        if (Build.VERSION.SDK_INT >= 21) {
            mTts.speak(text, TextToSpeech.QUEUE_ADD, null, id);
        } else {
            mTts.speak(text, TextToSpeech.QUEUE_ADD, null);
        }
    }

    private void flushQueue() {
        if (!mQueue.isEmpty()) {
            String text = mQueue.poll();
            speak(text);
        }
    }

    public void stop() { if (mTts != null) mTts.stop(); mQueue.clear(); }
    public void shutdown() { stop(); if (mTts != null) { mTts.shutdown(); mTts = null; } }
    public boolean isReady() { return mReady; }
}
```

# 九、MessageHandler.java — 消息类型处理

```java
package com.example.wechattts;

import de.robv.android.xposed.XposedBridge;

/**
 * 消息类型:
 *   1  = 文本  3  = 图片  34 = 语音
 *   43 = 视频  48 = 位置  49 = AppMsg
 */
public class MessageHandler {

    private final TtsEngine mTts;
    private final FilterManager mFilter;
    private final NicknameResolver mNick;

    public MessageHandler(TtsEngine tts, FilterManager filter, NicknameResolver nick) {
        this.mTts = tts; this.mFilter = filter; this.mNick = nick;
    }

    public void handleMessage(String talker, int msgType, String content, Object msgInfo) {
        String displayName = mNick.resolveDisplayName(talker);
        boolean isGroup = talker != null && talker.endsWith("@chatroom");

        switch (msgType) {
            case 1:  handleText(displayName, content, isGroup); break;
            case 3:  handleImage(displayName, isGroup); break;
            case 34: handleVoice(displayName, isGroup); break;
            case 43: handleVideo(displayName, isGroup); break;
            case 48: handleLocation(displayName, content, isGroup); break;
            case 49: handleAppMsg(displayName, content, isGroup); break;
        }
    }

    // --- 文字 ---
    private void handleText(String name, String content, boolean isGroup) {
        String text = cleanText(content);
        if (text.isEmpty()) return;
        // 跳过红包/转账系统消息
        if (text.contains("微信红包") || text.contains("领取红包") || text.contains("已被领取"))
            return;

        String speak = isGroup ? groupPrefix(null) + name + "说：" + text
                               : name + "说：" + text;
        mTts.speak(speak);
    }

    // --- 语音 ---
    private void handleVoice(String name, boolean isGroup) {
        String speak = isGroup ? groupPrefix(null) + name + "发来语音，请在手机上收听"
                               : name + "发来语音，请在手机上收听";
        mTts.speak(speak);
    }

    // --- 图片 ---
    private void handleImage(String name, boolean isGroup) {
        String speak = isGroup ? groupPrefix(null) + name + "发来一张照片"
                               : name + "发来一张照片";
        mTts.speak(speak);
    }

    // --- 视频 ---
    private void handleVideo(String name, boolean isGroup) {
        String speak = isGroup ? groupPrefix(null) + name + "发来一段视频"
                               : name + "发来一段视频";
        mTts.speak(speak);
    }

    // --- 位置 ---
    private void handleLocation(String name, String content, boolean isGroup) {
        String loc = parseLocation(content);
        String speak = isGroup ? groupPrefix(null) + name + "发来定位在：" + loc
                               : name + "发来定位在：" + loc;
        mTts.speak(speak);
    }

    // --- AppMsg (红包/转账) ---
    private void handleAppMsg(String name, String content, boolean isGroup) {
        if (content == null) return;
        if (content.contains("luckymoney") || content.contains("lucky money")) {
            String speak = isGroup ? groupPrefix(null) + name + "发来一个红包"
                                   : name + "发来一个红包";
            mTts.speak(speak);
        }
    }

    // ==================== 工具 ====================

    static String cleanText(String content) {
        if (content == null) return "";
        String t = content
            .replace("<![CDATA[", "").replace("]]>", "")
            .replaceAll("<[^>]+>", "")
            .replaceAll("https?://\\S+", "链接")
            .replaceAll("@\\S+\\s+", "")
            .replaceAll("\\[\\w+\\]", "")
            .replace("\n", " ").trim();
        return t.length() > 300 ? t.substring(0, 300) + "等长内容" : t;
    }

    static String parseLocation(String content) {
        if (content == null) return "未知位置";
        // 提取 label="xxx"
        int i = content.indexOf("label=\"");
        if (i >= 0) {
            int s = i + 7, e = content.indexOf("\"", s);
            if (e > s) return content.substring(s, e);
        }
        // 提取 poiname="xxx"
        i = content.indexOf("poiname=\"");
        if (i >= 0) {
            int s = i + 9, e = content.indexOf("\"", s);
            if (e > s) return content.substring(s, e);
        }
        return "未知位置";
    }

    private String groupPrefix(String talker) { return "群聊的"; }
}
```

# 十、RedPacketHandler.java — 红包金额播报

```java
package com.example.wechattts;

import java.lang.reflect.Field;
import de.robv.android.xposed.XposedBridge;

/**
 * 红包消息 AppMsg XML 结构:
 *   <wcpayinfo>
 *     <amount>88</amount>          <!-- 抢到金额(分) -->
 *     <totalamount>10000</totalamount>
 *     <sendname>张三</sendname>
 *     <wishing>恭喜发财</wishing>
 *   </wcpayinfo>
 *
 * 领取成功 resp (m1) 中包含 amount/sendname/wishing 等字段
 */
public class RedPacketHandler {

    private final TtsEngine mTts;
    private final FilterManager mFilter;
    private final NicknameResolver mNick;

    public RedPacketHandler(TtsEngine tts, FilterManager filter, NicknameResolver nick) {
        this.mTts = tts; this.mFilter = filter; this.mNick = nick;
    }

    public void handleReceiveResult(Object uiInstance, Object resp) {
        try {
            String amountStr = null;
            String sender = null;
            String wishing = null;

            // 1. 从 resp 反射提取
            if (resp != null) {
                String xml = reflectToString(resp);
                if (xml != null) {
                    amountStr = extractTag(xml, "amount");
                    if (amountStr == null) amountStr = extractTag(xml, "receiveamount");
                    if (amountStr == null) amountStr = extractTag(xml, "totalamount");
                    sender = extractTag(xml, "sendname");
                    wishing = extractTag(xml, "wishing");
                }
            }

            // 2. 从 UI 实例补充
            if (sender == null) sender = reflectField(uiInstance, "sendname", "sender", "sendername");
            if (wishing == null) wishing = reflectField(uiInstance, "wish", "desc", "greeting", "wishing");

            if (amountStr == null || amountStr.isEmpty()) return;

            String yuan = fenToYuan(amountStr);
            String name = sender != null ? sender : "好友";
            String wish = wishing != null ? wishing : "恭喜发财";

            String speak = name + "的红包：" + wish + "，你抢到了" + yuan + "元";
            mTts.speak(speak);

        } catch (Throwable t) {
            XposedBridge.log("[WeChatTTS] RedPktParseErr: " + t);
        }
    }

    /** 反射对象所有字段拼成字符串，用于 XML 解析 */
    private String reflectToString(Object obj) {
        if (obj == null) return null;
        try {
            StringBuilder sb = new StringBuilder();
            for (Field f : obj.getClass().getDeclaredFields()) {
                f.setAccessible(true);
                Object v = f.get(obj);
                if (v instanceof String) {
                    sb.append("<").append(f.getName()).append(">")
                      .append((String) v)
                      .append("</").append(f.getName()).append(">");
                }
            }
            return sb.length() > 0 ? sb.toString() : obj.toString();
        } catch (Exception e) { return obj.toString(); }
    }

    /** 反射查找字段值 */
    private String reflectField(Object obj, String... names) {
        if (obj == null) return null;
        for (String name : names) {
            try {
                for (Field f : obj.getClass().getDeclaredFields()) {
                    f.setAccessible(true);
                    if (f.getName().toLowerCase().contains(name.toLowerCase())) {
                        Object v = f.get(obj);
                        if (v instanceof String && !((String) v).isEmpty())
                            return (String) v;
                    }
                }
            } catch (Exception ignored) {}
        }
        return null;
    }

    /** XML 标签提取 */
    private static String extractTag(String xml, String tag) {
        int s = xml.indexOf("<" + tag + ">");
        if (s < 0) return null;
        s += tag.length() + 2;
        int e = xml.indexOf("</" + tag + ">", s);
        return e > s ? xml.substring(s, e) : null;
    }

    /** 分→元 */
    static String fenToYuan(String fen) {
        try {
            long f = Long.parseLong(fen.trim());
            return String.format("%.2f", f / 100.0);
        } catch (NumberFormatException e) { return fen; }
    }
}
```

```java
package com.example.wechattts;

import java.lang.reflect.Field;
import de.robv.android.xposed.XposedBridge;

/**
 * 转账 content XML 结构:
 *   <wcpaysign>
 *     <fee>5000</fee>             <!-- 金额(分) -->
 *     <pay_memo>备注文字</pay_memo>
 *     <payer_username>wxid_xxx</payer_username>
 *   </wcpaysign>
 */
public class TransferHandler {

    private final TtsEngine mTts;
    private final FilterManager mFilter;
    private final NicknameResolver mNick;

    public TransferHandler(TtsEngine tts, FilterManager filter, NicknameResolver nick) {
        this.mTts = tts; this.mFilter = filter; this.mNick = nick;
    }

    public void handleReceiveResult(Object uiInstance, Object resp) {
        try {
            String feeStr = null;
            String sender = null;
            String desc = null;

            if (resp != null) {
                String xml = reflectToString(resp);
                if (xml != null) {
                    feeStr = extractTag(xml, "fee");
                    sender = extractTag(xml, "payer_username");
                    desc = extractTag(xml, "pay_memo");
                }
            }

            if (sender == null) sender = reflectField(uiInstance, "payer", "sender", "fromuser");
            if (desc == null) desc = reflectField(uiInstance, "desc", "memo", "remark");

            if (feeStr == null || feeStr.isEmpty()) return;

            String yuan = RedPacketHandler.fenToYuan(feeStr);
            String name = sender != null ? mNick.resolveDisplayName(sender) : "好友";

            String speak = "收到" + name + "转账" + yuan + "元";
            if (desc != null && !desc.isEmpty()) speak += "，备注：" + desc;
            mTts.speak(speak);

        } catch (Throwable t) {
            XposedBridge.log("[WeChatTTS] TransferParseErr: " + t);
        }
    }

    private String reflectToString(Object obj) {
        if (obj == null) return null;
        try {
            StringBuilder sb = new StringBuilder();
            for (Field f : obj.getClass().getDeclaredFields()) {
                f.setAccessible(true);
                Object v = f.get(obj);
                if (v instanceof String) sb.append("<").append(f.getName()).append(">")
                    .append((String) v).append("</").append(f.getName()).append(">");
            }
            return sb.length() > 0 ? sb.toString() : obj.toString();
        } catch (Exception e) { return obj.toString(); }
    }

    private String reflectField(Object obj, String... names) {
        if (obj == null) return null;
        for (String name : names) {
            try {
                for (Field f : obj.getClass().getDeclaredFields()) {
                    f.setAccessible(true);
                    if (f.getName().toLowerCase().contains(name.toLowerCase())) {
                        Object v = f.get(obj);
                        if (v instanceof String && !((String) v).isEmpty())
                            return (String) v;
                    }
                }
            } catch (Exception ignored) {}
        }
        return null;
    }

    private static String extractTag(String xml, String tag) {
        int s = xml.indexOf("<" + tag + ">");
        if (s < 0) return null;
        s += tag.length() + 2;
        int e = xml.indexOf("</" + tag + ">", s);
        return e > s ? xml.substring(s, e) : null;
    }
}
```

# 十二、NicknameResolver.java — 昵称解析

```java
package com.example.wechattts;

import java.lang.reflect.Field;
import java.lang.reflect.Method;
import java.util.HashMap;
import java.util.Map;
import de.robv.android.xposed.XposedBridge;

/**
 * 微信联系人存储: qh3.o1
 * 联系人字段: field_conRemark(备注) > field_nickname(昵称)
 */
public class NicknameResolver {

    private Object mContactStg;
    private final Map<String, String> mCache = new HashMap<>();

    public NicknameResolver() {
        try {
            Class<?> j1Cls = Class.forName("j1");
            Method sMethod = j1Cls.getMethod("s", Class.class);
            Class<?> stgCls = Class.forName("qh3.o1");
            mContactStg = sMethod.invoke(null, stgCls);
        } catch (Throwable t) {
            XposedBridge.log("[WeChatTTS] NickResolver init fail: " + t);
        }
    }

    /** 解析显示名: 备注 > 昵称 > 默认 */
    public String resolveDisplayName(String talker) {
        if (talker == null || talker.isEmpty()) return "未知";
        String cached = mCache.get(talker);
        if (cached != null) return cached;

        String name = null;

        if (talker.endsWith("@chatroom")) {
            name = getContactName(talker); // 群也有昵称
        }
        if (name == null) {
            name = getContactName(talker);
        }
        if (name == null && talker.startsWith("gh_")) {
            name = "公众号";
        }
        if (name == null) {
            name = talker.length() > 20 ? talker.substring(0, 20) : talker;
        }

        mCache.put(talker, name);
        return name;
    }

    private String getContactName(String username) {
        try {
            if (mContactStg == null) return null;
            Method getMethod = mContactStg.getClass().getMethod("Lv", String.class);
            Object contact = getMethod.invoke(mContactStg, username);
            if (contact == null) return null;

            // 先取备注
            try {
                Field f = contact.getClass().getField("field_conRemark");
                String remark = (String) f.get(contact);
                if (remark != null && !remark.isEmpty()) return remark;
            } catch (NoSuchFieldException ignored) {}

            // 再取昵称
            try {
                Field f = contact.getClass().getField("field_nickname");
                String nick = (String) f.get(contact);
                if (nick != null && !nick.isEmpty()) return nick;
            } catch (NoSuchFieldException ignored) {}

        } catch (Throwable ignored) {}
        return null;
    }
}
```

# 十三、FilterManager.java — 过滤管理

```java
package com.example.wechattts;

import android.content.SharedPreferences;
import java.util.Calendar;
import java.util.HashSet;
import java.util.Set;

public class FilterManager {

    private final SharedPreferences mPrefs;

    public FilterManager(SharedPreferences prefs) {
        this.mPrefs = prefs;
        initDefaults();
    }

    private void initDefaults() {
        if (mPrefs.contains("tts_enabled")) return;
        mPrefs.edit()
            .putBoolean("tts_enabled", true)
            .putBoolean("tts_group_enabled", false)
            .putBoolean("tts_text", true)
            .putBoolean("tts_voice", true)
            .putBoolean("tts_image", true)
            .putBoolean("tts_video", true)
            .putBoolean("tts_location", true)
            .putBoolean("tts_redpkt", true)
            .putBoolean("tts_transfer", true)
            .putInt("tts_start_hour", 8)
            .putInt("tts_end_hour", 23)
            .putString("tts_list_mode", "none")
            .putStringSet("tts_list_data", new HashSet<>())
            .apply();
    }

    /** 主入口 */
    public boolean shouldProcess(String talker, int msgType) {
        if (!mPrefs.getBoolean("tts_enabled", true)) return false;
        if (!isInTimeRange()) return false;

        boolean isGroup = talker != null && talker.endsWith("@chatroom");
        if (isGroup && !mPrefs.getBoolean("tts_group_enabled", false)) return false;

        // 白名单/黑名单
        String mode = mPrefs.getString("tts_list_mode", "none");
        Set<String> list = mPrefs.getStringSet("tts_list_data", new HashSet<>());
        if (list == null) list = new HashSet<>();
        if ("whitelist".equals(mode) && !list.contains(talker)) return false;
        if ("blacklist".equals(mode) && list.contains(talker)) return false;

        // 消息类型
        switch (msgType) {
            case 1:  return mPrefs.getBoolean("tts_text", true);
            case 34: return mPrefs.getBoolean("tts_voice", true);
            case 3:  return mPrefs.getBoolean("tts_image", true);
            case 43: return mPrefs.getBoolean("tts_video", true);
            case 48: return mPrefs.getBoolean("tts_location", true);
            default: return true;
        }
    }

    private boolean isInTimeRange() {
        int start = mPrefs.getInt("tts_start_hour", 8);
        int end = mPrefs.getInt("tts_end_hour", 23);
        int now = Calendar.getInstance().get(Calendar.HOUR_OF_DAY);
        if (start <= end) return now >= start && now < end;
        else return now >= start || now < end; // 跨天
    }

    // Getters
    public boolean isGroupEnabled() { return mPrefs.getBoolean("tts_group_enabled", false); }
    public boolean isTextEnabled() { return mPrefs.getBoolean("tts_text", true); }
    public boolean isVoiceEnabled() { return mPrefs.getBoolean("tts_voice", true); }
    public boolean isImageEnabled() { return mPrefs.getBoolean("tts_image", true); }
    public boolean isVideoEnabled() { return mPrefs.getBoolean("tts_video", true); }
    public boolean isLocationEnabled() { return mPrefs.getBoolean("tts_location", true); }
    public String getListMode() { return mPrefs.getString("tts_list_mode", "none"); }
    public Set<String> getListData() { return mPrefs.getStringSet("tts_list_data", new HashSet<>()); }
    public int getStartHour() { return mPrefs.getInt("tts_start_hour", 8); }
    public int getEndHour() { return mPrefs.getInt("tts_end_hour", 23); }
}
```

```java
package com.example.wechattts;

import android.app.AlertDialog;

    @Override

            b.setText(names[i]);

        listContainer = new LinearLayout(this);

        listContainer.setPadding(0, 10, 0, 0);

                    if (list == null) list = new HashSet<>();

            Button del = new Button(this);

            del.setOnClickListener(v -> {
                Set<String> u = new HashSet<>(
                    mPrefs.getStringSet("tts_list_data", new HashSet<>()));
                if (u == null) u = new HashSet<>();
                u.remove(fi);
                mPrefs.edit().putStringSet("tts_list_data", u).apply();
                refreshList();
            });
            row.addView(del);
            listContainer.addView(row);
        }
    }
}
```
